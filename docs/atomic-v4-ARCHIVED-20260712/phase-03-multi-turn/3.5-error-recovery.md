# Unit 3.5 — Error Recovery: Slave Crash + Circuit Breaker Mid-Conversation

**Phase:** 3 | **ID:** 3.5 | **Dependencies:** 3.1

## Problem

If Chrome crashes during a conversation, the user gets an unhandled error. The system should detect the crash, respawn the slave, navigate back to the provider, and retry the send.

## Spec

### Conversation-Level Retry Wrapper

```typescript
// ConversationManager.send() with retry:

async send(conversationId: string, message: string): Promise<SendResult> {
  const MAX_RETRIES = 2
  
  for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
    try {
      return await this.sendInternal(conversationId, message)
    } catch (err) {
      if (attempt === MAX_RETRIES) {
        return {
          ok: false,
          messageId: '',
          blocks: [],
          text: '',
          latencyMs: 0,
          error: `Failed after ${MAX_RETRIES + 1} attempts: ${err}`,
        }
      }
      
      // Check if error is recoverable
      if (this.isRecoverableError(err)) {
        await this.recoverSlave(conversationId)
        await this.delay(1000 * (attempt + 1)) // exponential backoff
        continue
      }
      
      // Non-recoverable — return immediately
      return {
        ok: false,
        messageId: '',
        blocks: [],
        text: '',
        latencyMs: 0,
        error: String(err),
      }
    }
  }
}

private isRecoverableError(err: unknown): boolean {
  const msg = err instanceof Error ? err.message : String(err)
  return msg.includes('Slave not running') ||
         msg.includes('Circuit breaker') ||
         msg.includes('WebSocket connection failed') ||
         msg.includes('CDP command failed')
}

private async recoverSlave(conversationId: string): Promise<void> {
  const conv = await this.store.getConversation(conversationId)
  const account = await this.store.getAccount(conv.providerSessionId)
  
  // Kill any zombie slaves
  const slaves = this.governor.getAllSlaves({ providerId: conv.providerId })
  for (const slave of slaves) {
    if (slave.accountId === account.id) {
      try { await this.governor.kill(slave.slaveId) } catch {}
    }
  }
  
  // Spawn fresh slave with same profile
  await this.governor.spawn(conv.providerId, account.id, { visible: false })
  
  // Navigate to provider URL
  const url = PROVIDER_URLS[conv.providerId]
  const slave = this.governor.getAllSlaves({ providerId: conv.providerId })
    .find(s => s.accountId === account.id)
  
  if (slave) {
    await this.governor.cdp.send(slave.slaveId, 'Page.navigate', { url })
    await this.waitForPageLoad(slave.slaveId, 15_000)
  }
}
```

### Circuit Breaker Recovery

When the circuit breaker opens for a slave, the health monitor should attempt recovery:

```typescript
// In FleetSupervisor.ensureRunning():
if (instance.status === 'crashed' || instance.status === 'error') {
  const cb = this.getCircuit(instanceId)
  
  if (cb.state === 'open') {
    // Check if reset window has elapsed
    if (cb.openedAt && Date.now() - cb.openedAt >= this.opts.circuitBreakerResetMs) {
      cb.state = 'half_open'
      // Try one more time
      return this.spawn(instance.providerSlug, instance.accountId, {
        visible: false,
        debugPort: instance.debugPort,
      })
    }
    throw new CircuitOpenError(instanceId)
  }
  
  if (instance.consecutiveFailures >= this.opts.maxRestarts) {
    throw new CircuitOpenError(instanceId)
  }
  
  return this.spawn(instance.providerSlug, instance.accountId, {
    visible: false,
    debugPort: instance.debugPort,
  })
}
```

test('recovers from slave crash mid-conversation', async () => {
  // First send works

  await governor.kill(slave.slaveId)
