# Unit 1.5 — CDP-Based Login State Verification

**Phase:** 1 | **ID:** 1.5 | **Dependencies:** 1.4

## Problem

The `verify` endpoint in `setup-router.ts:89` checks if the current URL contains "login", "auth", "signin", or "sign-in" patterns. This is fragile — some providers redirect to auth subdomains or use unconventional URL patterns. A more robust approach checks for provider-specific login indicators in the DOM.

## Spec

### Verification Logic

```typescript
async function verifyLogin(
  cdp: BunCdpClient,
  providerId: string
): Promise<{ loggedIn: boolean; method: string; url: string }> {
  // 1. Get current page URL
  const targets = await cdp.send('Target.getTargets')
  const pageTarget = targets.targetInfos?.find(t => t.type === 'page')
  const url = pageTarget?.url ?? ''

  // 2. Check DOM indicators (most reliable)
  const indicator = LOGIN_INDICATORS[providerId]
  if (indicator?.loggedInSelector) {
    const result = await cdp.send('Runtime.evaluate', {
      expression: `(() => {
        const loggedIn = document.querySelector('${indicator.loggedInSelector}')
        const loggedOut = ${indicator.loggedOutSelector ? `document.querySelector('${indicator.loggedOutSelector}')` : 'null'}
        return JSON.stringify({ loggedIn: !!loggedIn, loggedOut: !!loggedOut })
      })()`,
      returnByValue: true,
    })
    const state = JSON.parse(result?.result?.value ?? '{}')
    if (state.loggedIn) return { loggedIn: true, method: 'dom_check', url }
    if (state.loggedOut) return { loggedIn: false, method: 'dom_check', url }
  }

  // 3. Fall back to URL pattern
  if (indicator?.urlPattern) {
    return { loggedIn: !indicator.urlPattern.test(url), method: 'url_pattern', url }
  }

  // 4. Generic fallback
  return { loggedIn: !/login|auth|signin|sign-in/i.test(url), method: 'url_pattern', url }
}
```

### Frontend Enhancement

The wizard should poll verification every 3 seconds while the user is on the login step, so the user doesn't have to manually click "Verify":

```typescript
// In provider-setup-wizard.tsx, step === 'login'
useEffect(() => {
  if (step !== 'login' || !launchResult) return
  const interval = setInterval(async () => {
    const resp = await fetch('/api/setup/verify', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ port: launchResult.debugPort, providerId: selectedProvider }),
    })
    const v = await resp.json()
    if (v.loggedIn) {
      clearInterval(interval)
      // Auto-complete
      handleVerifyComplete()
    }
  }, 3000)
  return () => clearInterval(interval)
}, [step, launchResult])
```

test('login verification detects logged-in state', async () => {
  // Mock CDP client
  const mockCdp = {
    send: async (method: string, params?: any) => {
      if (method === 'Target.getTargets') {
        return { targetInfos: [{ type: 'page', url: 'https://chatgpt.com/' }] }
      }
      if (method === 'Runtime.evaluate') {
        return { result: { value: '{"loggedIn": true, "loggedOut": false}' } }
      }
    }
  }
  
  const result = await verifyLogin(mockCdp as any, 'chatgpt')
  expect(result.loggedIn).toBe(true)
  expect(result.method).toBe('dom_check')
})
