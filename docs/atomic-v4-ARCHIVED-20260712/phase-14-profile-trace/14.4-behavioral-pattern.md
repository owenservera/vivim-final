# Unit 14.4 — BehavioralPatternEngine: Request Timing + Interaction Rhythm

**Phase:** 14 | **ID:** 14.4 | **Dependencies:** 11.2

## Problem

Even with perfect fingerprint spoofing, detection systems analyze **behavioral timing**:
- Bots send requests with metronomic regularity (exactly 5.000s between requests)
- Bots never pause to "read" the response
- Bots click immediately after page load (humans take 2-5s to scan)
- Bots type at constant speed with no bursts or pauses
- Bots don't mouse-move between actions

This module adds **realistic timing jitter** and **think-time pauses** between VIVIM operations.

## Spec

```typescript
// src/engines/stealth/behavioral-pattern-engine.ts

export class BehavioralPatternModule implements StealthModule {
  name = 'behavioral_pattern'
  detectionVector = 'Behavioral timing analysis (request rhythm, interaction cadence, think-time)'
  description = 'Adds human-like delays between operations: reading pauses, think-time, mouse wandering'
  priority = 25 // runs last — wraps around all other interactions

  configSchema = z.object({
    thinkTimeMinMs: z.number().default(1500),     // min pause after page load
    thinkTimeMaxMs: z.number().default(5000),     // max pause after page load
    betweenActionsMinMs: z.number().default(300), // min delay between sequential actions
    betweenActionsMaxMs: z.number().default(1500), // max delay between sequential actions
    readingPauseProbability: z.number().min(0).max(1).default(0.3),
    readingPauseMs: z.number().default(2000),
    mouseWanderProbability: z.number().min(0).max(1).default(0.4),
    jitterMs: z.number().default(200),           // random jitter added to all delays
    pageLoadSettleMs: z.number().default(1000),  // wait for page to settle after load
  })

  async apply(config: Record<string, unknown>, ctx: StealthContext): Promise<void> {
    // This module doesn't inject persistent JS
    // Instead it provides helper methods that add delays
    // It's called by ConversationManager between operations
  }

  // ── Timing Helpers ──────────────────────────────────────

  /**
   * Wait after page load before interacting (human scan time).
   */
  async waitForThinkTime(): Promise<void> {
    const cfg = this.config
    const thinkTime = this.randomBetween(cfg.thinkTimeMinMs, cfg.thinkTimeMaxMs)
    const jitter = this.randomBetween(0, cfg.jitterMs)
    await Bun.sleep(thinkTime + jitter)
  }

  /**
   * Wait between sequential actions (human reaction time).
   */
  async betweenActions(): Promise<void> {
    const cfg = this.config
    const delay = this.randomBetween(cfg.betweenActionsMinMs, cfg.betweenActionsMaxMs)
    const jitter = this.randomBetween(0, cfg.jitterMs)
    await Bun.sleep(delay + jitter)

    // Maybe add a "reading pause" (human reads response before next action)
    if (Math.random() < cfg.readingPauseProbability) {
      await Bun.sleep(cfg.readingPauseMs + this.randomBetween(0, cfg.jitterMs))
    }
  }

  /**
   * Wait for page to settle after navigation/load.
   */
  async waitForPageSettle(): Promise<void> {
    await Bun.sleep(this.config.pageLoadSettleMs + this.randomBetween(0, this.config.jitterMs))
  }

  /**
   * Optionally move mouse to a random position (human idle behavior).
   */
  async maybeMouseWander(ctx: StealthContext, slaveId: string): Promise<void> {
    if (Math.random() > this.config.mouseWanderProbability) return

    // Move mouse to a random visible position
    const x = this.randomBetween(100, 1000)
    const y = this.randomBetween(100, 700)

    await ctx.cdp.send(slaveId, 'Input.dispatchMouseEvent', {
      type: 'mouseMoved', x, y,
    })

    await Bun.sleep(this.randomBetween(200, 800))
  }

  /**
   * Random scroll during think time (human scans page).
   */
  async maybeIdleScroll(ctx: StealthContext, slaveId: string): Promise<void> {
    if (Math.random() > 0.3) return // 30% chance

    const direction = Math.random() > 0.5 ? 'down' : 'up'
    const amount = this.randomBetween(50, 300)

    await ctx.cdp.send(slaveId, 'Runtime.evaluate', {
      expression: `window.scrollBy(0, ${direction === 'down' ? amount : -amount})`,
    })

    await Bun.sleep(this.randomBetween(500, 2000))
  }

  private randomBetween(min: number, max: number): number {
    return min + Math.random() * (max - min)
  }
}
```

```typescript
test('betweenActions adds realistic delay', async () => {
  const start = Date.now()
  await module.betweenActions()
  const elapsed = Date.now() - start

  // Should be between min and max + jitter
  expect(elapsed).toBeGreaterThan(300)
  expect(elapsed).toBeLessThan(2000)
})

test('delays are non-deterministic', async () => {
  const delays: number[] = []
  for (let i = 0; i < 10; i++) {
    const start = Date.now()
    await module.betweenActions()
    delays.push(Date.now() - start)
  }

  // Should have variance (not all the same)
  const unique = new Set(delays)
  expect(unique.size).toBeGreaterThan(5)
})

test('mouseWander is probabilistic', async () => {
  let mouseMoved = false
  mockCdp.on('Input.dispatchMouseEvent', () => { mouseMoved = true })

  // Run 20 times — should sometimes move, sometimes not
  let moveCount = 0
  for (let i = 0; i < 20; i++) {
    mouseMoved = false
    await module.maybeMouseWander(ctx, slaveId)
    if (mouseMoved) moveCount++
  }

  // With 40% probability, expect roughly 4-12 out of 20
  expect(moveCount).toBeGreaterThan(2)
  expect(moveCount).toBeLessThan(16)
})
```
