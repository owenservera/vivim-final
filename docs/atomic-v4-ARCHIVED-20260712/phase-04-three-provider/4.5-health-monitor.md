# Unit 4.5 — Provider Health Monitoring for All Three Providers

**Phase:** 4 | **ID:** 4.5 | **Dependencies:** 4.1, 4.2, 4.3

## Problem

The ProviderHealthKernel computes health scores but isn't started in the server bootstrap. The frontend has no way to see provider health status.

## Spec

### Start Health Kernel in Bootstrap

```typescript
// In createServerWithEngines():
import { ProviderHealthKernel } from '../engines/provider-health.js'
import { HealthStoreImpl } from '../storage/impl/health-store-impl.js'

const healthStore = new HealthStoreImpl(db)
const healthKernel = new ProviderHealthKernel({
  governor,
  store: healthStore,
  eventBus,
  intervalMs: 30_000,
})
healthKernel.start()

onShutdown(async () => {
  healthKernel.stop()
})
```

### Health Dashboard Endpoint

```typescript
// GET /api/health/providers
if (pathname === '/api/health/providers' && method === 'GET') {
  const healthMap = healthKernel.getAllHealth()
  const result = []
  for (const [providerId, health] of healthMap) {
    result.push({
      providerId,
      status: health.status,
      score: health.score,
      signals: health.signals,
      fleet: health.fleet,
      parsers: health.parsers,
      capabilities: health.capabilities,
    })
  }
  return json(result)
}
```

### Frontend Health Display

```tsx
// web/sandbox/src/features/health-dashboard.tsx

export function HealthDashboard() {
  const [health, setHealth] = useState<ProviderHealth[]>([])
  
  useEffect(() => {
    const fetchHealth = () => {
      fetch('/api/health/providers')
        .then(r => r.json())
        .then(setHealth)
    }
    fetchHealth()
    const interval = setInterval(fetchHealth, 10_000)
    return () => clearInterval(interval)
  }, [])

  return (
    <div className="p-3 border-t">
      <h3 className="text-sm font-semibold mb-2">Provider Health</h3>
      {health.map(h => (
        <div key={h.providerId} className="flex items-center gap-2 mb-1">
          <HealthBadge status={h.status} />
          <span className="text-sm">{h.providerId}</span>
          <span className="text-xs text-gray-400">{h.score}/100</span>
        </div>
      ))}
    </div>
  )
}

function HealthBadge({ status }: { status: string }) {
  const colors = {
    healthy: 'bg-green-500',
    degraded: 'bg-yellow-500',
    unhealthy: 'bg-red-500',
    unknown: 'bg-gray-400',
  }
  return <span className={`w-2 h-2 rounded-full ${colors[status] || colors.unknown}`} />
}
```
