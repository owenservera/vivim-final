### Claude Parser Fix

Claude uses a different SSE format with event types:

```
event: message_start
data: {"type":"message_start","message":{...}}

event: content_block_start
data: {"type":"content_block_start","index":0,"content_block":{"type":"text","text":""}}

event: content_block_delta
data: {"type":"content_block_delta","index":0,"delta":{"type":"text_delta","text":"Hello"}}

event: content_block_stop
data: {"type":"content_block_stop","index":0}

event: message_stop
data: {"type":"message_stop"}
```

The parser must:
1. Skip non-delta events
2. Accumulate text from `content_block_delta` events
3. Detect completion on `message_stop`

```javascript
// Claude parser inline code
var parse = function(rawBody) {
  var blocks = [];
  var index = 0;
  var lines = rawBody.split('\n');
  var currentText = '';
  
  for (var i = 0; i < lines.length; i++) {
    var line = lines[i].trim();
    if (!line.startsWith('data:')) continue;
    var payload = line.slice(5).trim();
    
    try {
      var json = JSON.parse(payload);
      if (json.type === 'content_block_delta' && json.delta && json.delta.text) {
        currentText += json.delta.text;
      }
      if (json.type === 'message_stop') {
        if (currentText) {
          blocks.push({ kind: 'text', content: currentText, index: index++ });
          currentText = '';
        }
      }
    } catch (e) { }
  }
  
  if (currentText && blocks.length === 0) {
    blocks.push({ kind: 'text', content: currentText, index: 0 });
  }
  
  return blocks;
};
```

### Gemini Parser Fix

Gemini returns a non-standard format prefixed with `)]}'`:

```
)]}'

[["wrb.fr","...","[[\"Hello\",[\"response text\"]]]"]]
```

The parser must:
1. Strip the XSSI prefix `)]}'`
2. Parse the nested JSON arrays
3. Extract text from the deeply nested structure

```javascript
// Gemini parser inline code
var parse = function(rawBody) {
  var blocks = [];
  var index = 0;
  
  // Strip XSSI prefix
  var jsonStr = rawBody.replace(/^\)\]\}'\s*\n?/, '').trim();
  
  try {
    var data = JSON.parse(jsonStr);
    // Gemini response structure: [[["wrb.fr", null, "[[...]]"]]]
    // The actual text is deeply nested
    for (var i = 0; i < data.length; i++) {
      var item = data[i];
      if (!Array.isArray(item)) continue;
      for (var j = 0; j < item.length; j++) {
        var inner = item[j];
        if (typeof inner === 'string' && inner.startsWith('[[')) {
          try {
            var nested = JSON.parse(inner);
            // Extract text from nested structure
            var text = extractTextFromGeminiNested(nested);
            if (text) {
              blocks.push({ kind: 'text', content: text, index: index++ });
            }
          } catch (e) { }
        }
      }
    }
  } catch (e) {
    // If JSON parse fails, return raw as text
    if (rawBody.trim().length > 0) {
      blocks.push({ kind: 'text', content: rawBody, index: 0 });
    }
  }
  
  return blocks;
};

function extractTextFromGeminiNested(arr) {
  // Recursively search for text strings in the nested array
  var texts = [];
  function search(obj) {
    if (typeof obj === 'string' && obj.length > 5) {
      texts.push(obj);
    } else if (Array.isArray(obj)) {
      for (var i = 0; i < obj.length; i++) search(obj[i]);
    }
  }
  search(arr);
  return texts.join(' ');
}
```
