# Unit 8.3 — Request Queueing + Backpressure with Policy

**Phase:** 8 | **ID:** 8.3 | **Dependencies:** 7.3, 8.1

## Problem

Under heavy load (many concurrent conversation sends, capability executions), the system can overwhelm Chrome and the provider. There's no queueing mechanism — all requests compete for resources simultaneously.

## Re-Programmable Design

A **BackpressurePolicy** controls how requests are queued and throttled:

```typescript
interface BackpressurePolicy {
  maxConcurrent: number              // max simultaneous in-flight requests
  maxQueued: number                  // max waiting in queue
  queueTimeoutMs: number            // how long to wait in queue before rejecting
  priorityLevels: number            // number of priority levels (1 = flat)
  fairnessStrategy: 'fifo' | 'priority' | 'weighted_fair'
  shedStrategy: 'reject_newest' | 'reject_oldest' | 'reject_lowest_priority'
  perProviderMaxConcurrent: number   // per-provider limit
  onShed: 'reject' | 'return_stale' | 'return_error'
}
```

### Request Queue

```typescript
// New file: src/engines/request-queue.ts

interface QueuedRequest<T> {
  id: string
  priority: number
  enqueuedAt: number
  resolve: (value: T) => void
  reject: (error: Error) => void
  execute: () => Promise<T>
  providerId?: string
}

export class RequestQueue<T = unknown> {
  private queue: QueuedRequest<T>[] = []
  private inFlight = 0
  private perProviderInFlight = new Map<string, number>()
  private policy: BackpressurePolicy

  constructor(
    private policyStore: PolicyStore,
    private policyKey: string,
  ) {
    void this.loadPolicy()
  }

  private async loadPolicy(): Promise<void> {
    this.policy = await this.policyStore.get<BackpressurePolicy>(this.policyKey) ?? {
      maxConcurrent: 10,
      maxQueued: 50,
      queueTimeoutMs: 30_000,
      priorityLevels: 3,
      fairnessStrategy: 'priority',
      shedStrategy: 'reject_newest',
      perProviderMaxConcurrent: 5,
      onShed: 'reject',
    }
  }

  async submit(
    execute: () => Promise<T>,
    opts?: { priority?: number; providerId?: string },
  ): Promise<T> {
    const priority = opts?.priority ?? Math.floor(this.policy.priorityLevels / 2)

    // Check queue capacity
    if (this.queue.length >= this.policy.maxQueued) {
      this.shed()
    }

    // Check timeout
    return new Promise<T>((resolve, reject) => {
      const request: QueuedRequest<T> = {
        id: crypto.randomUUID(),
        priority,
        enqueuedAt: Date.now(),
        resolve,
        reject,
        execute,
        providerId: opts?.providerId,
      }

      const timer = setTimeout(() => {
        this.removeFromQueue(request.id)
        reject(new EngineError(`Request timed out in queue after ${this.policy.queueTimeoutMs}ms`))
      }, this.policy.queueTimeoutMs)

      // Wrap resolve/reject to clear timer
      const origResolve = resolve
      const origReject = reject
      request.resolve = (v: T) => { clearTimeout(timer); origResolve(v) }
      request.reject = (e: Error) => { clearTimeout(timer); origReject(e) }

      this.queue.push(request)
      this.processQueue()
    })
  }

  private processQueue(): void {
    while (this.canDispatch()) {
      const next = this.selectNext()
      if (!next) break
      this.dispatch(next)
    }
  }

  private canDispatch(): boolean {
    if (this.queue.length === 0) return false
    if (this.inFlight >= this.policy.maxConcurrent) return false
    return true
  }

  private selectNext(): QueuedRequest<T> | null {
    if (this.queue.length === 0) return null

    switch (this.policy.fairnessStrategy) {
      case 'fifo':
        return this.queue.shift() ?? null

      case 'priority':
        // Highest priority first
        this.queue.sort((a, b) => b.priority - a.priority)
        return this.queue.shift() ?? null

      case 'weighted_fair':
        // Weighted fair queuing — track virtual finish time
        // Simplified: proportional to priority weight
        this.queue.sort((a, b) => {
          const aWeight = a.enqueuedAt / (a.priority + 1)
          const bWeight = b.enqueuedAt / (b.priority + 1)
          return aWeight - bWeight
        })
        return this.queue.shift() ?? null

      default:
        return this.queue.shift() ?? null
    }
  }

  private async dispatch(request: QueuedRequest<T>): Promise<void> {
    this.inFlight++
    if (request.providerId) {
      const count = this.perProviderInFlight.get(request.providerId) ?? 0
      this.perProviderInFlight.set(request.providerId, count + 1)
    }

    try {
      const result = await request.execute()
      request.resolve(result)
    } catch (err) {
      request.reject(err instanceof Error ? err : new Error(String(err)))
    } finally {
      this.inFlight--
      if (request.providerId) {
        const count = this.perProviderInFlight.get(request.providerId) ?? 1
        this.perProviderInFlight.set(request.providerId, count - 1)
      }
      this.processQueue()
    }
  }

  private shed(): void {
    switch (this.policy.shedStrategy) {
      case 'reject_newest':
        throw new EngineError('Request rejected: queue full')
      case 'reject_oldest': {
        const oldest = this.queue.shift()
        oldest?.reject(new EngineError('Request shed: newer requests prioritized'))
        break
      }
      case 'reject_lowest_priority': {
        const idx = this.queue.reduce((minIdx, req, idx, arr) =>
          req.priority < (arr[minIdx]?.priority ?? 0) ? idx : minIdx, 0)
        const shed = this.queue.splice(idx, 1)[0]
        shed?.reject(new EngineError('Request shed: low priority'))
        break
      }
    }
  }

  getStats() {
    return {
      queueLength: this.queue.length,
      inFlight: this.inFlight,
      perProvider: Object.fromEntries(this.perProviderInFlight),
    }
  }
}
```

```typescript
test('queue throttles concurrent requests', async () => {
  await policyStore.set('backpressure.conversation', {
    maxConcurrent: 2,
    maxQueued: 10,
    ...defaults,
  })

  // Submit 5 requests
  const promises = Array.from({ length: 5 }, () =>
    queue.submit(async () => { await delay(100); return 'ok' })
  )

  // Only 2 should be in-flight at any time
  await Promise.all(promises)
  // All should succeed
  expect(promises.every(p => p.then(r => r === 'ok'))).toBeTruthy()
})

test('queue rejects when full', async () => {
  await policyStore.set('backpressure.conversation', {
    maxConcurrent: 1,
    maxQueued: 2,
    onShed: 'reject',
    ...defaults,
  })

  // Fill queue
  const slow = queue.submit(async () => { await delay(1000); return 'slow' })
  queue.submit(async () => 'queued1')
  queue.submit(async () => 'queued2')

  // This should be rejected
  await expect(queue.submit(async () => 'rejected')).rejects.toThrow('queue full')

  await slow
})
```
