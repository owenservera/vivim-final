# Unit 13.2 — HumanKeyboardEngine: Variable Rhythm Typing

**Phase:** 13 | **ID:** 13.2 | **Dependencies:** 11.2

## Problem

Current typing (`el.value = text` + single input event) inserts the entire text in one frame. Human detection flags this as instant text appearance with zero keystrokes. Real humans type character-by-character with variable delays following a log-normal distribution.

## Spec

```typescript
// src/engines/stealth/human-keyboard-engine.ts

export class HumanKeyboardModule implements StealthModule {
  name = 'human_keyboard'
  detectionVector = 'Typing rhythm analysis (keystroke timing, burst patterns)'
  description = 'Types text character-by-character with log-normal delay distribution and occasional bursts'
  priority = 21
  configSchema = z.object({
    minDelayMs: z.number().default(50),
    maxDelayMs: z.number().default(180),
    medianDelayMs: z.number().default(90),
    burstProbability: z.number().min(0).max(1).default(0.12),  // chance of fast burst
    burstLength: z.number().default(3),
    typoProbability: z.number().min(0).max(0.05).default(0.005), // occasional typo + correction
  })

  async apply(_config: Record<string, unknown>, ctx: StealthContext): Promise<void> {
    // This module doesn't inject a persistent script — it modifies how
    // the typing module dispatches keystrokes.
    // It's used as a helper by ConversationManager/composer-typing.
    //
    // The actual typing rhythm is implemented in the humanType() method below.
  }

  /**
   * Type text with human-like rhythm.
   * Called by ConversationManager instead of el.value = text.
   */
  async humanType(
    ctx: StealthContext,
    slaveId: string,
    selector: string,
    text: string,
  ): Promise<void> {
    const cfg = this.config

    // Focus the element
    await ctx.cdp.send(slaveId, 'DOM.querySelector', { selector })
    await ctx.cdp.send(slaveId, 'Input.focus', { selector })

    // Clear if needed
    await ctx.cdp.send(slaveId, 'Input.dispatchKeyEvent', {
      type: 'keyDown', key: 'a', code: 'KeyA', modifiers: ['Control'],
    })
    await ctx.cdp.send(slaveId, 'Input.dispatchKeyEvent', {
      type: 'keyUp', key: 'a', code: 'KeyA', modifiers: ['Control'],
    })
    await ctx.cdp.send(slaveId, 'Input.dispatchKeyEvent', {
      type: 'keyDown', key: 'Backspace', code: 'Backspace',
    })
    await ctx.cdp.send(slaveId, 'Input.dispatchKeyEvent', {
      type: 'keyUp', key: 'Backspace', code: 'Backspace',
    })

    let inBurst = false
    let burstRemaining = 0

    for (const char of text) {
      // Determine delay
      let delayMs: number

      if (inBurst && burstRemaining > 0) {
        // Fast burst typing
        delayMs = cfg.minDelayMs * 0.4
        burstRemaining--
        if (burstRemaining === 0) inBurst = false
      } else {
        // Log-normal distribution (human-like)
        // median = exp(mu), so mu = ln(median)
        const mu = Math.log(cfg.medianDelayMs)
        const sigma = 0.4 // shape parameter
        const z = this.boxMuller()
        delayMs = Math.exp(mu + sigma * z)
        delayMs = Math.max(cfg.minDelayMs, Math.min(cfg.maxDelayMs, delayMs))

        // Maybe start a burst
        if (Math.random() < cfg.burstProbability) {
          inBurst = true
          burstRemaining = cfg.burstLength
        }
      }

      // Occasionally typo and correct (very rare)
      if (Math.random() < cfg.typoProbability) {
        const wrongChar = String.fromCharCode(char.charCodeAt(0) + (Math.random() > 0.5 ? 1 : -1))
        await this.dispatchChar(ctx.cdp, slaveId, wrongChar)
        await Bun.sleep(delayMs)
        await ctx.cdp.send(slaveId, 'Input.dispatchKeyEvent', {
          type: 'keyDown', key: 'Backspace', code: 'Backspace',
        })
        await ctx.cdp.send(slaveId, 'Input.dispatchKeyEvent', {
          type: 'keyUp', key: 'Backspace', code: 'Backspace',
        })
        await Bun.sleep(delayMs * 0.5)
      }

      // Type the actual character
      await this.dispatchChar(ctx.cdp, slaveId, char)
      await Bun.sleep(delayMs)
    }

    // Dispatch input event to trigger React/framework handlers
    await ctx.cdp.send(slaveId, 'Runtime.evaluate', {
      expression: `document.querySelector(${JSON.stringify(selector)})?.dispatchEvent(new Event('input', {bubbles: true}))`,
    })
  }

  private async dispatchChar(cdp: CDPProxy, slaveId: string, char: string): Promise<void> {
    await cdp.send(slaveId, 'Input.dispatchKeyEvent', {
      type: 'keyDown',
      key: char,
      code: this.charToCode(char),
      text: char,
    })
    await cdp.send(slaveId, 'Input.dispatchKeyEvent', {
      type: 'keyUp',
      key: char,
      code: this.charToCode(char),
    })
  }

  private charToCode(char: string): string {
    if (char === ' ') return 'Space'
    if (char === '\n') return 'Enter'
    const upper = char.toUpperCase()
    if (/^[A-Z]$/.test(upper)) return `Key${upper}`
    if (/^[0-9]$/.test(char)) return `Digit${char}`
    return 'Unidentified'
  }

  private boxMuller(): number {
    const u1 = Math.random()
    const u2 = Math.random()
    return Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2)
  }
}
```

## Test

```typescript
test('typing takes realistic time', async () => {
  const start = Date.now()
  await module.humanType(ctx, slaveId, '#prompt-textarea', 'Hello world')
  const elapsed = Date.now() - start

  // 11 chars × ~90ms median = ~1000ms minimum
  expect(elapsed).toBeGreaterThan(500)
  expect(elapsed).toBeLessThan(5000)
})

test('typing dispatches individual key events', async () => {
  let keyEventCount = 0
  mockCdp.on('Input.dispatchKeyEvent', () => keyEventCount++)

  await module.humanType(ctx, slaveId, '#prompt-textarea', 'Hi')

  // 2 chars × 2 events (down + up) = 4 minimum
  expect(keyEventCount).toBeGreaterThanOrEqual(4)
})
```
