# Unit 13.3 — HumanScrollEngine: Natural Scroll Velocity Curves

**Phase:** 13 | **ID:** 13.3 | **Dependencies:** 11.2

## Problem

Current scrolling: `window.scrollBy(0, 300)` — instant jump, zero acceleration. Human scrolling has a velocity curve: slow start → fast middle → slow stop. Detection scripts analyze scroll event timing patterns.

## Spec

```typescript
// src/engines/stealth/human-scroll-engine.ts

export class HumanScrollModule implements StealthModule {
  name = 'human_scroll'
  detectionVector = 'Scroll velocity analysis (acceleration patterns, event frequency)'
  description = 'Scrolls with human-like velocity curves (ease-in, plateau, ease-out)'
  priority = 22
  configSchema = z.object({
    minSteps: z.number().default(5),
    maxSteps: z.number().default(15),
    stepDelayMinMs: z.number().default(16),
    stepDelayMaxMs: z.number().default(50),
    overshootProbability: z.number().min(0).max(1).default(0.2),
    pauseAtEndMs: z.number().default(200),
  })

  async apply(_config: Record<string, unknown>, ctx: StealthContext): Promise<void> {
    // Like human_keyboard, this module provides a helper method
    // rather than injecting persistent JS
  }

  /**
   * Scroll with human-like velocity.
   */
  async humanScroll(
    ctx: StealthContext,
    slaveId: string,
    direction: 'up' | 'down' | 'left' | 'right',
    totalPixels: number,
  ): Promise<void> {
    const cfg = this.config

    // Determine number of scroll steps
    const steps = cfg.minSteps + Math.floor(Math.random() * (cfg.maxSteps - cfg.minSteps))

    // Maybe overshoot
    let actualTotal = totalPixels
    let correction = 0
    if (Math.random() < cfg.overshootProbability) {
      correction = Math.floor(totalPixels * 0.1 * (Math.random() > 0.5 ? 1 : -1))
      actualTotal = totalPixels + correction
    }

    // Generate velocity curve using easing function
    // We use a bell-curve velocity: slow→fast→slow
    const stepSizes = this.generateBellCurveSteps(actualTotal, steps)

    // Dispatch scroll events
    const dy = direction === 'down' ? 1 : direction === 'up' ? -1 : 0
    const dx = direction === 'right' ? 1 : direction === 'left' ? -1 : 0

    for (let i = 0; i < stepSizes.length; i++) {
      const stepSize = stepSizes[i]!

      await ctx.cdp.send(slaveId, 'Runtime.evaluate', {
        expression: `window.scrollBy(${dx * stepSize}, ${dy * stepSize})`,
      })

      // Variable delay (slightly random)
      const delay = cfg.stepDelayMinMs + Math.random() * (cfg.stepDelayMaxMs - cfg.stepDelayMinMs)
      await Bun.sleep(delay)
    }

    // If overshot, correct
    if (correction !== 0) {
      await Bun.sleep(cfg.pauseAtEndMs)
      const correctionSteps = this.generateBellCurveSteps(Math.abs(correction), 3)
      const corrDir = correction > 0 ? -1 : 1
      for (const stepSize of correctionSteps) {
        await ctx.cdp.send(slaveId, 'Runtime.evaluate', {
          expression: `window.scrollBy(${dx * stepSize * corrDir}, ${dy * stepSize * corrDir})`,
        })
        await Bun.sleep(cfg.stepDelayMinMs)
      }
    }
  }

  /**
   * Generate step sizes that follow a bell curve (ease-in + ease-out).
   * Total of all steps = totalPixels.
   */
  private generateBellCurveSteps(totalPixels: number, steps: number): number[] {
    // Use sine bell curve: sin(pi * i / (steps-1))
    const weights: number[] = []
    let weightSum = 0

    for (let i = 0; i < steps; i++) {
      const t = steps === 1 ? 0.5 : i / (steps - 1)
      const weight = Math.sin(Math.PI * t) // 0 at edges, 1 at middle
      weights.push(weight)
      weightSum += weight
    }

    // Normalize to total pixels
    return weights.map(w => Math.max(1, Math.round((w / weightSum) * totalPixels)))
  }

  /**
   * Scroll to a specific element with human-like movement.
   */
  async scrollToElement(
    ctx: StealthContext,
    slaveId: string,
    selector: string,
  ): Promise<void> {
    // Get element position
    const posResult = await ctx.cdp.send(slaveId, 'Runtime.evaluate', {
      expression: `(() => {
        const el = document.querySelector(${JSON.stringify(selector)});
        if (!el) return null;
        const r = el.getBoundingClientRect();
        return JSON.stringify({ top: r.top + window.scrollY, windowH: window.innerHeight });
      })()`,
      returnByValue: true,
    })

    const pos = JSON.parse((posResult as any)?.result?.value ?? 'null')
    if (!pos) throw new Error(`Element not found: ${selector}`)

    // Target: element center should be at window center
    const targetScroll = pos.top - pos.windowH / 2
    const currentScroll = await this.getCurrentScroll(ctx, slaveId)
    const delta = targetScroll - currentScroll

    if (Math.abs(delta) > 5) {
      const direction = delta > 0 ? 'down' : 'up'
      await this.humanScroll(ctx, slaveId, direction, Math.abs(delta))
    }
  }

  private async getCurrentScroll(ctx: StealthContext, slaveId: string): Promise<number> {
    const result = await ctx.cdp.send(slaveId, 'Runtime.evaluate', {
      expression: 'window.scrollY',
      returnByValue: true,
    })
    return (result as any)?.result?.value ?? 0
  }
}
```

```typescript
test('scroll uses multiple steps with bell curve', async () => {
  let scrollCallCount = 0
  mockCdp.on('Runtime.evaluate', (params) => {
    if (params.expression?.includes('scrollBy')) scrollCallCount++
  })

  await module.humanScroll(ctx, slaveId, 'down', 300)

  expect(scrollCallCount).toBeGreaterThan(3) // multiple steps
})

test('overshoot produces correction scroll', async () => {
  // Force overshoot
  module.config.overshootProbability = 1.0

  const scrollValues: number[] = []
  mockCdp.on('Runtime.evaluate', (params) => {
    const match = params.expression?.match(/scrollBy\(\d+,\s*(-?\d+)\)/)
    if (match) scrollValues.push(parseInt(match[1]))
  })

  await module.humanScroll(ctx, slaveId, 'down', 300)

  const totalScrolled = scrollValues.reduce((a, b) => a + b, 0)
  // If overshot and corrected, total should be ~300 (not exact due to rounding)
  expect(Math.abs(totalScrolled - 300)).toBeLessThan(50)
})
```
