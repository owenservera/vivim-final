# Unit 7.7 — Configurable Retry Policy Engine

**Phase:** 7 | **ID:** 7.7 | **Dependencies:** 2.1

## Problem

The ConversationManager catches errors but doesn't retry. Transient failures (network blip, Chrome hiccup, provider rate limit) cause permanent message loss.

## Re-Programmable Design

A **RetryPolicy** is a configurable, per-operation policy — not a hardcoded retry count:

```typescript
interface RetryPolicy {
  maxAttempts: number
  backoffStrategy: 'fixed' | 'linear' | 'exponential' | 'exponential_jitter'
  initialDelayMs: number
  maxDelayMs: number
  multiplier: number              // for exponential
  retryableErrors: string[]       // error patterns that warrant retry
  nonRetryableErrors: string[]    // errors that should NOT be retried
  onRetry?: 'log' | 'emit' | 'silent'
}

// Policies can be per-operation:
// 'conversation.send' → different policy than 'fleet.spawn'
```

### Retry Engine

```typescript
// New file: src/engines/retry-engine.ts

export class RetryEngine {
  private policies = new Map<string, RetryPolicy>()

  constructor(private policyStore: PolicyStore) {
    void this.loadPolicies()
  }

  private async loadPolicies(): Promise<void> {
    // Load all retry policies from DB
    const policies = await this.policyStore.getAllByPrefix('retry.')
    for (const p of policies) {
      this.policies.set(p.key, p.value as RetryPolicy)
    }
  }

  async execute<T>(
    operationKey: string,
    fn: () => Promise<T>,
    opts?: { policy?: Partial<RetryPolicy> },
  ): Promise<T> {
    const policy = { ...this.getPolicy(operationKey), ...opts?.policy }
    let lastError: Error | null = null

    for (let attempt = 1; attempt <= policy.maxAttempts; attempt++) {
      try {
        return await fn()
      } catch (err) {
        lastError = err instanceof Error ? err : new Error(String(err))

        // Check if retryable
        if (!this.isRetryable(lastError, policy)) {
          throw lastError
        }

        // Don't sleep on last attempt
        if (attempt < policy.maxAttempts) {
          const delay = this.computeDelay(attempt, policy)
          if (policy.onRetry !== 'silent') {
            console.log(`[retry] ${operationKey} attempt ${attempt}/${policy.maxAttempts} failed: ${lastError.message}. Retrying in ${delay}ms`)
          }
          await Bun.sleep(delay)
        }
      }
    }

    throw lastError!
  }

  private getPolicy(key: string): RetryPolicy {
    return this.policies.get(key) ?? {
      maxAttempts: 3,
      backoffStrategy: 'exponential_jitter',
      initialDelayMs: 1000,
      maxDelayMs: 30_000,
      multiplier: 2,
      retryableErrors: ['timeout', 'connection', 'ECONNREFUSED', 'Slave not running', 'Circuit breaker'],
      nonRetryableErrors: ['not found', 'invalid', 'auth'],
      onRetry: 'log',
    }
  }

  private isRetryable(error: Error, policy: RetryPolicy): boolean {
    const msg = error.message.toLowerCase()

    // Check non-retryable first
    for (const pattern of policy.nonRetryableErrors) {
      if (msg.includes(pattern.toLowerCase())) return false
    }

    // Check retryable
    for (const pattern of policy.retryableErrors) {
      if (msg.includes(pattern.toLowerCase())) return true
    }

    return false // default: don't retry unknown errors
  }

  private computeDelay(attempt: number, policy: RetryPolicy): number {
    let delay: number

    switch (policy.backoffStrategy) {
      case 'fixed':
        delay = policy.initialDelayMs
        break
      case 'linear':
        delay = policy.initialDelayMs * attempt
        break
      case 'exponential':
        delay = policy.initialDelayMs * Math.pow(policy.multiplier, attempt - 1)
        break
      case 'exponential_jitter':
      default:
        delay = policy.initialDelayMs * Math.pow(policy.multiplier, attempt - 1)
        delay += Math.random() * delay * 0.1 // ±10% jitter
        break
    }

    return Math.min(delay, policy.maxDelayMs)
  }
}
```

## Test

```typescript
test('retries on transient failure', async () => {
  let attempts = 0
  const result = await retryEngine.execute('test', async () => {
    attempts++
    if (attempts < 3) throw new Error('timeout')
    return 'success'
  }, { policy: { maxAttempts: 3, initialDelayMs: 10, retryableErrors: ['timeout'] } as any })

  expect(result).toBe('success')
  expect(attempts).toBe(3)
})

test('does not retry non-retryable errors', async () => {
  let attempts = 0
  await expect(retryEngine.execute('test', async () => {
    attempts++
    throw new Error('auth required')
  })).rejects.toThrow('auth required')

  expect(attempts).toBe(1)
})
```
