# Unit 7.4 — Double-Send Protection: Idempotency Keys

**Phase:** 7 | **ID:** 7.4 | **Dependencies:** 7.3

## Problem

Network retries, user double-clicks, and WebSocket reconnects can cause the same message to be sent multiple times. The provider receives duplicate messages.

## Re-Programmable Design

An **IdempotencyPolicy** controls how duplicate sends are detected and handled:

```typescript
interface IdempotencyPolicy {
  enabled: boolean
  windowMs: number             // dedup window — same message within this window is rejected
  keyStrategy: 'message_hash' | 'client_key' | 'message_hash+conversation'
  maxRecentKeys: number        // how many recent keys to keep in memory
  onDuplicate: 'reject' | 'return_cached' | 'allow'
}
```

### Implementation

```typescript
// New file: src/engines/idempotency-guard.ts

export class IdempotencyGuard {
  private recent = new Map<string, { result: SendResult; timestamp: number }>()
  private policy: IdempotencyPolicy

  constructor(private policyStore: PolicyStore) {
    this.policy = {
      enabled: true,
      windowMs: 10_000,
      keyStrategy: 'message_hash+conversation',
      maxRecentKeys: 1000,
      onDuplicate: 'return_cached',
    }
    void this.loadPolicy()
  }

  async check(conversationId: string, message: string): Promise<{
    duplicate: boolean
    cachedResult?: SendResult
    key: string
  }> {
    if (!this.policy.enabled) return { duplicate: false, key: '' }

    const key = this.computeKey(conversationId, message)
    const now = Date.now()

    // Prune old entries
    this.prune(now)

    const existing = this.recent.get(key)
    if (existing && now - existing.timestamp < this.policy.windowMs) {
      // Duplicate detected
      switch (this.policy.onDuplicate) {
        case 'reject':
          throw new EngineError('Duplicate message rejected by idempotency policy')
        case 'return_cached':
          return { duplicate: true, cachedResult: existing.result, key }
        case 'allow':
          return { duplicate: false, key }
      }
    }

    return { duplicate: false, key }
  }

  record(key: string, result: SendResult): void {
    if (!this.policy.enabled || !key) return
    this.recent.set(key, { result, timestamp: Date.now() })

    // Enforce max size (LRU eviction)
    if (this.recent.size > this.policy.maxRecentKeys) {
      const oldest = [...this.recent.entries()].sort((a, b) => a[1].timestamp - b[1].timestamp)[0]
      if (oldest) this.recent.delete(oldest[0])
    }
  }

  private computeKey(conversationId: string, message: string): string {
    switch (this.policy.keyStrategy) {
      case 'message_hash':
        return hash(message)
      case 'client_key':
        return ''  // uses client-provided key instead
      case 'message_hash+conversation':
      default:
        return hash(`${conversationId}:${message}`)
    }
  }

  private prune(now: number): void {
    const cutoff = now - this.policy.windowMs
    for (const [key, entry] of this.recent) {
      if (entry.timestamp < cutoff) {
        this.recent.delete(key)
      }
    }
  }
}

function hash(s: string): string {
  let h = 0
  for (let i = 0; i < s.length; i++) {
    h = ((h << 5) - h + s.charCodeAt(i)) | 0
  }
  return `id_${h}`
}
```

### Integration

```typescript
// ConversationManager.send():
async send(conversationId: string, message: string, clientKey?: string): Promise<SendResult> {
  // Idempotency check
  const idempCheck = await this.idempotency.check(conversationId, message)
  if (idempCheck.duplicate && idempCheck.cachedResult) {
    return idempCheck.cachedResult
  }

  const result = await this.sendInternal(conversationId, message)
  this.idempotency.record(idempCheck.key, result)
  return result
}
```
