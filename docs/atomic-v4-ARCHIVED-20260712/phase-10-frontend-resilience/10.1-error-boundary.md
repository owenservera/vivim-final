# Unit 10.1 — Error Boundary + Crash Recovery

**Phase:** 10 | **ID:** 10.1 | **Dependencies:** 5.1

## Problem

An unhandled error in any React component crashes the entire sandbox app. The user sees a white screen with no recovery path.

## Spec

### React Error Boundary

```tsx
// web/sandbox/src/components/error-boundary.tsx

import { Component, ErrorInfo, ReactNode } from 'react'

interface State {
  hasError: boolean
  error: Error | null
}

export class ErrorBoundary extends Component<{ children: ReactNode }, State> {
  state: State = { hasError: false, error: null }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error }
  }

  componentDidCatch(error: Error, info: ErrorInfo): void {
    // Report to error tracker
    console.error('[ErrorBoundary]', error, info.componentStack)

    // POST to backend for tracking
    fetch('/api/errors', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        name: error.name,
        message: error.message,
        stack: error.stack,
        componentStack: info.componentStack,
        url: window.location.href,
        userAgent: navigator.userAgent,
      }),
    }).catch(() => {})
  }

  handleReload = (): void => {
    window.location.reload()
  }

  handleDismiss = (): void => {
    this.setState({ hasError: false, error: null })
  }

  render(): ReactNode {
    if (this.state.hasError) {
      return (
        <div className="flex flex-col items-center justify-center h-screen p-8">
          <div className="max-w-md text-center">
            <div className="text-5xl mb-4">⚠️</div>
            <h1 className="text-xl font-semibold text-gray-900 mb-2">
              Something went wrong
            </h1>
            <p className="text-sm text-gray-500 mb-4">
              {this.state.error?.message || 'An unexpected error occurred.'}
            </p>
            {this.state.error?.stack && (
              <details className="text-left mb-4">
                <summary className="text-xs text-gray-400 cursor-pointer">Stack trace</summary>
                <pre className="text-xs bg-gray-100 p-2 rounded mt-2 overflow-auto max-h-48">
                  {this.state.error.stack}
                </pre>
              </details>
            )}
            <div className="flex gap-2 justify-center">
              <button
                onClick={this.handleDismiss}
                className="px-4 py-2 text-gray-600 border rounded"
              >
                Try Again
              </button>
              <button
                onClick={this.handleReload}
                className="px-4 py-2 bg-blue-600 text-white rounded"
              >
                Reload App
              </button>
            </div>
          </div>
        </div>
      )
    }

    return this.props.children
  }
}
```

### Per-Component Boundaries

For critical areas, use nested error boundaries to isolate failures:

```tsx
<ErrorBoundary>
  <SandboxApp>
    <ConversationList />     // separate boundary
    <ErrorBoundary>
      <ConversationSurface /> // if this crashes, list still works
    </ErrorBoundary>
    <ErrorBoundary>
      <DebugPanel />         // if this crashes, chat still works
    </ErrorBoundary>
  </SandboxApp>
</ErrorBoundary>
```

### WebSocket Reconnection

```tsx
// Auto-reconnect on WebSocket disconnect
function useReconnectingWebSocket(url: string, onMessage: (data: string) => void) {
  const wsRef = useRef<WebSocket | null>(null)
  const reconnectDelay = useRef(1000)

  useEffect(() => {
    const connect = () => {
      const ws = new WebSocket(url)
      wsRef.current = ws

      ws.onopen = () => {
        reconnectDelay.current = 1000 // reset on success
      }

      ws.onmessage = (ev) => onMessage(ev.data as string)

      ws.onclose = () => {
        // Exponential backoff reconnect
        setTimeout(() => {
          reconnectDelay.current = Math.min(reconnectDelay.current * 2, 30000)
          connect()
        }, reconnectDelay.current)
      }

      ws.onerror = () => {
        // Will trigger onclose
      }
    }

    connect()

    return () => {
      wsRef.current?.close()
    }
  }, [url])
}
```
