# Unit 10.3 — Keyboard Shortcuts + Command Palette

**Phase:** 10 | **ID:** 10.3 | **Dependencies:** 6.1

## Problem

Power users can't navigate or trigger actions without mouse clicks. No keyboard shortcuts, no command palette.

## Re-Programmable Design

Shortcuts are **registered dynamically** via the ActionRegistry — not hardcoded. Any registered action can have a keyboard shortcut:

```typescript
interface ShortcutBinding {
  actionId: string        // from ActionRegistry
  keys: string            // e.g. 'Ctrl+Enter', 'Cmd+K', 'Ctrl+Shift+N'
  scope: string           // 'global' | 'conversation' | 'composer'
  when?: (ctx: ShortcutContext) => boolean  // conditional visibility
}
```

### Shortcut Manager

```tsx
// web/sandbox/src/hooks/use-shortcuts.ts

import { useEffect } from 'react'
import { ActionRegistry } from '@ui/actions/registry.js'

const shortcuts = new Map<string, ShortcutBinding>()

export function registerShortcut(binding: ShortcutBinding): void {
  shortcuts.set(binding.keys.toLowerCase(), binding)
}

export function useKeyboardShortcuts(context: ShortcutContext) {
  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      // Build key string
      const parts: string[] = []
      if (e.ctrlKey) parts.push('Ctrl')
      if (e.metaKey) parts.push('Cmd')
      if (e.shiftKey) parts.push('Shift')
      if (e.altKey) parts.push('Alt')
      parts.push(e.key)

      const keyStr = parts.join('+').toLowerCase()

      const binding = shortcuts.get(keyStr)
      if (!binding) return

      // Check scope
      if (binding.scope !== 'global' && binding.scope !== context.scope) return

      // Check condition
      if (binding.when && !binding.when(context)) return

      // Prevent default
      e.preventDefault()
      e.stopPropagation()

      // Dispatch action
      ActionRegistry.dispatch(binding.actionId, context.params ?? {}).catch(console.error)
    }

    window.addEventListener('keydown', handler, true)
    return () => window.removeEventListener('keydown', handler, true)
  }, [context])
}
```

### Command Palette

```tsx
// web/sandbox/src/features/command-palette.tsx

export function CommandPalette({ open, onClose }: { open: boolean; onClose: () => void }) {
  const [query, setQuery] = useState('')
  const [selectedIndex, setSelectedIndex] = useState(0)
  const inputRef = useRef<HTMLInputElement>(null)

  // Get all registered actions as commands
  const commands = useMemo(() => {
    const all = ActionRegistry.listWithMetadata()
    const filtered = query
      ? all.filter(cmd =>
          cmd.id.toLowerCase().includes(query.toLowerCase()) ||
          cmd.description.toLowerCase().includes(query.toLowerCase())
        )
      : all
    return filtered
  }, [query])

  // Keyboard navigation within palette
  const handleKeyDown = (e: React.KeyboardEvent) => {
    switch (e.key) {
      case 'ArrowDown':
        e.preventDefault()
        setSelectedIndex(i => Math.min(i + 1, commands.length - 1))
        break
      case 'ArrowUp':
        e.preventDefault()
        setSelectedIndex(i => Math.max(i - 1, 0))
        break
      case 'Enter':
        e.preventDefault()
        if (commands[selectedIndex]) {
          ActionRegistry.dispatch(commands[selectedIndex].id, {}).catch(console.error)
          onClose()
        }
        break
      case 'Escape':
        e.preventDefault()
        onClose()
        break
    }
  }

  if (!open) return null

  return (
    <div className="fixed inset-0 bg-black/30 flex items-start justify-center pt-32 z-50" onClick={onClose}>
      <div
        className="bg-white rounded-lg shadow-2xl w-full max-w-lg overflow-hidden"
        onClick={e => e.stopPropagation()}
      >
        <input
          ref={inputRef}
          type="text"
          value={query}
          onChange={e => { setQuery(e.target.value); setSelectedIndex(0) }}
          onKeyDown={handleKeyDown}
          placeholder="Type a command..."
          className="w-full px-4 py-3 text-base border-b outline-none"
          autoFocus
        />
        <div className="max-h-64 overflow-y-auto">
          {commands.map((cmd, i) => (
            <button
              key={cmd.id}
              onClick={() => {
                ActionRegistry.dispatch(cmd.id, {}).catch(console.error)
                onClose()
              }}
              onMouseEnter={() => setSelectedIndex(i)}
              className={`w-full text-left px-4 py-2 flex items-center justify-between ${
                i === selectedIndex ? 'bg-blue-50' : 'hover:bg-gray-50'
              }`}
            >
              <div>
                <div className="text-sm font-medium">{cmd.id}</div>
                <div className="text-xs text-gray-500">{cmd.description}</div>
              </div>
              {shortcutMap[cmd.id] && (
                <kbd className="text-xs bg-gray-100 px-2 py-1 rounded">
                  {shortcutMap[cmd.id]}
                </kbd>
              )}
            </button>
          ))}
          {commands.length === 0 && (
            <div className="px-4 py-8 text-center text-gray-400 text-sm">
              No commands found
            </div>
          )}
        </div>
      </div>
    </div>
  )
}
```

### Integration

```tsx
function SandboxApp() {
  const [paletteOpen, setPaletteOpen] = useState(false)
  const shortcutContext = useMemo(() => ({
    scope: 'global',
    composerFocused: false,
    composerHasText: false,
    params: {},
  }), [])

  useKeyboardShortcuts({ ...shortcutContext, openPalette: () => setPaletteOpen(true) })

  return (
    <>
      {/* Main app */}
      <CommandPalette open={paletteOpen} onClose={() => setPaletteOpen(false)} />
    </>
  )
}
```
