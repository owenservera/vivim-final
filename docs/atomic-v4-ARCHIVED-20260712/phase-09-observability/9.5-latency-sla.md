# Unit 9.5 — Latency SLA Monitoring + Alerting

**Phase:** 9 | **ID:** 9.5 | **Dependencies:** 9.2

## Problem

The MirrorEngine has latency budget constants but they're unused. No SLA monitoring means degrading performance isn't detected until users complain.

## Re-Programmable Design

An **SlaPolicy** defines per-operation latency targets and alert thresholds:

```typescript
interface SlaPolicy {
  operations: SlaOperation[]
  evaluationWindowMs: number     // evaluate over rolling window
  alertThreshold: number         // % of requests exceeding target to trigger alert
  cooldownMs: number             // don't re-alert within this window
}

interface SlaOperation {
  name: string                   // 'conversation.send'
  target: SlaTarget
}

interface SlaTarget {
  p50Ms: number                  // 50th percentile should be under this
  p95Ms: number                  // 95th percentile should be under this
  p99Ms: number                  // 99th percentile should be under this
}
```

### SLA Monitor

```typescript
// New file: src/engines/sla-monitor.ts

export class SlaMonitor {
  private samples = new Map<string, number[]>()  // operation → latencies
  private alertCooldowns = new Map<string, number>()
  private policy: SlaPolicy
  private exporters: MetricsExporter[]

  constructor(
    policy: SlaPolicy,
    private metrics: MetricsRegistry,
    private logger: StructuredLogger,
    private eventBus: CapabilityEventBus,
  ) {
    this.policy = policy
    // Periodic evaluation
    setInterval(() => void this.evaluate(), policy.evaluationWindowMs / 2)
  }

  record(operation: string, latencyMs: number): void {
    let samples = this.samples.get(operation)
    if (!samples) {
      samples = []
      this.samples.set(operation, samples)
    }
    samples.push(latencyMs)

    // Trim to window size
    const maxSamples = 10000
    if (samples.length > maxSamples) {
      samples.splice(0, samples.length - maxSamples)
    }

    // Record to metrics
    this.metrics.observe(`sla_${operation}_latency_ms`, latencyMs)
  }

  private async evaluate(): Promise<void> {
    const now = Date.now()

    for (const op of this.policy.operations) {
      const samples = this.samples.get(op.name)
      if (!samples || samples.length < 10) continue  // need minimum samples

      const percentiles = this.computePercentiles(samples, [50, 95, 99])
      const target = op.target

      // Check SLA violations
      const violations: string[] = []
      if (percentiles[50] > target.p50Ms) violations.push(`p50: ${percentiles[50]}ms > ${target.p50Ms}ms`)
      if (percentiles[95] > target.p95Ms) violations.push(`p95: ${percentiles[95]}ms > ${target.p95Ms}ms`)
      if (percentiles[99] > target.p99Ms) violations.push(`p99: ${percentiles[99]}ms > ${target.p99Ms}ms`)

      // Update gauges
      this.metrics.setGauge(`sla_${op.name}_p50_ms`, percentiles[50])
      this.metrics.setGauge(`sla_${op.name}_p95_ms`, percentiles[95])
      this.metrics.setGauge(`sla_${op.name}_p99_ms`, percentiles[99])

      if (violations.length > 0) {
        // Check cooldown
        const lastAlert = this.alertCooldowns.get(op.name) ?? 0
        if (now - lastAlert > this.policy.cooldownMs) {
          this.logger.warn(`SLA violation: ${op.name}`, {
            operation: op.name,
            violations,
            percentiles,
            target,
          })

          this.eventBus.emit({
            type: 'sla:violation',
            operation: op.name,
            violations,
            percentiles,
            target,
          })

          this.alertCooldowns.set(op.name, now)
        }
      }
    }
  }

  private computePercentiles(samples: number[], percentiles: number[]): Record<number, number> {
    const sorted = [...samples].sort((a, b) => a - b)
    const result: Record<number, number> = {}
    for (const p of percentiles) {
      const idx = Math.ceil(sorted.length * p / 100) - 1
      result[p] = sorted[Math.max(0, idx)] ?? 0
    }
    return result
  }

  getReport(): SlaReport {
    const report: SlaReport = {}
    for (const op of this.policy.operations) {
      const samples = this.samples.get(op.name) ?? []
      if (samples.length === 0) continue
      const percentiles = this.computePercentiles(samples, [50, 95, 99])
      report[op.name] = {
        sampleCount: samples.length,
        percentiles,
        target: op.target,
        meeting: percentiles[95] <= op.target.p95Ms,
      }
    }
    return report
  }
}
```

## Test

```typescript
test('SLA violation triggers alert', () => {
  // Record slow operations
  for (let i = 0; i < 100; i++) {
    slaMonitor.record('conversation.send', 20_000)  // above p95 target
  }

  // Trigger evaluation
  await slaMonitor.evaluate()

  // Should have emitted violation
  expect(eventBusEmitted).toHaveBeenCalledWith(expect.objectContaining({
    type: 'sla:violation',
    operation: 'conversation.send',
  }))
})
```
