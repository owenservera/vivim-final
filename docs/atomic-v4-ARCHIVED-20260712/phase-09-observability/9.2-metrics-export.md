# Unit 9.2 — Metrics Export Pipeline (Prometheus/OTLP)

**Phase:** 9 | **ID:** 9.2 | **Dependencies:** 9.1

## Problem

No system metrics are collected or exported. You can't monitor conversation latency, slave count, error rates, or throughput without manual instrumentation.

## Re-Programmable Design

A **MetricsExporter** plugin system (same pattern as LogTransport):

```typescript
interface MetricsExporter {
  name: string
  export(metrics: MetricSnapshot[]): Promise<void>
}

interface MetricSnapshot {
  name: string
  type: 'counter' | 'gauge' | 'histogram' | 'summary'
  value: number
  labels: Record<string, string>
  ts: number
}

interface MetricsPolicy {
  enabled: boolean
  exporters: ExporterConfig[]
  defaultHistogramBuckets: number[]
  scrapeIntervalMs: number
  prefix: string  // e.g. 'vivim_'
}
```

### Metrics Registry

```typescript
// New file: src/engines/metrics.ts

export class MetricsRegistry {
  private counters = new Map<string, MetricEntry>()
  private gauges = new Map<string, MetricEntry>()
  private histograms = new Map<string, HistogramEntry>()
  private exporters: MetricsExporter[] = []

  constructor(
    private policy: MetricsPolicy,
    private logger: StructuredLogger,
  ) {
    this.loadExporters()
  }

  // ── Counter ──────────────────────────────────────────

  increment(name: string, value = 1, labels?: Record<string, string>): void {
    const key = this.key(name, labels)
    const entry = this.counters.get(key) ?? { name, value: 0, labels: labels ?? {} }
    entry.value += value
    this.counters.set(key, entry)
  }

  // ── Gauge ────────────────────────────────────────────

  setGauge(name: string, value: number, labels?: Record<string, string>): void {
    const key = this.key(name, labels)
    this.gauges.set(key, { name, value, labels: labels ?? {} })
  }

  // ── Histogram ────────────────────────────────────────

  observe(name: string, value: number, labels?: Record<string, string>): void {
    const key = this.key(name, labels)
    let entry = this.histograms.get(key)
    if (!entry) {
      entry = {
        name,
        labels: labels ?? {},
        buckets: this.policy.defaultHistogramBuckets,
        counts: new Array(this.policy.defaultHistogramBuckets.length + 1).fill(0),
        sum: 0,
        count: 0,
      }
      this.histograms.set(key, entry)
    }
    entry.sum += value
    entry.count++
    for (let i = 0; i < entry.buckets.length; i++) {
      if (value <= entry.buckets[i]!) {
        entry.counts[i]!++
        break
      }
    }
    if (value > entry.buckets[entry.buckets.length - 1]!) {
      entry.counts[entry.counts.length - 1]!++
    }
  }

  // ── Timer helper ─────────────────────────────────────

  startTimer(name: string, labels?: Record<string, string>): () => void {
    const start = Date.now()
    return () => {
      this.observe(name, Date.now() - start, labels)
    }
  }

  // ── Export ───────────────────────────────────────────

  snapshot(): MetricSnapshot[] {
    const snapshots: MetricSnapshot[] = []
    const ts = Date.now()

    for (const entry of this.counters.values()) {
      snapshots.push({ ...entry, type: 'counter', ts })
    }
    for (const entry of this.gauges.values()) {
      snapshots.push({ ...entry, type: 'gauge', ts })
    }
    for (const entry of this.histograms.values()) {
      snapshots.push({ ...entry, type: 'histogram', ts })
    }

    return snapshots
  }

  async export(): Promise<void> {
    if (!this.policy.enabled) return
    const snapshots = this.snapshot()
    for (const exporter of this.exporters) {
      try {
        await exporter.export(snapshots)
      } catch (err) {
        this.logger.error('Metrics export failed', err instanceof Error ? err : undefined, { exporter: exporter.name })
      }
    }
  }

  // ── Core VIVIM Metrics (auto-instrumented) ───────────

  /** Called from ConversationManager */
  recordConversationSend(providerId: string, durationMs: number, ok: boolean): void {
    this.increment(`${this.policy.prefix}conversation_sends_total`, 1, {
      provider: providerId,
      status: ok ? 'success' : 'failure',
    })
    this.observe(`${this.policy.prefix}conversation_send_duration_ms`, durationMs, {
      provider: providerId,
    })
  }

  /** Called from FleetSupervisor */
  recordSlaveCount(providerId: string, count: number): void {
    this.setGauge(`${this.policy.prefix}fleet_slaves_active`, count, { provider: providerId })
  }

  /** Called from ChromeGovernor */
  recordCdpCommand(method: string, durationMs: number, ok: boolean): void {
    this.increment(`${this.policy.prefix}cdp_commands_total`, 1, { method, status: ok ? 'ok' : 'error' })
    this.observe(`${this.policy.prefix}cdp_command_duration_ms`, durationMs, { method })
  }
}
```

### Built-in Exporters

```typescript
class PrometheusExporter implements MetricsExporter {
  name = 'prometheus'
  private buffer: MetricSnapshot[] = []

  constructor(private config: { url?: string; pushIntervalMs?: number }) {}

  async export(metrics: MetricSnapshot[]): Promise<void> {
    if (this.config.url) {
      // Push gateway mode
      await this.pushToGateway(metrics)
    } else {
      // Buffer for /metrics scrape
      this.buffer = metrics
    }
  }

  /** Render Prometheus text format for /metrics endpoint */
  render(): string {
    const lines: string[] = []
    for (const m of this.buffer) {
      const labelStr = Object.entries(m.labels)
        .map(([k, v]) => `${k}="${v}"`)
        .join(',')
      const fullName = `vivim_${m.name}`
      lines.push(`${fullName}{${labelStr}} ${m.value}`)
    }
    return lines.join('\n')
  }
}
```

```typescript
test('counter increments', () => {
  registry.increment('test_counter', 1, { label: 'a' })
  registry.increment('test_counter', 2, { label: 'a' })
  const snap = registry.snapshot()
  const counter = snap.find(s => s.name === 'test_counter' && s.labels.label === 'a')
  expect(counter?.value).toBe(3)
})

test('histogram observes values', () => {
  const stop = registry.startTimer('test_duration')
  await delay(100)
  stop()
  const snap = registry.snapshot()
  const hist = snap.find(s => s.name === 'test_duration')
  expect(hist).toBeDefined()
  expect(hist.count).toBe(1)
})
```
