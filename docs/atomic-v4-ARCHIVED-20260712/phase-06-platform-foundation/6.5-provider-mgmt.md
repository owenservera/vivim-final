# Unit 6.5 — Provider Management UI: Add/Remove/Switch

**Phase:** 6 | **ID:** 6.5 | **Dependencies:** 6.1

## Problem

Users need to manage their provider accounts from the UI — add new providers, remove old ones, and switch between accounts.

## Spec

### Provider Management Panel

```tsx
// web/sandbox/src/features/provider-manager.tsx

export function ProviderManager({ onClose }: { onClose?: () => void }) {
  const [profiles, setProfiles] = useState<Profile[]>([])
  const [showAddWizard, setShowAddWizard] = useState(false)
  
  useEffect(() => {
    loadProfiles()
  }, [])
  
  const loadProfiles = async () => {
    const resp = await fetch('/api/setup/profiles')
    const data = await resp.json()
    setProfiles(data.profiles ?? [])
  }
  
  const handleRemove = async (providerId: string, accountSlug: string) => {
    // TODO: DELETE /api/setup/profiles/:id endpoint
    // For now, just remove from local state
    setProfiles(prev => prev.filter(p => !(p.providerId === providerId && p.accountSlug === accountSlug)))
  }
  
  return (
    <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50">
      <div className="bg-white rounded-lg shadow-xl w-full max-w-lg p-6">
        <div className="flex justify-between items-center mb-4">
          <h2 className="text-xl font-semibold">Provider Accounts</h2>
          <button onClick={onClose} className="text-gray-400 hover:text-gray-600">✕</button>
        </div>
        
        {/* Profile list */}
        <div className="space-y-2 mb-4">
          {profiles.map(p => (
            <div key={`${p.providerId}-${p.accountSlug}`} className="flex items-center justify-between p-3 border rounded">
              <div className="flex items-center gap-3">
                <ProviderIcon providerId={p.providerId} />
                <div>
                  <div className="font-medium">{p.providerId}</div>
                  <div className="text-xs text-gray-500">{p.accountSlug}</div>
                </div>
              </div>
              <div className="flex items-center gap-2">
                <LoginBadge state={p.loginState} />
                <button
                  onClick={() => handleRemove(p.providerId, p.accountSlug)}
                  className="text-red-500 text-sm"
                >
                  Remove
                </button>
              </div>
            </div>
          ))}
          {profiles.length === 0 && (
            <p className="text-gray-500 text-sm">No providers configured.</p>
          )}
        </div>
        
        {/* Add button */}
        <button
          onClick={() => setShowAddWizard(true)}
          className="w-full px-4 py-2 bg-blue-600 text-white rounded hover:bg-blue-700"
        >
          + Add Provider
        </button>
        
        {showAddWizard && (
          <ProviderSetupWizard onComplete={() => { setShowAddWizard(false); loadProfiles() }} />
        )}
      </div>
    </div>
  )
}
```
