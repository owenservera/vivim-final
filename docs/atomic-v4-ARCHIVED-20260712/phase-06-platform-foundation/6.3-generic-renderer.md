# Unit 6.3 — Capability UI: Generic Contract-Driven Renderer

**Phase:** 6 | **ID:** 6.3 | **Dependencies:** 6.1

## Problem

Each capability needs a UI component. Currently only the GenericCapabilityRenderer exists in the sandbox. It needs to map `uiComponent` types from the capability resolution to React components.

## Spec

### Component Mapping

```tsx
// web/sandbox/src/features/generic-capability-renderer.tsx

import { ComponentType } from 'react'

interface CapabilityContract {
  slug: string
  name: string
  uiComponent: string
  uiLabel: string
  uiIcon: string
  uiPosition: string
  uiInputSchema: Record<string, unknown>
  interactionMode: string
  uiStates: string[]
}

// Registry of bespoke renderers (override generic)
const bespokeRenderers = new Map<string, ComponentType<{ contract: CapabilityContract }>>()

export function registerBespokeRenderer(slug: string, component: ComponentType<{ contract: CapabilityContract }>) {
  bespokeRenderers.set(slug, component)
}

export function GenericCapabilityRenderer({ contract }: { contract: CapabilityContract }) {
  // Check for bespoke renderer
  const Bespoke = bespokeRenderers.get(contract.slug)
  if (Bespoke) return <Bespoke contract={contract} />

  // Map uiComponent to generic component
  switch (contract.uiComponent) {
    case 'button':
      return <CapabilityButton contract={contract} />
    case 'toggle':
      return <CapabilityToggle contract={contract} />
    case 'select':
      return <CapabilitySelect contract={contract} />
    case 'input':
      return <CapabilityInput contract={contract} />
    case 'modal':
      return <CapabilityModal contract={contract} />
    case 'panel':
      return <CapabilityPanel contract={contract} />
    case 'inline':
      return <CapabilityInline contract={contract} />
    default:
      return <CapabilityFallback contract={contract} />
  }
}
```

### Generic Components

```tsx
function CapabilityButton({ contract }: { contract: CapabilityContract }) {
  const [executing, setExecuting] = useState(false)
  
  const handleClick = async () => {
    setExecuting(true)
    try {
      await ActionRegistry.dispatch('capability.execute', {
        conversationId: getCurrentConversationId(),
        slug: contract.slug,
      })
    } finally {
      setExecuting(false)
    }
  }
  
  return (
    <button
      onClick={handleClick}
      disabled={executing}
      className="px-3 py-1.5 text-sm border rounded hover:bg-gray-50 disabled:opacity-50"
    >
      {contract.uiIcon && <span className="mr-1">{contract.uiIcon}</span>}
      {contract.uiLabel || contract.name}
      {executing && <span className="ml-1 animate-spin">⏳</span>}
    </button>
  )
}

function CapabilityToggle({ contract }: { contract: CapabilityContract }) {
  const [on, setOn] = useState(false)
  
  const handleToggle = async () => {
    const newState = !on
    setOn(newState) // optimistic
    try {
      await ActionRegistry.dispatch('capability.execute', {
        conversationId: getCurrentConversationId(),
        slug: contract.slug,
        params: { enabled: newState },
      })
    } catch {
      setOn(!newState) // revert
    }
  }
  
  return (
    <label className="flex items-center gap-2 cursor-pointer">
      <span className="text-sm">{contract.uiLabel}</span>
      <button
        onClick={handleToggle}
        className={`w-10 h-5 rounded-full transition-colors ${on ? 'bg-blue-500' : 'bg-gray-300'}`}
      >
        <span className={`block w-4 h-4 bg-white rounded-full transition-transform ${on ? 'translate-x-5' : 'translate-x-0.5'}`} />
      </button>
    </label>
  )
}

function CapabilitySelect({ contract }: { contract: CapabilityContract }) {
  const options = useMemo(() => {
    const schema = contract.uiInputSchema
    return (schema.options as Array<{ value: string; label: string }>) ?? []
  }, [contract])
  
  return (
    <select
      onChange={async (e) => {
        await ActionRegistry.dispatch('capability.execute', {
          conversationId: getCurrentConversationId(),
          slug: contract.slug,
          params: { value: e.target.value },
        })
      }}
      className="px-2 py-1 border rounded text-sm"
    >
      {options.map(opt => (
        <option key={opt.value} value={opt.value}>{opt.label}</option>
      ))}
    </select>
  )
}
```

### Position-Based Layout

```tsx
// Render capabilities grouped by position
function CapabilityLayout({ capabilities }: { capabilities: CapabilityContract[] }) {
  const grouped = useMemo(() => {
    const groups: Record<string, CapabilityContract[]> = {}
    for (const cap of capabilities) {
      const pos = cap.uiPosition || 'inline'
      if (!groups[pos]) groups[pos] = []
      groups[pos].push(cap)
    }
    // Sort by uiOrder within each group
    for (const pos of Object.keys(groups)) {
      groups[pos].sort((a, b) => (a.uiOrder ?? 999) - (b.uiOrder ?? 999))
    }
    return groups
  }, [capabilities])
  
  return (
    <>
      {/* Composer toolbar */}
      <div className="flex items-center gap-2 px-3 py-1 border-t">
        {grouped.composer?.map(cap => (
          <GenericCapabilityRenderer key={cap.slug} contract={cap} />
        ))}
      </div>
      
      {/* Header bar */}
      <div className="flex items-center gap-2 px-3 py-1">
        {grouped.header?.map(cap => (
          <GenericCapabilityRenderer key={cap.slug} contract={cap} />
        ))}
      </div>
      
      {/* Sidebar */}
      {grouped.sidebar && (
        <div className="w-48 border-r p-2 space-y-2">
          {grouped.sidebar.map(cap => (
            <GenericCapabilityRenderer key={cap.slug} contract={cap} />
          ))}
        </div>
      )}
    </>
  )
}
```
