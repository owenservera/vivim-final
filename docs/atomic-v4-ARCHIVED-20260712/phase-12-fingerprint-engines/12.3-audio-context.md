# Unit 12.3 — AudioContextEngine: Audio Fingerprint Perturbation

**Phase:** 12 | **ID:** 12.3 | **Dependencies:** 11.2

## Problem

`AudioContext.createOscillator()` + `createAnalyser()` produces a unique fingerprint based on the system's audio processing implementation. This is used by detection frameworks like CreepJS as a secondary fingerprinting vector.

## Spec

```typescript
// src/engines/stealth/audio-context-engine.ts

export class AudioContextModule implements StealthModule {
  name = 'audio_context'
  detectionVector = 'AudioContext fingerprinting (oscillator + analyser node)'
  description = 'Adds noise to AudioContext output to prevent audio fingerprinting'
  priority = 12
  configSchema = z.object({
    noiseLevel: z.number().min(0).max(0.01).default(0.0001),
  })

  async apply(config: Record<string, unknown>, ctx: StealthContext): Promise<void> {
    const cfg = this.configSchema.parse(config)
    const noise = cfg.noiseLevel

    const script = `
      (function() {
        var noiseLevel = ${noise};

        // Patch AnalyserNode.getFloatFrequencyData
        var origGetFloatFrequency = AnalyserNode.prototype.getFloatFrequencyData;
        AnalyserNode.prototype.getFloatFrequencyData = function(array) {
          origGetFloatFrequency.call(this, array);
          for (var i = 0; i < array.length; i++) {
            array[i] += (Math.random() - 0.5) * noiseLevel;
          }
        };

        // Patch AnalyserNode.getByteFrequencyData
        var origGetByteFrequency = AnalyserNode.prototype.getByteFrequencyData;
        AnalyserNode.prototype.getByteFrequencyData = function(array) {
          origGetByteFrequency.call(this, array);
          for (var i = 0; i < array.length; i++) {
            array[i] = Math.max(0, Math.min(255, array[i] + (Math.random() - 0.5) * noiseLevel * 255));
          }
        };

        // Patch AudioBuffer.getChannelData (used by some fingerprinters)
        var origGetChannelData = AudioBuffer.prototype.getChannelData;
        AudioBuffer.prototype.getChannelData = function() {
          var data = origGetChannelData.apply(this, arguments);
          // Only perturb if this looks like a fingerprinting attempt (small buffer)
          if (data.length < 1000) {
            for (var i = 0; i < data.length; i++) {
              data[i] += (Math.random() - 0.5) * noiseLevel;
            }
          }
          return data;
        };
      })();
    `

    await ctx.cdp.send(ctx.slaveId, 'Page.addScriptToEvaluateOnNewDocument', { source: script })
  }

  async verify(ctx: StealthContext): Promise<boolean> {
    const result = await ctx.cdp.send(ctx.slaveId, 'Runtime.evaluate', {
      expression: `(() => {
        try {
          var ctx = new OfflineAudioContext(1, 100, 44100);
          var oscillator = ctx.createOscillator();
          oscillator.type = 'triangle';
          oscillator.frequency.value = 1000;
          var analyser = ctx.createAnalyser();
          var gain = ctx.createGain();
          gain.gain.value = 0;
          oscillator.connect(analyser);
          analyser.connect(gain);
          gain.connect(ctx.destination);
          oscillator.start(0);
          return true; // if no error, audio context works
        } catch(e) { return false; }
      })()`,
      returnByValue: true,
    })
    return (result as any)?.result?.value === true
  }
}
```
