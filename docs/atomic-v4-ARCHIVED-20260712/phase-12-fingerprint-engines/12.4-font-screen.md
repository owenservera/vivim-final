# Unit 12.4 — FontScreenEngine: Font List + Screen Resolution Spoofing

**Phase:** 12 | **ID:** 12.4 | **Dependencies:** 11.2

## Problem

Two fingerprinting vectors: (1) Font availability probing — detection scripts measure element widths with different font families to detect which fonts are installed. (2) Screen resolution — `screen.width`, `screen.height`, `devicePixelRatio` are unique per display setup. Headless Chrome has `screen.width=0` or default values that don't match common displays.

```typescript
// src/engines/stealth/font-screen-engine.ts

const COMMON_FONTS = [
  'Arial', 'Arial Black', 'Arial Narrow', 'Book Antiqua', 'Bookman Old Style',
  'Calibri', 'Cambria', 'Cambria Math', 'Candara', 'Comic Sans MS', 'Consolas',
  'Constantia', 'Corbel', 'Courier', 'Courier New', 'Ebrima', 'Franklin Gothic Medium',
  'Gabriola', 'Gadugi', 'Georgia', 'Impact', 'Javanese Text', 'Leelawadee UI',
  'Lucida Console', 'Lucida Handwriting', 'Lucida Sans Unicode', 'Malgun Gothic',
  'Microsoft Sans Serif', 'MingLiU', 'MS Gothic', 'MS PGothic', 'MS Sans Serif',
  'MS Serif', 'Microsoft YaHei', 'Mongolian Baiti', 'Nirmala UI', 'Palatino Linotype',
  'Segoe Print', 'Segoe Script', 'Segoe UI', 'Segoe UI Emoji', 'Segoe UI Historic',
  'Segoe UI Symbol', 'SimSun', 'Sylfaen', 'Tahoma', 'Times', 'Times New Roman',
  'Trebuchet MS', 'Verdana', 'Webdings', 'Wingdings', 'Yu Gothic',
]

const COMMON_RESOLUTIONS = [
  { width: 1920, height: 1080, devicePixelRatio: 1 },
  { width: 1536, height: 864, devicePixelRatio: 1.25 },
  { width: 2560, height: 1440, devicePixelRatio: 1 },
  { width: 1440, height: 900, devicePixelRatio: 1 },
  { width: 1366, height: 768, devicePixelRatio: 1 },
  { width: 1280, height: 720, devicePixelRatio: 1 },
]

export class FontScreenModule implements StealthModule {
  name = 'font_screen'
  detectionVector = 'Font availability probing + screen resolution fingerprinting'
  description = 'Normalizes font list to common Windows fonts and spoofs screen resolution'
  priority = 13
  configSchema = z.object({
    fonts: z.array(z.string()).optional(),
    screen: z.object({
      width: z.number().optional(),
      height: z.number().optional(),
      devicePixelRatio: z.number().optional(),
    }).optional(),
    platform: z.enum(['windows', 'mac', 'linux']).default('windows'),
  })

  async apply(config: Record<string, unknown>, ctx: StealthContext): Promise<void> {
    const cfg = this.configSchema.parse(config)

    const fonts = cfg.fonts ?? COMMON_FONTS
    const screen = cfg.screen ?? this.pickResolution(cfg.platform)
    const platform = cfg.platform

    const script = `
      (function() {
        // ── Screen Resolution ──────────────────────────────
        var sw = ${screen.width}, sh = ${screen.height}, dpr = ${screen.devicePixelRatio || 1};

        Object.defineProperty(screen, 'width', { get: function() { return sw; } });
        Object.defineProperty(screen, 'height', { get: function() { return sh; } });
        Object.defineProperty(screen, 'availWidth', { get: function() { return sw; } });
        Object.defineProperty(screen, 'availHeight', { get: function() { return sh - 40; } });
        Object.defineProperty(screen, 'colorDepth', { get: function() { return 24; } });
        Object.defineProperty(screen, 'pixelDepth', { get: function() { return 24; } });

        Object.defineProperty(window, 'devicePixelRatio', { get: function() { return dpr; } });

        // ── Font Spoofing ──────────────────────────────────
        var allowedFonts = ${JSON.stringify(fonts.reduce((acc, f) => { acc[f.toLowerCase()] = true; return acc }, {}))};

        // Patch document.fonts.check (Font Loading API)
        if (document.fonts && document.fonts.check) {
          var origCheck = document.fonts.check.bind(document.fonts);
          document.fonts.check = function(font, text) {
            // Parse font family from font shorthand
            var match = /["']?([^"']+)["']?$/.exec(font);
            if (match && match[1]) {
              var family = match[1].toLowerCase();
              if (!allowedFonts[family]) return false;
            }
            return origCheck(font, text);
          };
        }

        // Patch offsetWidth/offsetHeight measurement (used for font probing)
        // This is tricky — we can't intercept offsetWidth globally without breaking layout
        // Instead, we intercept getComputedStyle for font-family and return only allowed fonts
        var origGetComputedStyle = window.getComputedStyle;
        window.getComputedStyle = function(elt, pseudoElt) {
          var style = origGetComputedStyle.call(this, elt, pseudoElt);
          var origFontFamily = Object.getOwnPropertyDescriptor(style, 'fontFamily');
          // Wrap fontFamily getter to filter fonts
          // Note: CSSStyleDeclaration is live, so we need a Proxy
          return new Proxy(style, {
            get: function(target, prop) {
              if (prop === 'fontFamily') {
                var val = target.fontFamily;
                // Filter to only allowed fonts
                return val.split(',').filter(function(f) {
                  var name = f.trim().replace(/["']/g, '').toLowerCase();
                  return allowedFonts[name];
                }).join(', ') || 'Arial';
              }
              return target[prop];
            }
          });
        };

        // ── Platform ───────────────────────────────────────
        Object.defineProperty(navigator, 'platform', {
          get: function() { return ${JSON.stringify(this.platformString(platform))}; }
        });
      })();
    `

    await ctx.cdp.send(ctx.slaveId, 'Page.addScriptToEvaluateOnNewDocument', { source: script })
  }

  private pickResolution(platform: string) {
    const idx = Math.floor(Math.random() * COMMON_RESOLUTIONS.length)
    return COMMON_RESOLUTIONS[idx]!
  }

  private platformString(platform: string): string {
    switch (platform) {
      case 'mac': return 'MacIntel'
      case 'linux': return 'Linux x86_64'
      case 'windows':
      default: return 'Win32'
    }
  }

  async verify(ctx: StealthContext): Promise<boolean> {
    const result = await ctx.cdp.send(ctx.slaveId, 'Runtime.evaluate', {
      expression: `(() => {
        return JSON.stringify({
          width: screen.width,
          height: screen.height,
          ratio: window.devicePixelRatio,
          platform: navigator.platform,
        });
      })()`,
      returnByValue: true,
    })
    const data = JSON.parse((result as any)?.result?.value ?? '{}')
    return data.width > 0 && data.height > 0 && data.ratio > 0
  }
}
```

## Test

```typescript
test('screen resolution is spoofed to non-zero values', async () => {
  await module.apply({ screen: { width: 1920, height: 1080, devicePixelRatio: 1 } }, ctx)
  const ok = await module.verify!(ctx)
  expect(ok).toBe(true)
})

test('platform is Win32', async () => {
  await module.apply({ platform: 'windows' }, ctx)
  const result = await ctx.cdp.send(slaveId, 'Runtime.evaluate', {
    expression: 'navigator.platform',
    returnByValue: true,
  })
  expect((result as any).result.value).toBe('Win32')
})
```
