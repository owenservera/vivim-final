# Unit 5.5 — Latency Budget Enforcement + Per-Stage Metrics

**Phase:** 5 | **ID:** 5.5 | **Dependencies:** 5.4

## Problem

The MirrorEngine has latency budget constants (`_LATENCY_BUDGETS`) but they're unused. The pipeline has no per-stage timing instrumentation. Users can't see where time is spent.

## Spec

### Stage Instrumentation in ConversationManager

```typescript
interface StageTiming {
  resolve: number
  recall: number
  ensure: number
  type: number
  submit: number
  capture: number
  parse: number
  store: number
  emit: number
  total: number
}

// In send():
async send(conversationId: string, message: string): Promise<SendResult> {
  const stages: Partial<StageTiming> = {}
  const totalStart = Date.now()
  
  // [1] RESOLVE
  const t0 = Date.now()
  const conv = await this.store.getConversation(conversationId)
  // ...
  stages.resolve = Date.now() - t0
  
  // [2] RECALL
  const t1 = Date.now()
  // ...
  stages.recall = Date.now() - t1
  
  // [3] ENSURE
  const t2 = Date.now()
  const slave = await this.governor.ensureRunningForAccount(...)
  stages.ensure = Date.now() - t2
  
  // [4] TYPE
  const t3 = Date.now()
  await typeMessage(...)
  stages.type = Date.now() - t3
  
  // [5] SUBMIT
  const t4 = Date.now()
  await submitMessage(...)
  stages.submit = Date.now() - t4
  
  // [6] CAPTURE
  const t5 = Date.now()
  const captureResult = await this.governor.cdp.capture(...)
  stages.capture = Date.now() - t5
  
  // [7] PARSE
  const t6 = Date.now()
  const parseResult = await this.parser.parse(...)
  stages.parse = Date.now() - t6
  
  // [8] STORE+EMIT
  const t7 = Date.now()
  await this.store.createMessage(...)
  stages.store = Date.now() - t7
  
  stages.total = Date.now() - totalStart
  
  // Record to MirrorEngine
  if (this.mirror) {
    for (const [stage, duration] of Object.entries(stages)) {
      await this.mirror.recordStageLatency(conversationId, stage, duration)
    }
  }
  
  // Include timing in result
  return {
    ok: true,
    messageId: msgRow.id,
    blocks: parseResult.blocks,
    text: extractText(parseResult.blocks),
    latencyMs: stages.total,
    timing: stages,  // per-stage breakdown
  }
}
```

### Budget Enforcement

```typescript
const LATENCY_BUDGETS = {
  resolve: 100,    // DB lookup + resolution
  recall: 200,     // Memory/context assembly
  ensure: 3000,    // Chrome ensure (may include spawn)
  type: 200,       // CDP type command
  submit: 100,     // CDP submit command
  capture: 60000,  // Wait for provider response
  parse: 100,      // Parse response body
  store: 50,       // DB writes
}

// After each stage, check budget:
function checkBudget(stage: string, actual: number, budget: number): boolean {
  if (actual > budget) {
    eventBus.emit({
      type: 'latency:budget_exceeded',
      stage,
      actual,
      budget,
      conversationId,
    })
    return false
  }
  return true
}
```

### Visual Budget Bar

```tsx
function BudgetBar({ timing }: { timing: StageTiming }) {
  const budget = LATENCY_BUDGETS
  return (
    <div className="flex h-1 rounded overflow-hidden mt-1">
      {Object.entries(timing).filter(([k]) => k !== 'total').map(([stage, ms]) => {
        const b = budget[stage] ?? 1000
        const overBudget = ms > b
        const width = Math.min(100, (ms / timing.total) * 100)
        return (
          <div
            key={stage}
            className={`h-full ${overBudget ? 'bg-red-400' : 'bg-green-400'}`}
            style={{ width: `${width}%` }}
            title={`${stage}: ${ms}ms / ${b}ms budget`}
          />
        )
      })}
    </div>
  )
}
```
