# Unit 11.3 — StealthProfile Store: Per-Provider Profile Config from DB

**Phase:** 11 | **ID:** 11.3 | **Dependencies:** none

## Problem

Stealth profiles and module profiles need to be persisted in DB, queryable, and attachable to providers via their fleet config.

## Spec

### Prisma Schema

```prisma
// prisma/schema.prisma — add:

model StealthLaunchProfile {
  id                 String   @id
  mode               String   // 'cdp_minimal' | 'cdp_stealth' | 'attach' | 'extension' | 'hidden'
  chromeArgsJson     String   @default("[]")
  stealthProfileId   String?  // FK to StealthModuleProfile
  attachPort         Int?
  extensionId        String?
  windowSizeJson     String   @default("{\"width\":1280,\"height\":720}")
  extraArgsJson      String   @default("[]")
  createdAt          Int      @default(0)
  updatedAt          Int      @default(0)

  @@map("stealth_launch_profile")
}

model StealthModuleProfile {
  id           String   @id
  name         String
  modulesJson  String   @default("[]")  // Array<{ name, enabled, config }>
  createdAt    Int      @default(0)
  updatedAt    Int      @default(0)

  @@map("stealth_module_profile")
}

model StealthPolicy {
  id                String   @id @default("default")
  defaultLaunchProfileId  String?
  defaultModuleProfileId  String?
  providerOverridesJson   String   @default("{}") // { "chatgpt": "stealth_max", ... }

  @@map("stealth_policy")
}
```

### Store Contract

```typescript
// src/storage/contracts/stealth-store.ts

export interface LaunchProfileRow {
  id: string
  mode: string
  chromeArgsJson: string
  stealthProfileId: string | null
  attachPort: number | null
  extensionId: string | null
  windowSizeJson: string
  extraArgsJson: string
}

export interface ModuleProfileRow {
  id: string
  name: string
  modulesJson: string
}

export interface StealthPolicyRow {
  defaultLaunchProfileId: string | null
  defaultModuleProfileId: string | null
  providerOverridesJson: string
}

export interface StealthProfileStore {
  // Launch profiles
  getAllLaunchProfiles(): Promise<LaunchProfileRow[]>
  getLaunchProfile(id: string): Promise<LaunchProfileRow | null>
  upsertLaunchProfile(profile: LaunchProfileRow): Promise<void>
  deleteLaunchProfile(id: string): Promise<void>

  // Module profiles
  getAllModuleProfiles(): Promise<ModuleProfileRow[]>
  getModuleProfile(id: string): Promise<ModuleProfileRow | null>
  upsertModuleProfile(profile: ModuleProfileRow): Promise<void>
  deleteModuleProfile(id: string): Promise<void>

  // Policy
  getPolicy(): Promise<StealthPolicyRow | null>
  setPolicy(policy: StealthPolicyRow): Promise<void>
}
```

### Implementation

```typescript
// src/storage/impl/stealth-store-impl.ts

export class StealthStoreImpl implements StealthProfileStore {
  constructor(private db: CapStoreDb) {}

  async getAllLaunchProfiles(): Promise<LaunchProfileRow[]> {
    const rows = await this.db.prisma.stealthLaunchProfile.findMany()
    return rows.map(r => ({
      id: r.id,
      mode: r.mode,
      chromeArgsJson: r.chromeArgsJson,
      stealthProfileId: r.stealthProfileId,
      attachPort: r.attachPort,
      extensionId: r.extensionId,
      windowSizeJson: r.windowSizeJson,
      extraArgsJson: r.extraArgsJson,
    }))
  }

  async getLaunchProfile(id: string): Promise<LaunchProfileRow | null> {
    const row = await this.db.prisma.stealthLaunchProfile.findUnique({ where: { id } })
    if (!row) return null
    return {
      id: row.id,
      mode: row.mode,
      chromeArgsJson: row.chromeArgsJson,
      stealthProfileId: row.stealthProfileId,
      attachPort: row.attachPort,
      extensionId: row.extensionId,
      windowSizeJson: row.windowSizeJson,
      extraArgsJson: row.extraArgsJson,
    }
  }

  async upsertLaunchProfile(profile: LaunchProfileRow): Promise<void> {
    const now = Date.now()
    await this.db.prisma.stealthLaunchProfile.upsert({
      where: { id: profile.id },
      create: {
        id: profile.id,
        mode: profile.mode,
        chromeArgsJson: profile.chromeArgsJson,
        stealthProfileId: profile.stealthProfileId,
        attachPort: profile.attachPort,
        extensionId: profile.extensionId,
        windowSizeJson: profile.windowSizeJson,
        extraArgsJson: profile.extraArgsJson,
        createdAt: now,
        updatedAt: now,
      },
      update: {
        mode: profile.mode,
        chromeArgsJson: profile.chromeArgsJson,
        stealthProfileId: profile.stealthProfileId,
        attachPort: profile.attachPort,
        extensionId: profile.extensionId,
        windowSizeJson: profile.windowSizeJson,
        extraArgsJson: profile.extraArgsJson,
        updatedAt: now,
      },
    })
  }

  async deleteLaunchProfile(id: string): Promise<void> {
    await this.db.prisma.stealthLaunchProfile.delete({ where: { id } })
  }

  // ... similar for module profiles and policy
}
```

### Provider Manifest Integration

Add `launch_profile` and `stealth_profile` fields to provider seed manifests:

```json
// seeds/providers/chatgpt.json — add to fleet_config:
{
  "fleet_config": {
    "port_range": [9252, 9280],
    "launch_profile_id": "stealth_default",
    "stealth_profile_id": "stealth_standard"
  }
}
```

ProviderRegistrar reads these and the LaunchProfileEngine picks them up when resolving.

## Test

```typescript
test('launch profile CRUD', async () => {
  await store.upsertLaunchProfile({ id: 'test', mode: 'cdp_minimal', ... })
  const profile = await store.getLaunchProfile('test')
  expect(profile).not.toBeNull()
  expect(profile!.mode).toBe('cdp_minimal')

  await store.deleteLaunchProfile('test')
  expect(await store.getLaunchProfile('test')).toBeNull()
})

test('policy overrides per provider', async () => {
  await store.setPolicy({
    defaultLaunchProfileId: 'default',
    defaultModuleProfileId: 'stealth_none',
    providerOverridesJson: JSON.stringify({ chatgpt: 'stealth_maximum' }),
  })

  const policy = await store.getPolicy()
  const overrides = JSON.parse(policy!.providerOverridesJson)
  expect(overrides.chatgpt).toBe('stealth_maximum')
})
```
