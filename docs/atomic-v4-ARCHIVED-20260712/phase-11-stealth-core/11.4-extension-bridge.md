# Unit 11.4 — ExtensionBridgeEngine: Browser Extension Interaction Mode

**Phase:** 11 | **ID:** 11.4 | **Dependencies:** 11.1

## Problem

CDP is the current interaction mode, but it leaves traces detectable by advanced anti-bot systems. A browser extension provides zero-footprint DOM interaction — content scripts run in the page context with no CDP artifacts.

This engine implements the "extension mode" from Option B in the anti-detection analysis. It's an alternative to CDP for all DOM operations (typing, clicking, capturing, observing).

## Spec

### Architecture

```
VIVIM Backend (Bun :9420)
  ↕ HTTP/WS (localhost)
VIVIM Extension Background Service Worker
  ↕ chrome.tabs.sendMessage
VIVIM Content Script (in provider page)
  → Full DOM access
  → fetch interception
  → Event dispatch
```

```typescript
// src/engines/extension-bridge-engine.ts

export interface ExtensionCommand {
  type: 'type_text' | 'click' | 'navigate' | 'scroll' | 'capture_fetch' | 'get_dom' | 'evaluate'
  target?: string         // CSS selector
  value?: string          // text to type, URL, JS expression
  options?: Record<string, unknown>
  requestId: string
}

export interface ExtensionResponse {
  requestId: string
  ok: boolean
  data?: unknown
  error?: string
}

export interface PendingExtensionRequest {
  resolve: (value: unknown) => void
  reject: (error: Error) => void
  timer: ReturnType<typeof setTimeout>
}

export class ExtensionBridgeEngine {
  private pending = new Map<string, PendingExtensionRequest>()
  private connectedTabs = new Map<string, { tabId: number; providerId: string }>()
  private ws: WebSocket | null = null
  private timeoutMs = 30_000

  constructor(
    private logger?: StructuredLogger,
  ) {
    void this.connectToExtension()
  }

  // ── Connection ──────────────────────────────────────────

  private async connectToExtension(): Promise<void> {
    // The extension connects to VIVIM backend via WebSocket
    // When it connects, it registers itself with a hello message
    // No polling needed — extension initiates connection
  }

  handleExtensionConnect(ws: WebSocket, tabId: number, providerId: string): void {
    this.ws = ws
    ws.onmessage = (ev) => {
      const msg = JSON.parse(ev.data as string) as ExtensionResponse
      const pending = this.pending.get(msg.requestId)
      if (!pending) return

      clearTimeout(pending.timer)
      this.pending.delete(msg.requestId)

      if (msg.ok) {
        pending.resolve(msg.data)
      } else {
        pending.reject(new Error(msg.error ?? 'Extension command failed'))
      }
    }
  }

  // ── DOM Operations (same interface as CDP) ───────────────

  async typeText(slaveId: string, selector: string, text: string, opts?: {
    delayMs?: number
    clearFirst?: boolean
  }): Promise<void> {
    return this.sendCommand(slaveId, {
      type: 'type_text',
      target: selector,
      value: text,
      options: opts,
      requestId: this.newRequestId(),
    }) as Promise<void>
  }

  async click(slaveId: string, selector: string): Promise<void> {
    return this.sendCommand(slaveId, {
      type: 'click',
      target: selector,
      requestId: this.newRequestId(),
    }) as Promise<void>
  }

  async navigate(slaveId: string, url: string): Promise<void> {
    return this.sendCommand(slaveId, {
      type: 'navigate',
      value: url,
      requestId: this.newRequestId(),
    }) as Promise<void>
  }

  async scroll(slaveId: string, direction: 'up' | 'down', amount?: number): Promise<void> {
    return this.sendCommand(slaveId, {
      type: 'scroll',
      value: direction,
      options: { amount },
      requestId: this.newRequestId(),
    }) as Promise<void>
  }

  async captureFetch(slaveId: string, pattern: string, timeoutMs?: number): Promise<string> {
    return this.sendCommand(slaveId, {
      type: 'capture_fetch',
      value: pattern,
      options: { timeoutMs },
      requestId: this.newRequestId(),
    }) as Promise<string>
  }

  async getDom(slaveId: string, selector?: string): Promise<string> {
    return this.sendCommand(slaveId, {
      type: 'get_dom',
      target: selector,
      requestId: this.newRequestId(),
    }) as Promise<string>
  }

  async evaluate(slaveId: string, expression: string): Promise<unknown> {
    return this.sendCommand(slaveId, {
      type: 'evaluate',
      value: expression,
      requestId: this.newRequestId(),
    })
  }

  // ── Internal ────────────────────────────────────────────

  private async sendCommand(slaveId: string, command: ExtensionCommand): Promise<unknown> {
    if (!this.ws) throw new EngineError('Extension not connected')

    return new Promise((resolve, reject) => {
      const timer = setTimeout(() => {
        this.pending.delete(command.requestId)
        reject(new EngineError(`Extension command timed out: ${command.type}`))
      }, this.timeoutMs)

      this.pending.set(command.requestId, { resolve, reject, timer })
      this.ws!.send(JSON.stringify({ ...command, slaveId }))
    })
  }

  private newRequestId(): string {
    return `ext_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`
  }

  get isConnected(): boolean {
    return this.ws !== null
  }
}
```

### Extension Content Script

```javascript
// vivim-extension/content.js — runs inside provider page

// Intercept fetch for response capture
const origFetch = window.fetch
const captureCallbacks = new Map()

window.fetch = async function(...args) {
  const response = await origFetch.apply(this, args)
  const url = (args[0]?.url ?? args[0]?.toString?.() ?? '').toString()

  for (const [pattern, callback] of captureCallbacks) {
    if (url.includes(pattern)) {
      const cloned = response.clone()
      cloned.text().then(body => callback(body)).catch(() => {})
    }
  }

  return response
}

// Listen for commands from background script
chrome.runtime.onMessage.addListener((msg, _sender, sendResponse) => {
  if (msg.type === 'vivim:command') {
    handleCommand(msg.command).then(
      result => sendResponse({ requestId: msg.command.requestId, ok: true, data: result }),
      error => sendResponse({ requestId: msg.command.requestId, ok: false, error: error.message }),
    )
    return true // async
  }
})

async function handleCommand(cmd) {
  switch (cmd.type) {
    case 'type_text': {
      const el = document.querySelector(cmd.target)
      if (!el) throw new Error(`Element not found: ${cmd.target}`)

      el.focus()

      if (cmd.options?.clearFirst) {
        el.value = ''
      }

      // Determine element type
      if (el.tagName === 'TEXTAREA') {
        // React-compatible textarea typing
        const nativeSetter = Object.getOwnPropertyDescriptor(
          HTMLTextAreaElement.prototype, 'value'
        )?.set
        nativeSetter?.call(el, cmd.value)
        el.dispatchEvent(new Event('input', { bubbles: true }))
      } else if (el.contentEditable === 'true') {
        // ContentEditable typing
        el.focus()
        document.execCommand('selectAll', false)
        document.execCommand('insertText', false, cmd.value)
      } else {
        el.value = cmd.value
        el.dispatchEvent(new Event('input', { bubbles: true }))
      }

      return { typed: true }
    }

    case 'click': {
      const el = document.querySelector(cmd.target)
      if (!el) throw new Error(`Element not found: ${cmd.target}`)
      el.click()
      return { clicked: true }
    }

    case 'navigate': {
      window.location.href = cmd.value
      return { navigated: true }
    }

    case 'scroll': {
      const px = cmd.options?.amount ?? 300
      const dy = cmd.value === 'down' ? px : -px
      window.scrollBy(0, dy)
      return { scrolled: true }
    }

    case 'capture_fetch': {
      return new Promise((resolve, reject) => {
        const pattern = cmd.value
        const timeout = cmd.options?.timeoutMs ?? 30000

        captureCallbacks.set(pattern, (body) => {
          captureCallbacks.delete(pattern)
          resolve(body)
        })

        setTimeout(() => {
          captureCallbacks.delete(pattern)
          reject(new Error(`Capture timeout for pattern: ${pattern}`))
        }, timeout)
      })
    }

    case 'get_dom': {
      if (cmd.target) {
        const el = document.querySelector(cmd.target)
        return el ? el.innerHTML : ''
      }
      return document.documentElement.innerHTML
    }

    case 'evaluate': {
      return eval(cmd.value)
    }
  }
}
```

### Extension Background Script

```javascript
// vivim-extension/background.js

// Connect to VIVIM backend
let vivimWs = null

function connectToVivim() {
  vivimWs = new WebSocket('ws://127.0.0.1:9420/ws')

  vivimWs.onopen = () => {
    vivimWs.send(JSON.stringify({
      type: 'hello',
      role: 'extension',
      sessionId: 'vivim-extension',
    }))
  }

  vivimWs.onmessage = async (event) => {
    const msg = JSON.parse(event.data)

    if (msg.type === 'vivim:command' && msg.slaveId) {
      // Find the tab for this slave
      const tab = await findTabForSlave(msg.slaveId)
      if (!tab) {
        vivimWs.send(JSON.stringify({
          requestId: msg.command.requestId,
          ok: false,
          error: 'Tab not found for slave',
        }))
        return
      }

      // Forward command to content script
      chrome.tabs.sendMessage(tab.id, {
        type: 'vivim:command',
        command: msg.command,
      }, (response) => {
        // Forward response back to VIVIM
        vivimWs.send(JSON.stringify(response))
      })
    }
  }

  vivimWs.onclose = () => {
    setTimeout(connectToVivim, 1000) // reconnect
  }
}

connectToVivim()

async function findTabForSlave(slaveId) {
  // slaveId format: chatgpt_work_timestamp
  // Map to tab by URL
  const [provider] = slaveId.split('_')
  const urlMap = {
    chatgpt: 'https://chatgpt.com/*',
    claude: 'https://claude.ai/*',
    gemini: 'https://gemini.google.com/*',
  }

  return new Promise(resolve => {
    chrome.tabs.query({ url: urlMap[provider] || '*://*/*' }, (tabs) => {
      resolve(tabs[0] ?? null)
    })
  })
}
```
