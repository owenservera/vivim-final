# Unit 11.2 — StealthModuleEngine: Registry + CDP Injection Pipeline

**Phase:** 11 | **ID:** 11.2 | **Dependencies:** 11.1, 11.3

## Problem

There's no system for registering, composing, and applying stealth modules to a running Chrome instance. Each module needs to inject JavaScript via CDP before any page loads.

## Spec

```typescript
export interface StealthModule {
  name: string                        // unique identifier: 'canvas_noise'
  detectionVector: string             // human-readable: 'Canvas fingerprinting'

export interface StealthModuleConfig {
```

### StealthModuleEngine

```typescript
// src/engines/stealth/stealth-module-engine.ts

export class StealthModuleEngine {
  private modules = new Map<string, StealthModule>()
  private profiles = new Map<string, StealthModuleProfile>()
  private applied = new Map<string, Set<string>>()  // slaveId → applied module names

  constructor(
    private store: StealthProfileStore,
    private eventBus?: CapabilityEventBus,
    private logger?: StructuredLogger,
  ) {
    void this.loadProfiles()
  }

  // ── Module Registration ──────────────────────────────────

  registerModule(module: StealthModule): void {
    if (this.modules.has(module.name)) {
      throw new EngineError(`Stealth module already registered: ${module.name}`)
    }
    this.modules.set(module.name, module)
    this.logger?.debug(`Registered stealth module: ${module.name}`, {
      detectionVector: module.detectionVector,
      priority: module.priority,
    })
  }

  unregisterModule(name: string): void {
    this.modules.delete(name)
  }

  listModules(): Array<{ name: string; detectionVector: string; description: string }> {
    return [...this.modules.values()].map(m => ({
      name: m.name,
      detectionVector: m.detectionVector,
      description: m.description,
    }))
  }

  // ── Profile Management ───────────────────────────────────

  private async loadProfiles(): Promise<void> {
    const rows = await this.store.getAllModuleProfiles()
    for (const row of rows) {
      this.profiles.set(row.id, {
        id: row.id,
        name: row.name,
        modules: JSON.parse(row.modulesJson ?? '[]'),
      })
    }
  }

  getProfile(id: string): StealthModuleProfile | null {
    return this.profiles.get(id) ?? null
  }

  async registerProfile(profile: StealthModuleProfile): Promise<void> {
    this.profiles.set(profile.id, profile)
    await this.store.upsertModuleProfile({
      id: profile.id,
      name: profile.name,
      modulesJson: JSON.stringify(profile.modules),
    })
  }

  // ── Core: Apply Profile to Slave ─────────────────────────

  async applyProfile(slaveId: string, profileId: string): Promise<void> {
    const profile = this.profiles.get(profileId)
    if (!profile) throw new EngineError(`Stealth profile not found: ${profileId}`)

    const appliedSet = new Set<string>()
    this.applied.set(slaveId, appliedSet)

    // Sort modules by priority (lower = first)
    const sorted = [...profile.modules]
      .filter(m => m.enabled)
      .sort((a, b) => {
        const pa = this.modules.get(a.name)?.priority ?? 100
        const pb = this.modules.get(b.name)?.priority ?? 100
        return pa - pb
      })

    const ctx: StealthContext = {
      cdp: this.getCdpProxy(slaveId),
      slaveId,
      logger: this.logger,
    }

    for (const moduleConfig of sorted) {
      const module = this.modules.get(moduleConfig.name)
      if (!module) {
        this.logger?.warn(`Stealth module not found, skipping: ${moduleConfig.name}`)
        continue
      }

      try {
        // Validate config
        const validated = module.configSchema.parse(moduleConfig.config)

        // Apply
        await module.apply(validated as Record<string, unknown>, ctx)
        appliedSet.add(moduleConfig.name)

        // Verify (optional)
        if (module.verify) {
          const ok = await module.verify(ctx)
          if (!ok) {
            this.logger?.warn(`Stealth module verification failed: ${moduleConfig.name}`)
          }
        }

        this.logger?.debug(`Applied stealth module: ${moduleConfig.name}`, { slaveId })
      } catch (err) {
        this.logger?.error(
          `Failed to apply stealth module: ${moduleConfig.name}`,
          err instanceof Error ? err : undefined,
          { slaveId, module: moduleConfig.name },
        )
      }
    }

    this.eventBus?.emit({
      type: 'stealth:profile_applied',
      slaveId,
      profileId,
      modulesApplied: [...appliedSet],
    })
  }

  // ── Query ────────────────────────────────────────────────

  getAppliedModules(slaveId: string): string[] {
    return [...(this.applied.get(slaveId) ?? [])]
  }

  async verifyAll(slaveId: string): Promise<Array<{ module: string; ok: boolean }>> {
    const applied = this.applied.get(slaveId) ?? new Set()
    const results: Array<{ module: string; ok: boolean }> = []
    const ctx: StealthContext = { cdp: this.getCdpProxy(slaveId), slaveId }

    for (const name of applied) {
      const module = this.modules.get(name)
      if (module?.verify) {
        results.push({ module: name, ok: await module.verify(ctx) })
      } else {
        results.push({ module: name, ok: true }) // no verify = assume ok
      }
    }
    return results
  }

  private getCdpProxy(slaveId: string): CDPProxy {
    // Access the Governor's CDP proxy
    // This is injected during server bootstrap
    return this.governor!.cdp
  }

  private governor?: ChromeGovernor

  setGovernor(governor: ChromeGovernor): void {
    this.governor = governor
  }
}
```

### NavigatorPatchModule (always needed)

```typescript
// src/engines/stealth/navigator-patch-module.ts

export class NavigatorPatchModule implements StealthModule {
  name = 'navigator_patch'
  detectionVector = 'navigator.webdriver + plugins + languages'
  description = 'Patches navigator.webdriver to undefined, adds realistic plugins and languages'
  priority = 0  // must be first
  configSchema = z.object({
    plugins: z.array(z.object({ name: z.string() })).optional(),
    languages: z.array(z.string()).optional(),
  }).default({})

  async apply(config: Record<string, unknown>, ctx: StealthContext): Promise<void> {
    const plugins = (config.plugins as any[]) ?? [
      { name: 'Chrome PDF Plugin' },
      { name: 'Chrome PDF Viewer' },
      { name: 'Native Client' },
    ]
    const languages = (config.languages as string[]) ?? ['en-US', 'en']

    const script = `
      Object.defineProperty(navigator, 'webdriver', { get: () => undefined });
      Object.defineProperty(navigator, 'plugins', {
        get: () => ${JSON.stringify(plugins)}.map(p => ({ ...p, length: 1 }))
      });
      Object.defineProperty(navigator, 'languages', { get: () => ${JSON.stringify(languages)} });
      window.chrome = window.chrome || {};
      window.chrome.runtime = window.chrome.runtime || {};
      const origQuery = window.navigator.permissions?.query;
      if (origQuery) {
        window.navigator.permissions.query = (parameters) =>
          parameters.name === 'notifications'
            ? Promise.resolve({ state: Notification.permission })
            : origQuery(parameters);
      }
    `

    await ctx.cdp.send(ctx.slaveId, 'Page.addScriptToEvaluateOnNewDocument', { source: script })
  }

  async verify(ctx: StealthContext): Promise<boolean> {
    const result = await ctx.cdp.send(ctx.slaveId, 'Runtime.evaluate', {
      expression: 'navigator.webdriver === undefined',
      returnByValue: true,
    })
    return (result as any)?.result?.value === true
  }
}
```

## Test

```typescript
test('applyProfile applies all enabled modules in priority order', async () => {
  const applied: string[] = []
  engine.registerModule({ name: 'a', priority: 10, apply: async () => applied.push('a') } as any)
  engine.registerModule({ name: 'b', priority: 0, apply: async () => applied.push('b') } as any)

  await engine.applyProfile('slave1', 'test_profile')

  expect(applied).toEqual(['b', 'a']) // priority 0 first
})

test('disabled modules are skipped', async () => {
  let applied = false
  engine.registerModule({ name: 'test', priority: 0, apply: async () => { applied = true } } as any)

  await engine.applyProfile('slave1', {
    id: 'test',
    name: 'test',
    modules: [{ name: 'test', enabled: false, config: {} }],
  })

  expect(applied).toBe(false)
})

test('verifyAll checks all applied modules', async () => {
  engine.registerModule({
    name: 'test',
    priority: 0,
    apply: async () => {},
    verify: async () => true,
  } as any)

  await engine.applyProfile('slave1', 'test_profile')
  const results = await engine.verifyAll('slave1')
  expect(results[0].ok).toBe(true)
})
```
