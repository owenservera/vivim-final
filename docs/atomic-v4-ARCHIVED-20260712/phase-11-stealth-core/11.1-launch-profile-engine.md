# Unit 11.1 — LaunchProfileEngine: Multi-Mode Launch Strategy

**Phase:** 11 | **ID:** 11.1 | **Dependencies:** 1.1

## Problem

`launcher.ts:buildChromeArgs()` hardcodes one launch mode with bot-signal flags. There's no way to switch between attach/extension/cdp-stealth/cdp-minimal modes per provider.

## Spec

### Engine Interface

```typescript
// src/engines/launch-profile-engine.ts

export type LaunchMode = 'cdp_minimal' | 'cdp_stealth' | 'attach' | 'extension' | 'hidden'

export interface LaunchProfile {
  id: string
  mode: LaunchMode
  chromeArgs: string[]               // explicit args (not hardcoded)
  stealthProfileId: string | null     // which stealth modules to apply
  attachPort: number | null           // for 'attach' mode
  extensionId: string | null          // for 'extension' mode
  windowSize: { width: number; height: number }
  extraArgs: string[]
}

export interface LaunchProfilePolicy {
  defaultProfileId: string            // which profile to use if provider doesn't specify
  providerOverrides: Record<string, string>  // providerId → profileId
}

export class LaunchProfileEngine {
  private profiles = new Map<string, LaunchProfile>()

  constructor(
    private store: StealthProfileStore,
    private logger?: StructuredLogger,
  ) {
    void this.loadProfiles()
  }

  private async loadProfiles(): Promise<void> {
    const rows = await this.store.getAllProfiles()
    for (const row of rows) {
      this.profiles.set(row.id, this.mapRow(row))
    }
  }

  async resolve(providerId: string): Promise<LaunchProfile> {
    // Check provider override
    const policy = await this.store.getPolicy()
    const overrideId = policy?.providerOverrides?.[providerId]
    const profileId = overrideId ?? policy?.defaultProfileId ?? 'default'

    const profile = this.profiles.get(profileId)
    if (!profile) {
      // Return minimal safe default
      return this.getDefaultProfile()
    }
    return profile
  }

  buildArgs(profile: LaunchProfile, opts: { debugPort: number; profileDir: string }): string[] {
    switch (profile.mode) {
      case 'cdp_minimal':
        return this.buildMinimalArgs(opts)
      case 'cdp_stealth':
        return this.buildStealthArgs(profile, opts)
      case 'hidden':
        return this.buildHiddenArgs(profile, opts)
      case 'attach':
        return [] // no args — attaching to existing
      case 'extension':
        return this.buildExtensionArgs(profile, opts)
      default:
        return this.buildMinimalArgs(opts)
    }
  }

  private buildMinimalArgs(opts: { debugPort: number; profileDir: string }): string[] {
    return [
      `--remote-debugging-port=${opts.debugPort}`,
      `--user-data-dir=${opts.profileDir}`,
      '--no-first-run',
      '--no-default-browser-check',
    ]
    // NOTE: No --disable-extensions, no --disable-sync, no --disable-background-networking
    // These were the bot signals. Minimal = let Chrome be Chrome.
  }

  private buildStealthArgs(profile: LaunchProfile, opts: { debugPort: number; profileDir: string }): string[] {
    const args = this.buildMinimalArgs(opts)
    // Add profile-specific args (from DB, not hardcoded)
    args.push(...profile.chromeArgs)
    // No headless — stealth uses hidden mode
    args.push('--window-position=-32000,-32000')
    args.push(`--window-size=${profile.windowSize.width},${profile.windowSize.height}`)
    return args
  }

  private buildHiddenArgs(profile: LaunchProfile, opts: { debugPort: number; profileDir: string }): string[] {
    const args = this.buildMinimalArgs(opts)
    args.push('--window-position=-32000,-32000')
    args.push(`--window-size=${profile.windowSize.width},${profile.windowSize.height}`)
    return args
  }

  private buildExtensionArgs(profile: LaunchProfile, opts: { debugPort: number; profileDir: string }): string[] {
    const args = this.buildMinimalArgs(opts)
    // Load the VIVIM extension
    args.push(`--load-extension=${profile.extensionId ?? '/path/to/vivim-extension'}`)
    return args
  }

  private getDefaultProfile(): LaunchProfile {
    return {
      id: 'default',
      mode: 'cdp_minimal',
      chromeArgs: [],
      stealthProfileId: null,
      attachPort: null,
      extensionId: null,
      windowSize: { width: 1280, height: 720 },
      extraArgs: [],
    }
  }

  async registerProfile(profile: LaunchProfile): Promise<void> {
    this.profiles.set(profile.id, profile)
    await this.store.upsertProfile(this.toRow(profile))
  }

  async updateProfile(id: string, patch: Partial<LaunchProfile>): Promise<void> {
    const existing = this.profiles.get(id)
    if (!existing) throw new EngineError(`Profile not found: ${id}`)
    const updated = { ...existing, ...patch }
    this.profiles.set(id, updated)
    await this.store.upsertProfile(this.toRow(updated))
  }

  async deleteProfile(id: string): Promise<void> {
    this.profiles.delete(id)
    await this.store.deleteProfile(id)
  }

  listProfiles(): LaunchProfile[] {
    return [...this.profiles.values()]
  }

  getProfile(id: string): LaunchProfile | null {
    return this.profiles.get(id) ?? null
  }

  private mapRow(row: LaunchProfileRow): LaunchProfile {
    return {
      id: row.id,
      mode: row.mode as LaunchMode,
      chromeArgs: JSON.parse(row.chromeArgsJson ?? '[]'),
      stealthProfileId: row.stealthProfileId,
      attachPort: row.attachPort,
      extensionId: row.extensionId,
      windowSize: JSON.parse(row.windowSizeJson ?? '{"width":1280,"height":720}'),
      extraArgs: JSON.parse(row.extraArgsJson ?? '[]'),
    }
  }

  private toRow(profile: LaunchProfile): LaunchProfileRow {
    return {
      id: profile.id,
      mode: profile.mode,
      chromeArgsJson: JSON.stringify(profile.chromeArgs),
      stealthProfileId: profile.stealthProfileId,
      attachPort: profile.attachPort,
      extensionId: profile.extensionId,
      windowSizeJson: JSON.stringify(profile.windowSize),
      extraArgsJson: JSON.stringify(profile.extraArgs),
    }
  }
}
```

```typescript
test('resolve returns provider override profile', async () => {
  await store.setPolicy({
    defaultProfileId: 'default',
    providerOverrides: { chatgpt: 'stealth_default' },
  })

  const profile = await engine.resolve('chatgpt')
  expect(profile.mode).toBe('cdp_stealth')
  expect(profile.stealthProfileId).toBe('stealth_basic')
})

test('minimal args have no bot signals', () => {
  const args = engine.buildArgs(
    { ...defaultProfile, mode: 'cdp_minimal' },
    { debugPort: 9300, profileDir: '/tmp/test' },
  )
  expect(args).not.toContain('--disable-extensions')
  expect(args).not.toContain('--disable-background-networking')
  expect(args).not.toContain('--disable-sync')
  expect(args).not.toContain('--headless=new')
})

test('can switch profile at runtime', async () => {
  await engine.registerProfile({ id: 'test', mode: 'attach', ... })
  const p1 = await engine.resolve('chatgpt')
  expect(p1.mode).not.toBe('attach')

  await store.setPolicy({ defaultProfileId: 'test', providerOverrides: {} })
  await engine.loadProfiles()
  const p2 = await engine.resolve('chatgpt')
  expect(p2.mode).toBe('attach')
})
```
