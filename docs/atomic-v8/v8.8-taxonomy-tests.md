### `tests/unit/provider-taxonomy/prisma-store.test.ts`

```typescript
import { describe, it, expect, beforeEach } from 'bun:test';
import { PrismaClient } from '@prisma/client';
import { PrismaProviderTaxonomyStore } from '../../src/storage/impl/provider-taxonomy-store.js';

describe('PrismaProviderTaxonomyStore', () => {
  let prisma: PrismaClient;
  let store: PrismaProviderTaxonomyStore;

  beforeEach(() => {
    prisma = new PrismaClient({ datasources: { db: { url: 'file::memory:' } } });
    store = new PrismaProviderTaxonomyStore(prisma);
  });

  it('creates a taxonomy row', async () => {
    const result = await store.create({
      providerId: 'test-provider',
      platformCategory: 'social_messaging',
      interactionPattern: 'message',
    });
    expect(result.id).toBeDefined();
    expect(result.platformCategory).toBe('social_messaging');
  });

  it('upserts by provider and category', async () => {
    const first = await store.upsertByProviderAndCategory(
      'test-provider', 'social_messaging', 'message',
      { messageTypesJson: '["text"]' }
    );
    const second = await store.upsertByProviderAndCategory(
      'test-provider', 'social_messaging', 'message',
      { messageTypesJson: '["text", "image"]' }
    );
    expect(first.id).toBe(second.id);
    expect(JSON.parse(second.messageTypesJson)).toContain('image');
  });
});
```

### `tests/unit/provider-taxonomy/seed.test.ts`

```typescript
import { describe, it, expect } from 'bun:test';
import { platformTaxonomySeeds } from '../../seeds/taxonomy/platform-taxonomy.seed.js';

describe('Platform taxonomy seeds', () => {
  it('has entries for all 6 platforms', () => {
    const slugs = platformTaxonomySeeds.map(s => s.providerSlug);
    expect(slugs).toContain('facebook');
    expect(slugs).toContain('instagram');
    expect(slugs).toContain('linkedin');
    expect(slugs).toContain('whatsapp');
    expect(slugs).toContain('telegram');
    expect(slugs).toContain('x_twitter');
  });

  it('each entry has required fields', () => {
    for (const seed of platformTaxonomySeeds) {
      expect(seed.platformCategory).toBeDefined();
      expect(seed.interactionPattern).toBeDefined();
      expect(seed.messageTypes.length).toBeGreaterThan(0);
      expect(seed.capabilities.length).toBeGreaterThan(0);
      expect(seed.constraints).toBeDefined();
      expect(seed.authRequirements.length).toBeGreaterThan(0);
      expect(seed.discoveryHints.composerSelector).toBeDefined();
      expect(seed.nlpEntityTypes.length).toBeGreaterThan(0);
      expect(seed.nlpIntentPatterns.length).toBeGreaterThan(0);
    }
  });
});
```

### `tests/unit/provider-taxonomy/resolver.test.ts`

```typescript
import { describe, it, expect, beforeEach } from 'bun:test';
import { ProviderTaxonomyResolver } from '../../src/engines/provider-taxonomy/provider-taxonomy-resolver.js';

describe('ProviderTaxonomyResolver', () => {
  let resolver: ProviderTaxonomyResolver;

  beforeEach(() => {
    // Mock store with seed data
    const mockStore = { /* ... */ };
    resolver = new ProviderTaxonomyResolver(mockStore as any);
  });

  it('resolves "post a photo to instagram"', async () => {
    const results = await resolver.resolveIntent('post a photo to instagram');
    expect(results.length).toBeGreaterThan(0);
    expect(results[0].providerSlug).toBe('instagram');
    expect(results[0].matchedIntent).toBe('post_media');
  });

  it('resolves "send a message on whatsapp"', async () => {
    const results = await resolver.resolveIntent('send a message on whatsapp');
    expect(results.length).toBeGreaterThan(0);
    expect(results[0].providerSlug).toBe('whatsapp');
    expect(results[0].matchedIntent).toBe('send_message');
  });

  it('returns empty for unmatched input', async () => {
    const results = await resolver.resolveIntent('xyzzy');
    expect(results.length).toBe(0);
  });
});
```
