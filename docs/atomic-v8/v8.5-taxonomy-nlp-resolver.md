## Context

The NLP resolver bridges user intent to provider capabilities via the taxonomy. When a user says "post a photo to Instagram", the resolver extracts entities (photo, Instagram), matches intent patterns (post → post_media), and returns the matched provider + capability + confidence. This feeds directly into the NLCL HybridResolver.

- `src/engines/nlcl/entity-resolution.ts` — existing entity resolution engine.

### Engine — `src/engines/provider-taxonomy/provider-taxonomy-resolver.ts`

```typescript
import type { ProviderCapabilityTaxonomy } from '@prisma/client';
import type { ProviderTaxonomyStore } from '../../storage/contracts/provider-taxonomy-store.js';
import type {
  ResolvedIntent,
  PlatformCapability,
  PlatformConstraints,
  NLPEntityType,
  NLPIntentPattern,
} from './types.js';

export interface ProviderTaxonomyResolverConfig {
  minConfidence?: number;  // default: 0.6
  maxResults?: number;     // default: 5
}

export class ProviderTaxonomyResolver {
  private minConfidence: number;
  private maxResults: number;

  constructor(
    private store: ProviderTaxonomyStore,
    config?: ProviderTaxonomyResolverConfig
  ) {
    this.minConfidence = config?.minConfidence ?? 0.6;
    this.maxResults = config?.maxResults ?? 5;
  }

  /**
   * Resolve user intent to provider capability using taxonomy.
   */
  async resolveIntent(
    userInput: string,
    providerSlug?: string
  ): Promise<ResolvedIntent[]> {
    // Get taxonomies
    const taxonomies = providerSlug
      ? await this.getTaxonomiesForProvider(providerSlug)
      : await this.store.list({ isActive: 1 });

    const results: ResolvedIntent[] = [];

    for (const taxonomy of taxonomies) {
      const intentPatterns: NLPIntentPattern[] = JSON.parse(taxonomy.nlpIntentPatternsJson);
      const entityTypes: NLPEntityType[] = JSON.parse(taxonomy.nlpEntityTypesJson);

      // Extract entities
      const extractedEntities: Record<string, string> = {};
      for (const entityType of entityTypes) {
        const regex = new RegExp(entityType.extractionPattern, 'gi');
        const match = userInput.match(regex);
        if (match) {
          extractedEntities[entityType.name] = match[0];
        }
      }

      // Match intent
      for (const intentPattern of intentPatterns) {
        if (intentPattern.confidence < this.minConfidence) continue;

        const intentRegex = new RegExp(
          `\\b(${intentPattern.patterns.join('|')})\\b`,
          'gi'
        );
        const intentMatch = userInput.match(intentRegex);

        if (intentMatch) {
          const hasAllRequired = intentPattern.requiredEntities.every(
            (entity) => entity in extractedEntities
          );

          if (hasAllRequired) {
            const provider = await this.getProvider(taxonomy.providerId);
            if (!provider) continue;

            results.push({
              providerSlug: provider.slug,
              providerId: provider.id,
              taxonomyId: taxonomy.id,
              matchedIntent: intentPattern.intent,
              confidence: intentPattern.confidence,
              extractedEntities,
              capabilities: JSON.parse(taxonomy.capabilitiesJson),
              messageTypes: JSON.parse(taxonomy.messageTypesJson),
              constraints: JSON.parse(taxonomy.constraintsJson),
            });
          }
        }
      }
    }

    // Sort by confidence descending, take top N
    return results
      .sort((a, b) => b.confidence - a.confidence)
      .slice(0, this.maxResults);
  }

  /**
   * Get platform capabilities for a provider.
   */
  async getCapabilities(providerSlug: string): Promise<{
    capabilities: PlatformCapability[];
    messageTypes: string[];
    constraints: PlatformConstraints;
  } | null> {
    const taxonomies = await this.store.getByProviderId(
      (await this.getProviderBySlug(providerSlug))?.id ?? ''
    );

    if (taxonomies.length === 0) return null;

    // Merge capabilities from all taxonomies for this provider
    const allCapabilities: PlatformCapability[] = [];
    const allMessageTypes: string[] = [];
    const constraints: PlatformConstraints = {
      rateLimits: { requests: Infinity, window: 'second' },
      maxMessageLength: Infinity,
      supportedMediaTypes: [],
    };

    for (const taxonomy of taxonomies) {
      allCapabilities.push(...JSON.parse(taxonomy.capabilitiesJson));
      allMessageTypes.push(...JSON.parse(taxonomy.messageTypesJson));

      const taxConstraints: PlatformConstraints = JSON.parse(taxonomy.constraintsJson);
      constraints.maxMessageLength = Math.min(constraints.maxMessageLength, taxConstraints.maxMessageLength);
      constraints.rateLimits = constraints.rateLimits.requests < taxConstraints.rateLimits.requests
        ? constraints.rateLimits
        : taxConstraints.rateLimits;
      constraints.supportedMediaTypes.push(...taxConstraints.supportedMediaTypes);
    }

    // Deduplicate
    const uniqueCapabilities = allCapabilities.filter(
      (cap, i, arr) => arr.findIndex((c) => c.slug === cap.slug) === i
    );
    const uniqueMessageTypes = [...new Set(allMessageTypes)];
    const uniqueMediaTypes = [...new Set(constraints.supportedMediaTypes)];

    return {
      capabilities: uniqueCapabilities,
      messageTypes: uniqueMessageTypes,
      constraints: { ...constraints, supportedMediaTypes: uniqueMediaTypes },
    };
  }

  /**
   * Get discovery hints for DOM automation.
   */
  async getDiscoveryHints(providerSlug: string): Promise<{
    composerSelector: string;
    sendButtonSelector: string;
    messageListSelector: string;
  } | null> {
    const provider = await this.getProviderBySlug(providerSlug);
    if (!provider) return null;

    const taxonomy = await this.store.getByProviderId(provider.id);
    if (taxonomy.length === 0) return null;

    // Use first taxonomy's discovery hints
    return JSON.parse(taxonomy[0].discoveryHintsJson);
  }

  /**
   * Get all taxonomies for a provider by slug.
   */
  private async getTaxonomiesForProvider(slug: string): Promise<ProviderCapabilityTaxonomy[]> {
    const provider = await this.getProviderBySlug(slug);
    if (!provider) return [];
    return this.store.getByProviderId(provider.id);
  }
```

## Acceptance Criteria

1. `ProviderTaxonomyResolver` compiles with zero type errors.
2. `resolveIntent('post a photo to instagram')` returns a result with `providerSlug: 'instagram'`, `matchedIntent: 'post_media'`, `confidence >= 0.8`.
3. `getCapabilities('whatsapp')` returns capabilities including `send_message`, `send_template`, `send_media`.
4. `getDiscoveryHints('telegram')` returns selectors for composer, send button, message list.
5. Results are sorted by confidence descending and limited to `maxResults`.
