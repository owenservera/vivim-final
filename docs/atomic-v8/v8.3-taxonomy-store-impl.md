## Interface

### Impl — `src/storage/impl/provider-taxonomy-store.ts`

```typescript
import { PrismaClient } from '@prisma/client';
import { ulid } from 'ulid';
import type {
  ProviderTaxonomyStore,
  ProviderTaxonomyInput,
  ProviderTaxonomyRow,
} from '../contracts/provider-taxonomy-store.js';

export class PrismaProviderTaxonomyStore implements ProviderTaxonomyStore {
  constructor(private prisma: PrismaClient) {}

  async create(input: ProviderTaxonomyInput): Promise<ProviderTaxonomyRow> {
    const now = Math.floor(Date.now() / 1000);
    return this.prisma.providerCapabilityTaxonomy.create({
      data: {
        id: ulid(),
        providerId: input.providerId,
        platformCategory: input.platformCategory,
        interactionPattern: input.interactionPattern,
        messageTypesJson: input.messageTypesJson ?? '[]',
        capabilitiesJson: input.capabilitiesJson ?? '[]',
        constraintsJson: input.constraintsJson ?? '{}',
        authRequirementsJson: input.authRequirementsJson ?? '[]',
        discoveryHintsJson: input.discoveryHintsJson ?? '{}',
        nlpEntityTypesJson: input.nlpEntityTypesJson ?? '[]',
        nlpIntentPatternsJson: input.nlpIntentPatternsJson ?? '[]',
        seedDataVersion: input.seedDataVersion ?? 1,
        isActive: input.isActive ?? 1,
        createdAt: now,
        updatedAt: now,
      },
    });
  }

  async getById(id: string): Promise<ProviderTaxonomyRow | null> {
    return this.prisma.providerCapabilityTaxonomy.findUnique({ where: { id } });
  }

  async getByProviderId(providerId: string): Promise<ProviderTaxonomyRow[]> {
    return this.prisma.providerCapabilityTaxonomy.findMany({
      where: { providerId },
    });
  }

  async getByCategory(category: string): Promise<ProviderTaxonomyRow[]> {
    return this.prisma.providerCapabilityTaxonomy.findMany({
      where: { platformCategory: category },
    });
  }

  async getByPattern(pattern: string): Promise<ProviderTaxonomyRow[]> {
    return this.prisma.providerCapabilityTaxonomy.findMany({
      where: { interactionPattern: pattern },
    });
  }

  async getByProviderAndCategory(providerId: string, category: string): Promise<ProviderTaxonomyRow | null> {
    return this.prisma.providerCapabilityTaxonomy.findFirst({
      where: { providerId, platformCategory: category },
    });
  }

  async list(opts?: { category?: string; pattern?: string; isActive?: number }): Promise<ProviderTaxonomyRow[]> {
    return this.prisma.providerCapabilityTaxonomy.findMany({
      where: {
        ...(opts?.category ? { platformCategory: opts.category } : {}),
        ...(opts?.pattern ? { interactionPattern: opts.pattern } : {}),
        ...(opts?.isActive !== undefined ? { isActive: opts.isActive } : {}),
      },
    });
  }

  async update(id: string, patch: Partial<Omit<ProviderTaxonomyInput, 'providerId'>>): Promise<ProviderTaxonomyRow> {
    const now = Math.floor(Date.now() / 1000);
    return this.prisma.providerCapabilityTaxonomy.update({
      where: { id },
      data: {
        ...patch,
        updatedAt: now,
      },
    });
  }

  async delete(id: string): Promise<void> {
    await this.prisma.providerCapabilityTaxonomy.delete({ where: { id } });
  }

  async upsertByProviderAndCategory(
    providerId: string,
    category: string,
    pattern: string,
    input: Omit<ProviderTaxonomyInput, 'providerId' | 'platformCategory' | 'interactionPattern'>
  ): Promise<ProviderTaxonomyRow> {
    const now = Math.floor(Date.now() / 1000);
    const existing = await this.prisma.providerCapabilityTaxonomy.findFirst({
      where: { providerId, platformCategory: category },
    });

    if (existing) {
      return this.prisma.providerCapabilityTaxonomy.update({
        where: { id: existing.id },
        data: { ...input, updatedAt: now },
      });
    }

    return this.prisma.providerCapabilityTaxonomy.create({
      data: {
        id: ulid(),
        providerId,
        platformCategory: category,
        interactionPattern: pattern,
        messageTypesJson: input.messageTypesJson ?? '[]',
        capabilitiesJson: input.capabilitiesJson ?? '[]',
        constraintsJson: input.constraintsJson ?? '{}',
        authRequirementsJson: input.authRequirementsJson ?? '[]',
        discoveryHintsJson: input.discoveryHintsJson ?? '{}',
        nlpEntityTypesJson: input.nlpEntityTypesJson ?? '[]',
        nlpIntentPatternsJson: input.nlpIntentPatternsJson ?? '[]',
        seedDataVersion: input.seedDataVersion ?? 1,
        isActive: input.isActive ?? 1,
        createdAt: now,
        updatedAt: now,
      },
    });
  }
}
```

## Acceptance Criteria

1. `PrismaProviderTaxonomyStore` implements all methods of `ProviderTaxonomyStore`.
2. `upsertByProviderAndCategory` creates if not found, updates if found.
3. All methods use ULID for new IDs and unix timestamps for `createdAt`/`updatedAt`.
4. `bun run typecheck` passes.
