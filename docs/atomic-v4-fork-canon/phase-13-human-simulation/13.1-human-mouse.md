# Unit 13.1 — HumanMouseEngine: Bezier-Curve Mouse Movement

**Phase:** 13 | **ID:** 13.1 | **Dependencies:** 11.2

## Problem

Direct `element.click()` via CDP dispatches a click at the exact center of the element with zero mouse movement. Human detection analyzes mouse paths: humans move in curves with acceleration/deceleration, not teleport-to-target.

## Spec

```typescript
// src/engines/stealth/human-mouse-engine.ts

export class HumanMouseModule implements StealthModule {
  name = 'human_mouse'
  detectionVector = 'Mouse movement biometrics (acceleration curves, click distribution)'
  description = 'Moves mouse along bezier curves with human-like acceleration and Gaussian click offset'
  priority = 20
  configSchema = z.object({
    bezierPoints: z.number().min(5).max(100).default(25),
    speedMin: z.number().default(300),   // px/sec min
    speedMax: z.number().default(800),   // px/sec max
    clickOffsetRadius: z.number().default(5), // Gaussian offset from center (pixels)
    overshootProbability: z.number().min(0).max(1).default(0.15),
  })

  async apply(config: Record<string, unknown>, ctx: StealthContext): Promise<void> {
    const cfg = this.configSchema.parse(config)

    const script = `
      (function() {
        var BEZIER_POINTS = ${cfg.bezierPoints};
        var SPEED_MIN = ${cfg.speedMin};
        var SPEED_MAX = ${cfg.speedMax};
        var CLICK_OFFSET = ${cfg.clickOffsetRadius};
        var OVERSHOOT_PROB = ${cfg.overshootProbability};

        // Expose human mouse API for VIVIM commands
        window.__vivimHumanMouse = {
          async moveTo(targetX, targetY) {
            var startX = window.__vivimMouseX || 0;
            var startY = window.__vivimMouseY || 0;
            var dist = Math.sqrt(Math.pow(targetX - startX, 2) + Math.pow(targetY - startY, 2));
            var speed = SPEED_MIN + Math.random() * (SPEED_MAX - SPEED_MIN);
            var durationMs = (dist / speed) * 1000;

            // Generate bezier control points
            var cp1x = startX + (targetX - startX) * 0.3 + (Math.random() - 0.5) * dist * 0.3;
            var cp1y = startY + (targetY - startY) * 0.3 + (Math.random() - 0.5) * dist * 0.3;
            var cp2x = startX + (targetX - startX) * 0.7 + (Math.random() - 0.5) * dist * 0.3;
            var cp2y = startY + (targetY - startY) * 0.7 + (Math.random() - 0.5) * dist * 0.3;

            // Overshoot?
            var endX = targetX, endY = targetY;
            if (Math.random() < OVERSHOOT_PROB) {
              endX = targetX + (Math.random() - 0.5) * 20;
              endY = targetY + (Math.random() - 0.5) * 20;
            }

            // Dispatch mouseMoved events along the curve
            for (var i = 0; i <= BEZIER_POINTS; i++) {
              var t = i / BEZIER_POINTS;
              // Ease (bell curve velocity)
              var easedT = t < 0.5 ? 2 * t * t : -1 + (4 - 2 * t) * t;

              var x = Math.pow(1 - easedT, 3) * startX +
                      3 * Math.pow(1 - easedT, 2) * easedT * cp1x +
                      3 * (1 - easedT) * easedT * easedT * cp2x +
                      Math.pow(easedT, 3) * endX;

              var y = Math.pow(1 - easedT, 3) * startY +
                      3 * Math.pow(1 - easedT, 2) * easedT * cp1y +
                      3 * (1 - easedT) * easedT * easedT * cp2y +
                      Math.pow(easedT, 3) * endY;

              window.__vivimMouseX = x;
              window.__vivimMouseY = y;

              var event = new MouseEvent('mousemove', {
                bubbles: true,
                cancelable: true,
                clientX: x,
                clientY: y,
                screenX: x,
                screenY: y,
              });
              document.elementFromPoint(x, y)?.dispatchEvent(event);

              // Wait proportional to velocity (easedT controls speed)
              var stepDelay = (durationMs / BEZIER_POINTS);
              await new Promise(function(r) { setTimeout(r, stepDelay); });
            }

            // If overshot, correct to actual target
            if (endX !== targetX || endY !== targetY) {
              await new Promise(function(r) { setTimeout(r, 50 + Math.random() * 100); });
              for (var i = 0; i <= 5; i++) {
                var t = i / 5;
                var x = endX + (targetX - endX) * t;
                var y = endY + (targetY - endY) * t;
                var event = new MouseEvent('mousemove', {
                  bubbles: true, cancelable: true, clientX: x, clientY: y,
                });
                document.elementFromPoint(x, y)?.dispatchEvent(event);
                await new Promise(function(r) { setTimeout(r, 20); });
              }
            }
          },

          getClickOffset(centerX, centerY) {
            // Gaussian distribution around center (Box-Muller transform)
            var u1 = Math.random(), u2 = Math.random();
            var z0 = Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
            var z1 = Math.sqrt(-2 * Math.log(u1)) * Math.sin(2 * Math.PI * u2);
            return {
              x: centerX + z0 * CLICK_OFFSET / 2,
              y: centerY + z1 * CLICK_OFFSET / 2,
            };
          }
        };

        window.__vivimMouseX = 0;
        window.__vivimMouseY = 0;
      })();
    `

    await ctx.cdp.send(ctx.slaveId, 'Page.addScriptToEvaluateOnNewDocument', { source: script })
  }

  // Called by ConversationManager instead of direct CDP click:
  async humanClick(ctx: StealthContext, selector: string): Promise<void> {
    // Get element position
    const posResult = await ctx.cdp.send(ctx.slaveId, 'Runtime.evaluate', {
      expression: `(() => {
        const el = document.querySelector(${JSON.stringify(selector)});
        if (!el) return null;
        const r = el.getBoundingClientRect();
        return JSON.stringify({ x: r.x + r.width/2, y: r.y + r.height/2, w: r.width, h: r.height });
      })()`,
      returnByValue: true,
    })

    const pos = JSON.parse((posResult as any)?.result?.value ?? 'null')
    if (!pos) throw new Error(`Element not found: ${selector}`)

    // Get click offset (Gaussian)
    const offsetResult = await ctx.cdp.send(ctx.slaveId, 'Runtime.evaluate', {
      expression: `window.__vivimHumanMouse.getClickOffset(${pos.x}, ${pos.y})`,
      returnByValue: true,
    })
    const offset = (offsetResult as any)?.result?.value ?? { x: pos.x, y: pos.y }

    // Move mouse to target (human path)
    await ctx.cdp.send(ctx.slaveId, 'Runtime.evaluate', {
      expression: `window.__vivimHumanMouse.moveTo(${offset.x}, ${offset.y})`,
      awaitPromise: true,
    })

    // Dispatch click
    await ctx.cdp.send(ctx.slaveId, 'Input.dispatchMouseEvent', {
      type: 'mousePressed', x: offset.x, y: offset.y, button: 'left', clickCount: 1,
    })
    await new Promise(r => setTimeout(r, 50 + Math.random() * 100))
    await ctx.cdp.send(ctx.slaveId, 'Input.dispatchMouseEvent', {
      type: 'mouseReleased', x: offset.x, y: offset.y, button: 'left', clickCount: 1,
    })
  }
}
```

## Test

```typescript
test('mouse movement follows curve (not straight line)', async () => {
  await module.apply({}, ctx)

  // Track mousemove events
  await ctx.cdp.send(slaveId, 'Runtime.evaluate', {
    expression: `window.__mousePath = []; document.addEventListener('mousemove', e => window.__mousePath.push({x: e.clientX, y: e.clientY}))`,
  })

  await module.humanClick(ctx, '#button')

  const path = await ctx.cdp.send(slaveId, 'Runtime.evaluate', {
    expression: 'JSON.stringify(window.__mousePath)',
    returnByValue: true,
  })

  const points = JSON.parse((path as any).result.value)
  expect(points.length).toBeGreaterThan(10) // multiple points along curve
  // Path should not be a straight line
  const isStraight = points.every((p, i) => i === 0 || points[0].x === p.x)
  expect(isStraight).toBe(false)
})
```
