# Unit 12.1 — CanvasNoiseEngine: Canvas Fingerprint Perturbation

**Phase:** 12 | **ID:** 12.1 | **Dependencies:** 11.2

## Problem

Canvas fingerprinting works by asking the browser to draw text/shapes to a canvas and reading back pixel data. The exact pixel output depends on GPU, driver, OS, font rendering — creating a unique fingerprint. Headless/hidden Chrome uses software rendering (SwiftShader), producing a completely different fingerprint from real GPUs.

## Spec

```typescript
// src/engines/stealth/canvas-noise-engine.ts

export class CanvasNoiseModule implements StealthModule {
  name = 'canvas_noise'
  detectionVector = 'Canvas fingerprinting (toDataURL, toBlob, getImageData)'
  description = 'Adds subtle pixel noise to canvas readback to prevent fingerprint matching'
  priority = 10
  configSchema = z.object({
    noiseLevel: z.number().min(0).max(0.1).default(0.01),
    targets: z.array(z.enum(['toDataURL', 'toBlob', 'getImageData'])).default([
      'toDataURL', 'toBlob', 'getImageData',
    ]),
  })

  async apply(config: Record<string, unknown>, ctx: StealthContext): Promise<void> {
    const cfg = this.configSchema.parse(config)
    const noise = cfg.noiseLevel * 255  // scale to pixel range
    const targets = JSON.stringify(cfg.targets)

    const script = `
      (function() {
        var noise = ${noise};
        var targets = ${targets};

        function perturbImageData(imageData) {
          var data = imageData.data;
          for (var i = 0; i < data.length; i += 4) {
            // Add noise to RGB channels only (not alpha)
            data[i]     = Math.max(0, Math.min(255, data[i]     + (Math.random() - 0.5) * noise));
            data[i + 1] = Math.max(0, Math.min(255, data[i + 1] + (Math.random() - 0.5) * noise));
            data[i + 2] = Math.max(0, Math.min(255, data[i + 2] + (Math.random() - 0.5) * noise));
          }
          return imageData;
        }

        if (targets.indexOf('toDataURL') !== -1) {
          var origToDataURL = HTMLCanvasElement.prototype.toDataURL;
          HTMLCanvasElement.prototype.toDataURL = function() {
            var ctx2d = this.getContext('2d');
            if (ctx2d && this.width > 0 && this.height > 0) {
              try {
                var imageData = ctx2d.getImageData(0, 0, this.width, this.height);
                ctx2d.putImageData(perturbImageData(imageData), 0, 0);
              } catch(e) { /* tainted canvas — skip */ }
            }
            return origToDataURL.apply(this, arguments);
          };
        }

        if (targets.indexOf('toBlob') !== -1) {
          var origToBlob = HTMLCanvasElement.prototype.toBlob;
          HTMLCanvasElement.prototype.toBlob = function(callback) {
            var ctx2d = this.getContext('2d');
            if (ctx2d && this.width > 0 && this.height > 0) {
              try {
                var imageData = ctx2d.getImageData(0, 0, this.width, this.height);
                ctx2d.putImageData(perturbImageData(imageData), 0, 0);
              } catch(e) { /* tainted canvas */ }
            }
            return origToBlob.apply(this, arguments);
          };
        }

        if (targets.indexOf('getImageData') !== -1) {
          var origGetImageData = CanvasRenderingContext2D.prototype.getImageData;
          CanvasRenderingContext2D.prototype.getImageData = function() {
            var imageData = origGetImageData.apply(this, arguments);
            return perturbImageData(imageData);
          };
        }
      })();
    `

    await ctx.cdp.send(ctx.slaveId, 'Page.addScriptToEvaluateOnNewDocument', { source: script })
  }

  async verify(ctx: StealthContext): Promise<boolean> {
    const result = await ctx.cdp.send(ctx.slaveId, 'Runtime.evaluate', {
      expression: `(() => {
        const canvas = document.createElement('canvas');
        canvas.width = 16; canvas.height = 16;
        const ctx = canvas.getContext('2d');
        ctx.fillStyle = 'red';
        ctx.fillRect(0, 0, 16, 16);
        const hash1 = canvas.toDataURL();
        const hash2 = canvas.toDataURL();
        return hash1 !== hash2; // noise makes each call different
      })()`,
      returnByValue: true,
    })
    return (result as any)?.result?.value === true
  }
}
```

## Test

```typescript
test('canvas noise makes fingerprints non-deterministic', async () => {
  await module.apply({ noiseLevel: 0.05 }, ctx)

  const result = await ctx.cdp.send(slaveId, 'Runtime.evaluate', {
    expression: `(() => {
      const c = document.createElement('canvas'); c.width=16; c.height=16;
      const x = c.getContext('2d'); x.fillRect(0,0,16,16);
      return [c.toDataURL(), c.toDataURL()];
    })()`,
    returnByValue: true,
  })

  const [hash1, hash2] = (result as any).result.value
  expect(hash1).not.toBe(hash2) // noise changes each call
})
```
