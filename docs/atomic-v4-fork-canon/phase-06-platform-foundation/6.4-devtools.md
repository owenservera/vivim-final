### Events Tab

```tsx
function EventsTab() {
  const [events, setEvents] = useState<any[]>([])
  const wsRef = useRef<WebSocket | null>(null)
  
  useEffect(() => {
    const ws = new WebSocket('ws://localhost:9420/ws')
    ws.onopen = () => {
      ws.send(JSON.stringify({ type: 'subscribe', entityType: '*', entityId: '*' }))
    }
    ws.onmessage = (ev) => {
      const msg = JSON.parse(ev.data)
      setEvents(prev => [{ ...msg, _ts: Date.now() }, ...prev].slice(0, 100))
    }
    wsRef.current = ws
    return () => ws.close()
  }, [])
  
  return (
    <div className="p-2 space-y-1">
      {events.map((ev, i) => (
        <div key={i} className="text-xs font-mono bg-white p-1 rounded truncate">
          <span className="text-gray-400">{new Date(ev._ts).toLocaleTimeString()}</span>
          {' '}
          <span className="text-blue-600">{ev.type}</span>
          {ev.conversationId && <span className="text-gray-400"> {ev.conversationId.slice(0, 8)}</span>}
        </div>
      ))}
    </div>
  )
}
```

### Capabilities Tab

```tsx
function CapabilitiesTab() {
  const [capabilities, setCapabilities] = useState<any[]>([])
  
  useEffect(() => {
    fetch('/api/providers')
      .then(r => r.json())
      .then(providers => {
        if (providers[0]) {
          return fetch(`/api/providers/${providers[0].id}/capabilities`)
        }
      })
      .then(r => r?.json())
      .then(data => setCapabilities(data?.capabilities ?? []))
      .catch(() => {})
  }, [])
  
  return (
    <div className="p-2 space-y-1">
      {capabilities.map(cap => (
        <div key={cap.id} className="bg-white p-2 rounded text-xs">
          <div className="flex justify-between">
            <span className="font-medium">{cap.slug}</span>
            <span className="text-gray-400">{cap.uiPosition}</span>
          </div>
          <div className="text-gray-500">{cap.name}</div>
          <div className="mt-1 flex gap-1">
            <span className="text-gray-400">{cap.uiComponent}</span>
            <span className="text-gray-400">tier: {cap.minPlanTier}</span>
          </div>
        </div>
      ))}
    </div>
  )
}
```

### Fleet Tab

```tsx
function FleetTab() {
  const [slaves, setSlaves] = useState<any[]>([])
  
  useEffect(() => {
    const interval = setInterval(async () => {
      const resp = await fetch('/api/fleet/status')
      const data = await resp.json()
      setSlaves(data)
    }, 5000)
    return () => clearInterval(interval)
  }, [])
  
  return (
    <div className="p-2 space-y-1">
      {slaves.length === 0 && <p className="text-xs text-gray-400">No active slaves</p>}
      {slaves.map(s => (
        <div key={s.slaveId} className="bg-white p-2 rounded text-xs">
          <div className="flex justify-between">
            <span>{s.providerId}/{s.accountId}</span>
            <StatusBadge status={s.status} />
          </div>
          <div className="text-gray-400">
            port:{s.debugPort} pid:{s.pid} circuit:{s.circuitState}
          </div>
        </div>
      ))}
    </div>
  )
}
```

### Health Tab

```tsx
function HealthTab() {
  const [health, setHealth] = useState<any[]>([])
  
  useEffect(() => {
    const interval = setInterval(async () => {
      const resp = await fetch('/api/health/providers')
      if (resp.ok) setHealth(await resp.json())
    }, 10000)
    return () => clearInterval(interval)
  }, [])
  
  return (
    <div className="p-2 space-y-2">
      {health.map(h => (
        <div key={h.providerId} className="bg-white p-2 rounded">
          <div className="flex justify-between items-center">
            <span className="text-sm font-medium">{h.providerId}</span>
            <HealthBadge status={h.status} />
          </div>
          <div className="text-xs text-gray-400 mt-1">Score: {h.score}/100</div>
          {h.signals?.map(sig => (
            <div key={sig.signal} className="text-xs flex justify-between mt-1">
              <span className="text-gray-500">{sig.signal}</span>
              <span>{sig.contribution}/{sig.weight}</span>
            </div>
          ))}
        </div>
      ))}
    </div>
  )
}
```

### Timing Tab

```tsx
function TimingTab() {
  const [timings, setTimings] = useState<Record<string, any>>({})
  
  // Listen for conversation:complete events with timing data
  useEffect(() => {
    const unsubscribe = eventBus.on('conversation:complete', (event) => {
      if (event.timing) {
        setTimings(prev => ({
          ...prev,
          [event.conversationId]: event.timing,
        }))
      }
    })
    return unsubscribe
  }, [])
  
  return (
    <div className="p-2 space-y-2">
      {Object.entries(timings).map(([convId, timing]) => (
        <div key={convId} className="bg-white p-2 rounded text-xs">
          <div className="font-medium">{convId.slice(0, 8)}: {timing.total}ms</div>
          <BudgetBar timing={timing} />
        </div>
      ))}
    </div>
  )
}
```
