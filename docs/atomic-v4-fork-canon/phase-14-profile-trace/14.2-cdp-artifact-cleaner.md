# Unit 14.2 — CDPArtifactCleaner: Remove CDP Traces from Page

**Phase:** 14 | **ID:** 14.2 | **Dependencies:** 11.2

## Problem

CDP leaves detectable artifacts in the page context:
1. `window.cdc_*` variables injected by ChromeDriver/CDP
2. `error.stack` traces include CDP-injected script sources
3. `PerformanceObserver` entries show CDP-related timing
4. `document.$cdc_*` or `document.$wdc_*` variables
5. `Runtime.evaluate` execution context appears in some timing APIs

Advanced detection frameworks (DataDome, PerimeterX, CreepJS) check for these artifacts.

## Spec

```typescript
// src/engines/stealth/cdp-artifact-cleaner.ts

export class CdpArtifactCleanerModule implements StealthModule {
  name = 'cdp_artifact_cleaner'
  detectionVector = 'CDP/WebDriver artifacts in page context (cdc_*, stack traces, performance entries)'
  description = 'Removes CDP-injected variables, cleans error stacks, filters performance entries'
  priority = 1 // must run FIRST — before any page script sees artifacts

  configSchema = z.object({
    removeCdcVars: z.boolean().default(true),
    patchErrorStack: z.boolean().default(true),
    filterPerfEntries: z.boolean().default(true),
    patchRuntimeMarker: z.boolean().default(true),
  })

  async apply(config: Record<string, unknown>, ctx: StealthContext): Promise<void> {
    const cfg = this.configSchema.parse(config)

    const script = `
      (function() {
        ${cfg.removeCdcVars ? this.removeCdcVarsScript() : ''}
        ${cfg.patchErrorStack ? this.patchErrorStackScript() : ''}
        ${cfg.filterPerfEntries ? this.filterPerfEntriesScript() : ''}
        ${cfg.patchRuntimeMarker ? this.patchRuntimeMarkerScript() : ''}
      })();
    `

    await ctx.cdp.send(ctx.slaveId, 'Page.addScriptToEvaluateOnNewDocument', { source: script })
  }

  private removeCdcVarsScript(): string {
    return `
      // Remove cdc_ variables from document and window
      var props = Object.getOwnPropertyNames(document).filter(function(p) {
        return /^\\$?(cdc_|wdc_)/.test(p)
      })
      props.forEach(function(p) {
        try { delete document[p] } catch(e) {
          // If can't delete, overwrite with undefined
          try { Object.defineProperty(document, p, { value: undefined, writable: false }) } catch(e2) {}
        }
      })

      // Same for window
      var winProps = Object.getOwnPropertyNames(window).filter(function(p) {
        return /^\\$?(cdc_|wdc_)/.test(p)
      })
      winProps.forEach(function(p) {
        try { delete window[p] } catch(e) {
          try { Object.defineProperty(window, p, { value: undefined, writable: false }) } catch(e2) {}
        }
      })

      // Periodically check for re-injected variables (CDP may re-inject on navigation)
      setInterval(function() {
        var re = /^\\$?(cdc_|wdc_)/
        Object.getOwnPropertyNames(document).forEach(function(p) {
          if (re.test(p)) {
            try { delete document[p] } catch(e) {}
          }
        })
      }, 1000)
    `
  }

  private patchErrorStackScript(): string {
    return `
      // Patch Error.stack to remove CDP script traces
      // CDP-injected scripts have URLs like "/__puppeteer_evaluation_script__" or empty string
      var origStack = Object.getOwnPropertyDescriptor(Error.prototype, 'stack')

      if (origStack && origStack.get) {
        Object.defineProperty(Error.prototype, 'stack', {
          get: function() {
            var stack = origStack.get.call(this)
            if (!stack || typeof stack !== 'string') return stack
            // Remove lines referencing CDP-injected scripts
            return stack.split('\\n').filter(function(line) {
              return !line.includes('__puppeteer_evaluation_script__') &&
                     !line.includes('__cdp_evaluation_script__') &&
                     !line.includes('debugger eval') &&
                     !line.includes('<anonymous>:0:0') &&
                     !line.match(/^\\s+at\\s+eval\\s+/)  // bare eval traces
            }).join('\\n')
          },
          set: function(val) {
            // Allow manual stack setting
            Object.defineProperty(this, 'stack', { value: val, writable: true, configurable: true })
          }
        })
      }
    `
  }

  private filterPerfEntriesScript(): string {
    return `
      // Patch PerformanceObserver to filter CDP-related entries
      if (typeof PerformanceObserver !== 'undefined') {
        var origObserve = PerformanceObserver.prototype.observe
        PerformanceObserver.prototype.observe = function(options) {
          var self = this
          var origCallback = self.callback

          // Wrap the callback to filter entries
          // Note: PerformanceObserver callback is set in constructor, not observe()
          // So we need a different approach: patch getEntries*

          var origGetEntries = performance.getEntries.bind(performance)
          performance.getEntries = function() {
            return origGetEntries().filter(function(entry) {
              return !entry.name?.includes('__puppeteer') &&
                     !entry.name?.includes('__cdp') &&
                     !entry.name?.includes('devtools')
            })
          }

          var origGetEntriesByType = performance.getEntriesByType.bind(performance)
          performance.getEntriesByType = function(type) {
            return origGetEntriesByType(type).filter(function(entry) {
              return !entry.name?.includes('__puppeteer') &&
                     !entry.name?.includes('__cdp') &&
                     !entry.name?.includes('devtools')
            })
          }

          return origObserve.call(this, options)
        }
      }
    `
  }

  private patchRuntimeMarkerScript(): string {
    return `
      // CDP sets a runtime marker that some detection scripts check
      // The marker is typically: window.outerWidth === 0 (headless) or
      // navigator.webdriver === true (already patched by navigator_patch module)
      //
      // Additional: patch the Permissions API to not reveal automation
      if (navigator.permissions) {
        var origQuery = navigator.permissions.query.bind(navigator.permissions)
        navigator.permissions.query = function(desc) {
          if (desc.name === 'notifications') {
            return Promise.resolve({ state: Notification.permission, onchange: null })
          }
          return origQuery(desc)
        }
      }

      // Patch navigator.connection (sometimes used for fingerprinting)
      if (navigator.connection) {
        Object.defineProperty(navigator.connection, 'rtt', { get: function() { return 50 } })
      }
    `
  }

  async verify(ctx: StealthContext): Promise<boolean> {
    // Check no cdc_ variables exist
    const result = await ctx.cdp.send(ctx.slaveId, 'Runtime.evaluate', {
      expression: `(() => {
        var props = Object.getOwnPropertyNames(document).concat(Object.getOwnPropertyNames(window));
        return props.filter(function(p) { return /^\\$?(cdc_|wdc_)/.test(p) }).length;
      })()`,
      returnByValue: true,
    })
    return (result as any)?.result?.value === 0
  }
}
```

## Test

```typescript
test('no cdc_ variables after cleanup', async () => {
  // Inject fake cdc_ variable (simulating CDP injection)
  await ctx.cdp.send(slaveId, 'Runtime.evaluate', {
    expression: 'document.$cdc_asdjflk = "test"',
  })

  await module.apply({}, ctx)

  // Navigate to trigger new document context
  await ctx.cdp.send(slaveId, 'Page.navigate', { url: 'about:blank' })
  await Bun.sleep(500)

  const ok = await module.verify!(ctx)
  expect(ok).toBe(true)
})

test('error stack is cleaned', async () => {
  await module.apply({}, ctx)
  await ctx.cdp.send(slaveId, 'Page.navigate', { url: 'about:blank' })
  await Bun.sleep(500)

  const result = await ctx.cdp.send(slaveId, 'Runtime.evaluate', {
    expression: `(() => {
      try { throw new Error('test') } catch(e) { return e.stack }
    })()`,
    returnByValue: true,
  })
  const stack = (result as any)?.result?.value ?? ''
  expect(stack).not.toContain('__puppeteer')
  expect(stack).not.toContain('__cdp')
})
```
