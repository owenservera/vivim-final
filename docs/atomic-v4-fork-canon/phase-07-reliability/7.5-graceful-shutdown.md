# Unit 7.5 — Graceful Chrome Shutdown on SIGTERM

**Phase:** 7 | **ID:** 7.5 | **Dependencies:** 1.1

## Problem

`server/index.ts:56` has a `gracefulShutdown()` function with shutdown hooks, but Governor's `killAll()` is never registered as a hook. SIGTERM leaves zombie Chrome processes consuming memory and debug ports.

## Re-Programmable Design

A **ShutdownPolicy** controls how Chrome processes are terminated:

```typescript
interface ShutdownPolicy {
  chromeTermination: 'kill_all' | 'leave_running' | 'graceful_close'
  gracefulTimeoutMs: number    // wait this long for Chrome to close naturally
  closeTabsFirst: boolean      // close all tabs before killing process
  saveProfileOnExit: boolean   // flush profile to disk before kill
}
```

### Implementation

```typescript
// In createServerWithEngines(), after governor boot:

const shutdownPolicy = await policyStore.get<ShutdownPolicy>('server.shutdown')

onShutdown(async () => {
  console.log('[shutdown] Terminating Chrome fleet...')

  switch (shutdownPolicy?.chromeTermination ?? 'graceful_close') {
    case 'leave_running':
      console.log('[shutdown] Leaving Chrome instances running')
      return

    case 'graceful_close': {
      // Close all tabs first (gives Chrome time to save session)
      if (shutdownPolicy?.closeTabsFirst !== false) {
        const allSlaves = governor.getAllSlaves()
        for (const slave of allSlaves) {
          try {
            const targets = await governor.cdp.send(slave.slaveId, 'Target.getTargets')
            for (const target of (targets as any)?.targetInfos ?? []) {
              if (target.type === 'page') {
                await governor.cdp.send(slave.slaveId, 'Target.closeTarget', { targetId: target.targetId })
              }
            }
          } catch { /* slave may already be dead */ }
        }
        // Wait for graceful close
        await Bun.sleep(shutdownPolicy?.gracefulTimeoutMs ?? 3000)
      }
      await governor.killAll()
      break
    }

    case 'kill_all':
    default:
      await governor.killAll()
  }

  // Disconnect CDP transport
  await cdpTransport.disconnectAll()

  // Flush profile metadata
  if (shutdownPolicy?.saveProfileOnExit !== false) {
    for (const slave of governor.getAllSlaves()) {
      try {
        await profileAllocator.release(slave.providerId, slave.accountId)
      } catch { /* best effort */ }
    }
  }

  console.log('[shutdown] Chrome fleet terminated')
})

// Also register for SIGTERM and SIGINT (already done in server/index.ts)
// But add Windows-specific CTRL_CLOSE_EVENT handler:
if (process.platform === 'win32') {
  // On Windows, SIGTERM doesn't work the same way
  // Use 'beforeExit' as additional safety net
  process.on('beforeExit', () => {
    void gracefulShutdown('beforeExit')
  })
}
```

### FleetSupervisor.killAll Enhancement

```typescript
async killAll(): Promise<void> {
  const killPromises: Promise<void>[] = []

  for (const id of this.instances.keys()) {
    killPromises.push(
      this.kill(id).catch(err => {
        console.error(`[fleet] Failed to kill ${id}:`, err)
      })
    )
  }

  // Wait for all kills with timeout
  await Promise.race([
    Promise.allSettled(killPromises),
    new Promise(resolve => setTimeout(resolve, 10_000)), // 10s max
  ])

  // Force kill any remaining
  for (const [id, instance] of this.instances) {
    if (instance.pid) {
      try {
        await killChrome(instance.pid)
      } catch { /* already dead */ }
    }
  }

  this.instances.clear()
  this.stopHealthProbe()
}
```

## Test

```typescript
test('SIGTERM kills all Chrome processes', async () => {
  await governor.spawn('chatgpt', 'work')
  await governor.spawn('claude', 'personal')

  const slaves = governor.getAllSlaves()
  expect(slaves.length).toBe(2)

  // Simulate SIGTERM
  process.emit('SIGTERM')

  // Wait for shutdown
  await delay(5000)

  // Verify no Chrome processes remain
  for (const slave of slaves) {
    if (slave.pid) {
      expect(await isChromeRunning(slave.pid)).toBe(false)
    }
  }
})
```
