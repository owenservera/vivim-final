# Unit 7.3 — Conversation Locking: Configurable Lock Policy

**Phase:** 7 | **ID:** 7.3 | **Dependencies:** 2.1

## Problem

Two concurrent `send()` calls to the same conversation will both try to type into the same Chrome textarea simultaneously, corrupting the message.

## Re-Programmable Design

A **LockPolicy** controls how conversation access is serialized:

```typescript
interface LockPolicy {
  scope: 'conversation' | 'provider' | 'global'  // lock granularity
  strategy: 'mutex' | 'queue' | 'reject'          // what to do with concurrent access
  timeoutMs: number                                 // how long to wait for lock
  queueMaxSize: number                              // max queued requests (for 'queue' strategy)
}
```

### Lock Manager

```typescript
// New file: src/engines/lock-manager.ts

export class LockManager {
  private locks = new Map<string, LockEntry>()
  private queues = new Map<string, Array<{ resolve: () => void; reject: (e: Error) => void }>>()
  private policy: LockPolicy

  constructor(
    private policyStore: PolicyStore,
  ) {
    this.policy = { scope: 'conversation', strategy: 'mutex', timeoutMs: 60_000, queueMaxSize: 5 }
    void this.loadPolicy()
  }

  private async loadPolicy(): Promise<void> {
    const stored = await this.policyStore.get<LockPolicy>('conversation.lock')
    if (stored) this.policy = stored
  }

  async acquire(key: string): Promise<LockHandle> {
    const lockKey = this.resolveKey(key)

    // Check if already locked
    const existing = this.locks.get(lockKey)
    if (!existing) {
      // Acquire immediately
      this.locks.set(lockKey, { acquiredAt: Date.now(), holder: key })
      return this.createHandle(lockKey)
    }

    // Already locked — apply strategy
    switch (this.policy.strategy) {
      case 'reject':
        throw new EngineError(`Resource locked: ${lockKey}`)

      case 'queue': {
        const queue = this.queues.get(lockKey) ?? []
        if (queue.length >= this.policy.queueMaxSize) {
          throw new EngineError(`Queue full for: ${lockKey}`)
        }
        return new Promise((resolve, reject) => {
          const timer = setTimeout(() => {
            reject(new EngineError(`Lock timeout after ${this.policy.timeoutMs}ms: ${lockKey}`))
          }, this.policy.timeoutMs)

          queue.push({
            resolve: () => {
              clearTimeout(timer)
              this.locks.set(lockKey, { acquiredAt: Date.now(), holder: key })
              resolve(this.createHandle(lockKey))
            },
            reject: (e) => {
              clearTimeout(timer)
              reject(e)
            },
          })
          this.queues.set(lockKey, queue)
        })
      }

      case 'mutex':
      default: {
        // Same as queue but with timeout
        return new Promise((resolve, reject) => {
          const timer = setTimeout(() => {
            reject(new EngineError(`Lock timeout after ${this.policy.timeoutMs}ms: ${lockKey}`))
          }, this.policy.timeoutMs)

          const check = () => {
            if (!this.locks.has(lockKey)) {
              clearTimeout(timer)
              this.locks.set(lockKey, { acquiredAt: Date.now(), holder: key })
              resolve(this.createHandle(lockKey))
            } else {
              setTimeout(check, 50)
            }
          }
          check()
        })
      }
    }
  }

  private resolveKey(key: string): string {
    // Scope determines lock granularity
    switch (this.policy.scope) {
      case 'global': return 'global'
      case 'provider': return key.split(':')[0] ?? key  // providerId
      case 'conversation':
      default: return key
    }
  }

  private createHandle(lockKey: string): LockHandle {
    return {
      release: () => {
        this.locks.delete(lockKey)
        // Process queue
        const queue = this.queues.get(lockKey)
        if (queue && queue.length > 0) {
          const next = queue.shift()!
          next.resolve()
        }
      },
    }
  }
}

interface LockEntry {
  acquiredAt: number
  holder: string
}

interface LockHandle {
  release(): void
}
```

### Integration with ConversationManager

```typescript
// ConversationManager.send():
async send(conversationId: string, message: string): Promise<SendResult> {
  const lock = await this.lockManager.acquire(conversationId)
  try {
    return await this.sendInternal(conversationId, message)
  } finally {
    lock.release()
  }
}
```

test('concurrent sends are serialized', async () => {
  const p1 = cm.send(conv.id, 'Message 1')
  const p2 = cm.send(conv.id, 'Message 2')

  const [r1, r2] = await Promise.all([p1, p2])
  // Both should succeed, but executed sequentially
  expect(r1.ok).toBe(true)
  expect(r2.ok).toBe(true)
})

test('reject strategy throws on concurrent access', async () => {
  // Configure policy: strategy = 'reject'
  await policyStore.set('conversation.lock', { strategy: 'reject', ... })

  const p1 = cm.send(conv.id, 'Message 1')
  await expect(cm.send(conv.id, 'Message 2')).rejects.toThrow('Resource locked')

  await p1 // cleanup
})
