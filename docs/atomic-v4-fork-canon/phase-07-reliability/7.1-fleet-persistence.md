# Unit 7.1 — Fleet State Persistence: Survive Server Restart

**Phase:** 7 | **ID:** 7.1 | **Dependencies:** 1.7

## Problem

`FleetSupervisor.instances` is a `new Map()` in memory (`fleet-supervisor.ts:97`). When the server restarts, all instance tracking is lost. Chrome processes keep running but become invisible to the supervisor.

## Re-Programmable Design

A **FleetPersistencePolicy** controls how fleet state is persisted:

```typescript
interface FleetPersistencePolicy {
  backend: 'sqlite' | 'file' | 'memory'  // configurable
  syncMode: 'eager' | 'lazy' | 'batched' // when to write
  batchIntervalMs: number                // for batched mode
  restoreOnBoot: boolean                 // auto-restore on restart
  staleAfterMs: number                   // instances older than this are ignored
}
```

### Fleet State Table

```sql
CREATE TABLE fleet_instance_state (
  id TEXT PRIMARY KEY,
  provider_slug TEXT NOT NULL,
  account_id TEXT NOT NULL,
  debug_port INTEGER NOT NULL,
```

### Implementation

```typescript
// FleetSupervisor modifications:

private async persistInstance(instance: FleetInstance): Promise<void> {
  const policy = await this.policyStore.get<FleetPersistencePolicy>('fleet.persistence')
  if (!policy || policy.backend === 'memory') return

  await this.store.upsertFleetState({
    id: instance.id,
    providerSlug: instance.providerSlug,
    accountId: instance.accountId,
    debugPort: instance.debugPort,
    profileDir: instance.profileDir,
    status: instance.status,
    pid: instance.pid,
    consecutiveFailures: instance.consecutiveFailures,
    createdAt: instance.createdAt,
    lastSeenAt: Date.now(),
  })
}

// On any state change, persist (eager) or queue (batched)
private onInstanceChanged(instance: FleetInstance): void {
  const mode = this.policy?.syncMode ?? 'lazy'
  switch (mode) {
    case 'eager': void this.persistInstance(instance); break
    case 'batched': this.batchQueue.add(instance); break
    case 'lazy': this.dirty.add(instance.id); break
  }
}

// On boot, restore known instances
async boot(): Promise<void> {
  const policy = await this.policyStore.get<FleetPersistencePolicy>('fleet.persistence')
  if (policy?.restoreOnBoot !== false) {
    await this.restoreInstances()
  }
  await this.portReaper.reap(this.opts.portRange)
}

private async restoreInstances(): Promise<void> {
  const staleAfter = this.policy?.staleAfterMs ?? 300_000 // 5 min default
  const cutoff = Date.now() - staleAfter
  const persisted = await this.store.getRecentFleetState(cutoff)

  for (const row of persisted) {
    // Check if Chrome process is actually alive
    if (row.pid && await isChromeRunning(row.pid)) {
      // Adopt: re-add to in-memory map
      this.instances.set(row.id, {
        id: row.id,
        providerSlug: row.providerSlug,
        accountId: row.accountId,
        debugPort: row.debugPort,
        profileDir: row.profileDir,
        status: 'running',
        pid: row.pid,
        consecutiveFailures: row.consecutiveFailures,
        lastHealthCheck: Date.now(),
        createdAt: row.createdAt,
      })
      this.portReaper.trackPid(row.debugPort, row.pid)
      console.log(`[fleet] Adopted instance ${row.id} (pid=${row.pid})`)
    } else {
      // Stale — mark as stopped, clean up DB row
      await this.store.updateFleetState(row.id, { status: 'stopped', pid: null })
    }
  }
}
```

### Store Contract

```typescript
interface FleetPersistenceStore {
  upsertFleetState(input: FleetStateInput): Promise<void>
  getRecentFleetState(since: number): Promise<FleetStateRow[]>
  updateFleetState(id: string, patch: Record<string, unknown>): Promise<void>
```

## Test

```typescript
test('fleet state survives restart', async () => {
  await supervisor.spawn('chatgpt', 'work')
  const beforeRestart = supervisor.getAllInstances()
  expect(beforeRestart).toHaveLength(1)

  // Simulate restart: create new supervisor with same store
  const newSupervisor = new FleetSupervisor(store, opts)
  await newSupervisor.boot()

  const afterRestart = newSupervisor.getAllInstances()
  expect(afterRestart).toHaveLength(1)
  expect(afterRestart[0].providerSlug).toBe('chatgpt')
  expect(afterRestart[0].status).toBe('running')
})
```
