# Unit 10.2 — Loading + Skeleton States for All Async Surfaces

**Phase:** 10 | **ID:** 10.2 | **Dependencies:** 5.1

## Problem

Async data fetches show nothing while loading. The user can't tell if the app is working or frozen.

### Loading State Pattern

```tsx
// Async data component with loading state:
function ConversationSurface({ conversationId }: { conversationId: string }) {
  const { data: messages, loading, error } = useAsync(
    () => fetch(`/api/conversations/${conversationId}/messages`).then(r => r.json()),
    [conversationId]
  )

  if (loading) {
    return (
      <div className="flex-1 p-4">
        <MessageSkeleton />
        <MessageSkeleton />
      </div>
    )
  }

  if (error) {
    return (
      <div className="flex-1 p-4 text-center text-gray-500">
        <p>Failed to load messages</p>
        <button onClick={retry} className="text-blue-500">Retry</button>
      </div>
    )
  }

  return (
    <div className="flex-1 overflow-y-auto px-4 py-2">
      {messages.map(msg => <MessageBubble key={msg.id} message={msg} />)}
    </div>
  )
}
```

### useAsync Hook

```tsx
// web/sandbox/src/hooks/use-async.ts

import { useState, useEffect, useCallback } from 'react'

interface AsyncState<T> {
  data: T | null
  loading: boolean
  error: Error | null
}

export function useAsync<T>(
  fn: () => Promise<T>,
  deps: unknown[],
): AsyncState<T> & { retry: () => void } {
  const [state, setState] = useState<AsyncState<T>>({
    data: null,
    loading: true,
    error: null,
  })
  const [retryCount, setRetryCount] = useState(0)

  const execute = useCallback(() => {
    setState(prev => ({ ...prev, loading: true, error: null }))
    fn()
      .then(data => setState({ data, loading: false, error: null }))
      .catch(error => setState({ data: null, loading: false, error }))
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, deps)

  useEffect(() => {
    execute()
  }, [execute, retryCount])

  return {
    ...state,
    retry: () => setRetryCount(c => c + 1),
  }
}
```
