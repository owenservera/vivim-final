# Unit 5.3 — Virtual Scrolling for Long Conversations

**Phase:** 5 | **ID:** 5.3 | **Dependencies:** 5.2

## Problem

Conversations with 100+ messages cause slow initial render and janky scrolling because all message DOM nodes are present simultaneously.

## Spec

### Windowing Strategy

Only render messages within the viewport ± 5 messages (overscan):

```tsx
import { useRef, useState, useEffect, useMemo } from 'react'

const ITEM_HEIGHT_ESTIMATE = 120  // estimated px per message
const OVERSCAN = 5  // extra messages above/below viewport

export function VirtualMessageList({ messages }: { messages: Message[] }) {
  const containerRef = useRef<HTMLDivElement>(null)
  const [scrollTop, setScrollTop] = useState(0)
  const [viewportHeight, setViewportHeight] = useState(600)
  
  useEffect(() => {
    const el = containerRef.current
    if (!el) return
    
    const onScroll = () => setScrollTop(el.scrollTop)
    const onResize = () => setViewportHeight(el.clientHeight)
    
    el.addEventListener('scroll', onScroll, { passive: true })
    onResize()
    
    return () => {
      el.removeEventListener('scroll', onScroll)
    }
  }, [])
  
  const { startIndex, endIndex, totalHeight, offsetY } = useMemo(() => {
    const total = messages.length * ITEM_HEIGHT_ESTIMATE
    const start = Math.max(0, Math.floor(scrollTop / ITEM_HEIGHT_ESTIMATE) - OVERSCAN)
    const visibleCount = Math.ceil(viewportHeight / ITEM_HEIGHT_ESTIMATE) + OVERSCAN * 2
    const end = Math.min(messages.length, start + visibleCount)
    const offset = start * ITEM_HEIGHT_ESTIMATE
    
    return {
      startIndex: start,
      endIndex: end,
      totalHeight: total,
      offsetY: offset,
    }
  }, [scrollTop, viewportHeight, messages.length])
  
  const visibleMessages = messages.slice(startIndex, endIndex)
  
  return (
    <div
      ref={containerRef}
      className="flex-1 overflow-y-auto"
      style={{ position: 'relative' }}
    >
      {/* Spacer to create scroll height */}
      <div style={{ height: totalHeight, position: 'relative' }}>
        {/* Rendered messages */}
        <div style={{ transform: `translateY(${offsetY}px)` }}>
          {visibleMessages.map((msg, i) => (
            <MessageBubble
              key={msg.id}
              message={msg}
              style={{ minHeight: ITEM_HEIGHT_ESTIMATE }}
            />
          ))}
        </div>
      </div>
    </div>
  )
}
```

### Auto-Scroll to Bottom

```tsx
const [autoScroll, setAutoScroll] = useState(true)

useEffect(() => {
  if (!autoScroll || !containerRef.current) return
  containerRef.current.scrollTop = containerRef.current.scrollHeight
}, [messages.length, streamingMessage])

// Detect manual scroll up
const onScroll = () => {
  const el = containerRef.current
  if (!el) return
  setScrollTop(el.scrollTop)
  const atBottom = el.scrollHeight - el.scrollTop - el.clientHeight < 50
  setAutoScroll(atBottom)
}
```

### Streaming Message Position

When streaming, the streaming message should always be visible at the bottom:

```tsx
{streamingMessage && (
  <div style={{ position: 'sticky', bottom: 0 }}>
    <StreamingMessageBubble message={streamingMessage} />
  </div>
)}
```

## Performance Target

| Messages | DOM Nodes (before) | DOM Nodes (after) | FPS |
|----------|-------------------|-------------------|-----|
| 50 | 50 | ~15 | 60 |
| 200 | 200 | ~15 | 60 |
| 1000 | 1000 | ~15 | 60 |

## Test

```typescript
test('virtual list renders only visible items', () => {
  const messages = Array.from({ length: 1000 }, (_, i) => ({
    id: `msg_${i}`,
    role: i % 2 ? 'assistant' : 'user',
    content: `Message ${i}`,
    createdAt: Date.now() + i,
  }))
  
  render(<VirtualMessageList messages={messages} />)
  
  // Should render far fewer than 1000 DOM nodes
  const messageNodes = screen.queryAllByTestId('message-bubble')
  expect(messageNodes.length).toBeLessThan(30)
})
```
