# 16.1 — Kernel REST API

**Phase:** 16 · **Domain:** Kernel Surfaces · **Status:** DRAFT

## Purpose

HTTP routes that expose kernel and oracle data. Provides the primary API surface for
external tools, dashboards, and the frontend.

## Interface

```
GET  /api/kernel/topology              → SystemTopology
GET  /api/kernel/health                → HealthSnapshot
GET  /api/kernel/engines               → EngineDescriptor[]
GET  /api/kernel/engines/:id           → EngineDescriptor
GET  /api/kernel/trace/:traceId        → TraceSpan[]
GET  /api/kernel/provenance/:traceId   → CausalNode[]
GET  /api/kernel/spans?limit=N         → TraceSpan[]
GET  /api/kernel/config/:engineId      → Record<string, unknown>
PUT  /api/kernel/config/:engineId      → { ok: true }
POST /api/kernel/diagnose              → DiagnosticIssue[]
POST /api/kernel/diagnose/:engineId    → DiagnosticIssue[]
POST /api/kernel/heal                  → HealAction
POST /api/kernel/heal/:issueId         → HealAction
GET  /api/kernel/oracle?q=...          → QueryResult
GET  /api/kernel/events?limit=N        → OracleEvent[]
```

## Design

### Route Registration

```typescript
// src/server/routes/kernel-routes.ts
import { Router } from 'express'  // or whatever HTTP lib

export function createKernelRoutes(kernel: Kernel): Router {
  const router = Router()

  router.get('/topology', async (req, res) => {
    const topology = kernel.registry.describe()
    res.json(topology)
  })

  router.get('/health', async (req, res) => {
    const health = await kernel.oracle.health()
    res.json(health)
  })

  router.get('/engines', async (req, res) => {
    const engines = kernel.registry.listEngines(req.query)
    res.json(engines)
  })

  router.get('/engines/:id', async (req, res) => {
    const engine = kernel.registry.getEngine(req.params.id)
    if (!engine) return res.status(404).json({ error: 'Engine not found' })
    res.json(engine)
  })

  router.get('/trace/:traceId', async (req, res) => {
    const spans = kernel.tracer.getTrace(req.params.traceId)
    res.json(spans)
  })

  router.get('/provenance/:traceId', async (req, res) => {
    const chain = kernel.provenance.getChain(req.params.traceId)
    res.json(chain)
  })

  router.get('/spans', async (req, res) => {
    const limit = parseInt(req.query.limit as string) || 50
    const spans = kernel.tracer.getRecentSpans(limit)
    res.json(spans)
  })

  router.get('/config/:engineId', async (req, res) => {
    const config = kernel.config.readScoped(req.params.engineId)
    res.json(config)
  })

  router.put('/config/:engineId', async (req, res) => {
    await kernel.config.updateScoped(req.params.engineId, req.body)
    res.json({ ok: true })
  })

  router.post('/diagnose', async (req, res) => {
    const issues = await kernel.oracle.diagnose()
    res.json(issues)
  })

  router.post('/diagnose/:engineId', async (req, res) => {
    const issues = await kernel.oracle.diagnoseEngine(req.params.engineId)
    res.json(issues)
  })

  router.post('/heal', async (req, res) => {
    const action = await kernel.oracle.heal(req.body.issueId)
    res.json(action)
  })

  router.post('/heal/:issueId', async (req, res) => {
    const action = await kernel.oracle.heal(req.params.issueId)
    res.json(action)
  })

  router.get('/oracle', async (req, res) => {
    const result = await kernel.oracle.query({ type: 'all', filter: { q: req.query.q } })
    res.json(result)
  })

  router.get('/events', async (req, res) => {
    const limit = parseInt(req.query.limit as string) || 50
    const events = kernel.oracle.events.getRecentEvents(limit)
    res.json(events)
  })

  return router
}
```

### Integration

```typescript
// src/server/index.ts — in createServerWithEngines():
const kernelRoutes = createKernelRoutes(kernel)
app.use('/api/kernel', kernelRoutes)
```
