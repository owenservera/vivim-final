## Context

Same first-class-local model treatment as 5.1 but for `llama.cpp --server`, which speaks an **OpenAI-compatible** HTTP protocol (different response envelope than Ollama's `/api/generate`). The adapter parses the OpenAI-style `{ choices: [{ message: { content } }] }` shape.

## Existing Baseline

- `src/engines/local-model-adapter.ts` — `LocalModelProvider` already enumerates `'llamacpp'`; we extend `generate()` to detect the OpenAI-compat envelope, or add a `generateOpenAi(prompt)` method.
- `src/engines/provider-registrar.ts` + `seeds/providers/` — `llamacpp.json` follows `ProviderManifestSchema`.
- `OllamaProviderPlugin` (5.1) — `LlamacppProviderPlugin` mirrors it (shared `ProviderPlugin` interface).
- Default endpoint `http://localhost:8080` (llama.cpp default).

### Extension in `src/engines/local-model-adapter.ts`

```typescript
export interface LocalModelResponse {
  ok: boolean
  response: string
  model: string
  // ... existing fields ...
}

// OpenAI-compatible envelope parser (llama.cpp --server)
async generateOpenAi(prompt: string): Promise<LocalModelResponse> {
  const res = await fetch(`${this.config.endpoint}/v1/chat/completions`, {
    method: 'POST',
    headers: { 'content-type': 'application/json' },
    body: JSON.stringify({
      model: this.config.model,
      messages: [{ role: 'user', content: prompt }],
      max_tokens: this.config.maxTokens,
      temperature: this.config.temperature,
    }),
  })
  const data = (await res.json()) as { choices?: Array<{ message?: { content?: string } }> }
  const text = data.choices?.[0]?.message?.content ?? ''
  return { ok: res.ok, response: text, model: this.config.model, promptEvalCount: 0, evalCount: 0, durationMs: 0 }
}
```

### New file: `src/engines/provider-plugin-llamacpp.ts`

```typescript
import { LocalModelAdapter } from './local-model-adapter.js'
import type { ProviderPlugin } from './provider-plugin-ollama.js'

export class LlamacppProviderPlugin implements ProviderPlugin {
  readonly providerId = 'llamacpp'
  private adapter: LocalModelAdapter
  constructor(endpoint = 'http://localhost:8080', model = 'default') {
    this.adapter = new LocalModelAdapter({
      provider: 'llamacpp', endpoint, model,
      timeoutMs: 60_000, maxTokens: 4096, temperature: 0.7,
    })
  }
  async onAction(action: string, input: Record<string, unknown>): Promise<unknown> {
    if (action === 'send_message') return this.adapter.generateOpenAi(String(input.message ?? ''))
    if (action === 'select_model') { /* reconfigure */ return { ok: true } }
    throw new EngineError(`llamacpp plugin: unsupported action ${action}`)
  }
}
```
