**Produces:** `workspace_layout` + `workspace_panel` tables; `WorkspaceManager.{getLayout, setLayout, openPanel, closePanel, rearrange}`.

---

## Context

The workspace is the user-facing shell: a persisted, ordered tree of panels. Each panel points at either a `canvasInstanceId` (a live canvas from 3.3) or a `builtinSurfaceId` (one of the surfaces in 4.4–4.10, e.g. `conversation-list`, `memory-browser`). The layout is per-user and changes emit `workspace:layout_changed` so the frontend (4.3) and agent (4.11) stay in sync.

### New file: `src/engines/workspace-manager.ts`

```typescript
import type { WorkspaceStore } from '../storage/contracts/workspace-store.js'
import type { CapabilityEventBus } from './capability-event-bus.js'
import { EngineError } from '../errors.js'
import { newId } from '../ids.js'

export interface PanelSpec {
  kind: 'canvas' | 'builtin'
  canvasInstanceId?: string
  builtinSurfaceId?: string
  title?: string
  parentPanelId?: string | null
}

export class WorkspaceManager {
  constructor(private store: WorkspaceStore, private bus?: CapabilityEventBus) {}

  async getLayout(userId: string, name = 'default'): Promise<WorkspaceLayoutRow | null> {
    return this.store.getLayout(userId, name)
  }

  async setLayout(userId: string, panels: PanelSpec[], name = 'default'): Promise<WorkspaceLayoutRow> {
    const existing = await this.store.getLayout(userId, name)
    const layoutId = existing?.id ?? newId()
    if (existing) {
      for (const p of existing.panels) await this.store.deletePanel(p.id)
    }
    const created: WorkspacePanelRow[] = []
    for (let i = 0; i < panels.length; i++) {
      created.push(await this.store.createPanel({ id: newId(), layoutId, position: i, ...panels[i] }))
    }
    const layout: WorkspaceLayoutRow = {
      id: layoutId, userId, name, isDefault: name === 'default',
      panels: created, createdAt: existing?.createdAt ?? Date.now(), updatedAt: Date.now(),
    }
    await this.store.upsertLayout(layout)
    this.bus?.emit('workspace:layout_changed', { userId, layout })
    return layout
  }

  async openPanel(userId: string, spec: PanelSpec, name = 'default'): Promise<WorkspacePanelRow> {
    const layout = await this.requireLayout(userId, name)
    const panel = await this.store.createPanel({
      id: newId(), layoutId: layout.id, position: layout.panels.length, ...spec,
    })
    this.bus?.emit('workspace:layout_changed', { userId, layout: await this.store.getLayout(userId, name) })
    return panel
  }

  async closePanel(panelId: string): Promise<void> {
    await this.store.deletePanel(panelId)
  }

  async rearrange(userId: string, orderedPanelIds: string[], name = 'default'): Promise<WorkspaceLayoutRow> {
    const layout = await this.requireLayout(userId, name)
    const byId = new Map(layout.panels.map((p) => [p.id, p]))
    if (orderedPanelIds.length !== layout.panels.length) {
      throw new EngineError('rearrange must include all panel ids')
    }
    for (let i = 0; i < orderedPanelIds.length; i++) {
      const p = byId.get(orderedPanelIds[i])
      if (!p) throw new EngineError(`Unknown panel ${orderedPanelIds[i]}`)
      await this.store.updatePanel(p.id, { position: i })
    }
    const updated = await this.store.getLayout(userId, name)
    if (updated) this.bus?.emit('workspace:layout_changed', { userId, layout: updated })
    return (await this.store.getLayout(userId, name))!
  }

  private async requireLayout(userId: string, name: string): Promise<WorkspaceLayoutRow> {
    const layout = await this.store.getLayout(userId, name)
    if (!layout) throw new EngineError(`No workspace layout ${name} for ${userId}`)
    return layout
  }
}
```

## Acceptance Criteria

1. `setLayout` persists panels; `getLayout` returns them with `position` order intact.
2. `openPanel` appends a panel and emits `workspace:layout_changed`.
3. `closePanel` removes the panel; `rearrange` reorders and validates the full id set.
4. A panel with `kind:'canvas'` must carry `canvasInstanceId`; a `builtin` panel carries `builtinSurfaceId` (store-level guard).
5. `bun run devops gate` passes.
