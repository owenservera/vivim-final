# 16.4 — Kernel Frontend Surface

**Phase:** 16 · **Domain:** Kernel Surfaces · **Status:** DRAFT

## Purpose

React component for the Oracle dashboard in the VIVIM UI. Provides visual access to
system topology, health, diagnostics, traces, provenance, and config.

## Interface

### Routes

```
/oracle                    → OracleDashboard (main layout)
/oracle/topology           → TopologyView (force-directed graph)
/oracle/diagnostics        → DiagnosticsView (issue list)
/oracle/traces             → TraceExplorer (trace list + span tree)
/oracle/config/:engineId   → ConfigEditor (JSON editor)
```

### TopologyView

```typescript
// Force-directed graph of engines + dependency edges
// Nodes: engines (colored by health status)
// Edges: dependency relationships
// Click node → show engine details panel

export function TopologyView() {
  const [topology, setTopology] = useState<TopologyDescription>()

  useEffect(() => {
    fetch('/api/kernel/topology').then(r => r.json()).then(setTopology)
  }, [])

  return (
    <div className="topology-view">
      <h2>System Topology</h2>
      <ForceGraph
        nodes={topology?.engines.map(e => ({
          id: e.id,
          label: e.id,
          color: healthColor(e.health?.status),
          size: e.dependencies.length + 1
        }))}
        edges={topology?.dependencyGraph}
        onNodeClick={(id) => navigate(`/oracle/config/${id}`)}
      />
    </div>
  )
}
```

### DiagnosticsView

```typescript
// Filterable issue list with severity badges
// Severity: critical (red), warning (yellow), info (blue)
// Click issue → show details + suggested fix + heal button

export function DiagnosticsView() {
  const [issues, setIssues] = useState<DiagnosticIssue[]>()
  const [filter, setFilter] = useState<string>('all')

  useEffect(() => {
    fetch('/api/kernel/diagnose', { method: 'POST' })
      .then(r => r.json()).then(setIssues)
  }, [])

  const filtered = issues?.filter(i => filter === 'all' || i.severity === filter)

  return (
    <div className="diagnostics-view">
      <h2>Diagnostic Issues</h2>
      <FilterBar value={filter} onChange={setFilter} options={['all', 'critical', 'warning', 'info']} />
      <IssueList issues={filtered} onHeal={handleHeal} />
    </div>
  )
}
```

### ConfigEditor

```typescript
// JSON editor for engine config
// Reads config from API, allows editing, saves back

export function ConfigEditor({ engineId }: { engineId: string }) {
  const [config, setConfig] = useState<Record<string, unknown>>()

  useEffect(() => {
    fetch(`/api/kernel/config/${engineId}`).then(r => r.json()).then(setConfig)
  }, [engineId])

  const handleSave = async (newConfig: Record<string, unknown>) => {
    await fetch(`/api/kernel/config/${engineId}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(newConfig)
    })
  }

  return (
    <div className="config-editor">
      <h2>Config: {engineId}</h2>
      <JsonEditor value={config} onChange={handleSave} />
    </div>
  )
}
```
