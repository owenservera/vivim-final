# 0.1 — KernelRegistry

**Phase:** 0 · **Domain:** Kernel Core · **Status:** DRAFT

## Purpose

Central registry that every engine, store, capability, and route calls at construction time.
Provides the system's live topology map — what exists, what's wired, what's healthy.

## Interface

```typescript
export interface EngineDescriptor {
  id: string
  kind: 'engine' | 'store' | 'capability' | 'route' | 'surface'
  layer?: string
  dependencies: string[]
  status: 'registered' | 'wired' | 'running' | 'error' | 'stopped'
  config: Record<string, unknown>
  health?: HealthState
  metadata: Record<string, unknown>
  registeredAt: number
  updatedAt: number
}

export interface HealthState {
  status: 'healthy' | 'degraded' | 'unhealthy' | 'unknown'
  score?: number               // 0-100
  lastCheck: number
  details?: Record<string, unknown>
}

export interface SystemTopology {
  engines: EngineDescriptor[]
  stores: StoreDescriptor[]
  capabilities: CapabilityDescriptor[]
  routes: RouteDescriptor[]
  timestamp: number
}

export class KernelRegistry {
  registerEngine(desc: Omit<EngineDescriptor, 'registeredAt' | 'updatedAt'>): void
  registerStore(desc: StoreDescriptor): void
  registerCapability(desc: CapabilityDescriptor): void
  registerRoute(desc: RouteDescriptor): void

  describe(): SystemTopology
  listEngines(filter?: { layer?: string; kind?: string; status?: string }): EngineDescriptor[]
  getEngine(id: string): EngineDescriptor | null
  getDependencies(id: string): string[]
  dependentsOf(id: string): string[]   // engines that depend on this

  updateHealth(id: string, health: HealthState): void
  markWired(id: string): void
  markRunning(id: string): void
  markError(id: string, error: string): void
  markStopped(id: string): void

  // Events
  onRegister(callback: (desc: EngineDescriptor) => void): () => void
  onStatusChange(callback: (id: string, from: string, to: string) => void): () => void
}
```

## Store Contract

- `src/storage/contracts/kernel-store.ts`: `KernelStore`
  - `upsertEngine(desc)`: persist engine descriptor
  - `upsertStore(desc)`: persist store descriptor
  - `upsertCapability(desc)`: persist capability descriptor
  - `getEngine(id)`: load engine by id
  - `listEngines(filter?)`: list all engines
  - `saveTopology(snapshot)`: persist topology snapshot
  - `getLastTopology()`: load last topology snapshot
