# 15.4 — OracleEventStream

**Phase:** 15 · **Domain:** Kernel Oracle · **Status:** DRAFT

## Purpose

Real-time system state over WebSocket. Broadcasts oracle events (health changes,
issue detection, healing actions) to all connected clients.

## Interface

```typescript
export interface OracleEvent {
  kind: 'health-changed' | 'issue-detected' | 'issue-resolved' | 'heal-started' | 'heal-completed' | 'heal-failed' | 'topology-changed'
  engineId?: string
  data: unknown
  timestamp: number
}

export class OracleEventStream {
  subscribe(callback: (event: OracleEvent) => void): () => void
  getRecentEvents(limit?: number): OracleEvent[]

  startPeriodicScan(intervalMs?: number): void
  stopPeriodicScan(): void

  // Internal — called by oracle subsystems
  emit(event: Omit<OracleEvent, 'timestamp'>): void
}
```

## Design

### Event Types

```typescript
// Health changes
{ kind: 'health-changed', engineId: 'chatgpt', data: { from: 'healthy', to: 'degraded', score: 65 } }

// Issue detection
{ kind: 'issue-detected', engineId: 'chrome-governor', data: { issueId: '...', category: 'stub', severity: 'warning' } }

// Issue resolution
{ kind: 'issue-resolved', engineId: 'chrome-governor', data: { issueId: '...', resolution: 'auto-healed' } }

// Healing
{ kind: 'heal-started', engineId: 'fleet-supervisor', data: { actionId: '...', kind: 'restart-engine' } }
{ kind: 'heal-completed', engineId: 'fleet-supervisor', data: { actionId: '...', result: 'success' } }
{ kind: 'heal-failed', engineId: 'fleet-supervisor', data: { actionId: '...', error: 'timeout' } }

// Topology
{ kind: 'topology-changed', data: { added: ['new-engine'], removed: [], statusChanges: [...] } }
```

### Periodic Scan

```typescript
startPeriodicScan(intervalMs = 60_000) {
  this.scanTimer = setInterval(async () => {
    const issues = await this.diagnostic.scan()
    for (const issue of issues) {
      if (!this.knownIssues.has(issue.id)) {
        this.knownIssues.set(issue.id, issue)
        this.emit({ kind: 'issue-detected', engineId: issue.engineId, data: issue })
      }
    }
    // Check for resolved issues
    for (const [id, issue] of this.knownIssues) {
      if (!issues.find(i => i.id === id)) {
        this.emit({ kind: 'issue-resolved', engineId: issue.engineId, data: { issueId: id } })
        this.knownIssues.delete(id)
      }
    }
  }, intervalMs)
}
```
