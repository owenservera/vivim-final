# Unit 3.6 — Selector Healing: Auto-Detect + Repair Broken Selectors

**Phase:** 3 | **ID:** 3.6 | **Dependencies:** 3.2

## Problem

When a provider updates their UI, selectors break. The system should detect broken selectors, try alternative selectors, and update the DB with the working one — all automatically.

```typescript
// ConversationManager constructor:
constructor(
  private governor: ChromeGovernor,
  // ... existing deps ...
  private healer?: SelectorHealer,
) {}

// In send(), when composer selector fails:
async findWorkingComposer(slaveId, providerId): Promise<string | null> {
  const selectors = COMPOSER_SELECTORS[providerId] ?? ['textarea']
  
  for (const selector of selectors) {
    const found = await this.testSelector(slaveId, selector)
    if (found) {
      // Heal: update DB with working selector
      if (this.healer) {
        await this.healer.recordHit(providerId, 'send_message', selector)
      }
      return selector
    }
  }
  
  // All selectors failed — try DOM heuristic
  const heuristic = await this.findComposerHeuristic(slaveId)
  if (heuristic) {
    // Record as new selector
    if (this.healer) {
      await this.healer.proposeSelector(providerId, 'send_message', heuristic, 0.8)
    }
    return heuristic
  }
  
  return null
}

private async findComposerHeuristic(slaveId: string): Promise<string | null> {
  // Look for any input-like element that can accept text
  const result = await this.governor.cdp.send(slaveId, 'Runtime.evaluate', {
    expression: `(() => {
      // Try textarea
      const ta = document.querySelector('textarea');
      if (ta) return 'textarea';
      // Try contenteditable
      const ce = document.querySelector('[contenteditable="true"]');
      if (ce) {
        if (ce.id) return '#' + ce.id;
        if (ce.className) return '.' + ce.className.split(' ')[0];
        return '[contenteditable="true"]';
      }
      // Try role=textbox
      const tb = document.querySelector('[role="textbox"]');
      if (tb) return '[role="textbox"]';
      return null;
    })()`,
    returnByValue: true,
  })
  return result?.result?.value ?? null
}
```

### Selector Health Tracking

```typescript
// In provider_capability table:
// selector_hit_count: incremented when selector works
// selector_miss_count: incremented when selector fails
// selector_last_miss_at: timestamp of last miss
// selector_last_error: error message from last miss

// After each send:
if (selectorWorked) {
  await store.updateCapabilitySelector(providerId, 'send_message', {
    hitCount: { increment: 1 },
    lastError: null,
  })
} else {
  await store.updateCapabilitySelector(providerId, 'send_message', {
    missCount: { increment: 1 },
    lastMissAt: Date.now(),
    lastError: 'Selector not found',
  })
  // Emit drift event
  eventBus.emit({ type: 'capability:selector_drifted', providerId, capabilitySlug: 'send_message' })
}
```
