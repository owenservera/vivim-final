# Unit 8.1 — Idle Slave TTL + Configurable Eviction Policy

**Phase:** 8 | **ID:** 8.1 | **Dependencies:** 7.1

## Problem

Slaves stay running indefinitely once spawned. Chrome processes consume 200-500MB RAM each. With 3 providers × multiple accounts, memory usage grows unbounded.

## Re-Programmable Design

An **EvictionPolicy** controls when idle slaves are cleaned up:

```typescript
interface EvictionPolicy {
  enabled: boolean
  idleTtlMs: number                    // kill slave after this much idle time
  maxConcurrentSlaves: number          // hard limit on total slaves
  maxPerProvider: number               // limit per provider
  evictionStrategy: 'lru' | 'lfu' | 'fifo' | 'idle_first'
  checkIntervalMs: number              // how often to check for evictable slaves
  warmPoolSize: number                 // keep at least N slaves pre-warmed
  onEvict: 'kill' | 'suspend'          // 'suspend' = freeze process (if supported)
  minUptimeBeforeEvictMs: number       // don't evict freshly spawned slaves
}
```

### Eviction Manager

```typescript
// New file: src/engines/eviction-manager.ts

export class EvictionManager {
  private timer: ReturnType<typeof setInterval> | null = null
  private accessLog = new Map<string, { lastAccess: number; accessCount: number; createdAt: number }>()

  constructor(
    private governor: ChromeGovernor,
    private policyStore: PolicyStore,
    private eventBus: CapabilityEventBus,
  ) {
    // Track access patterns
    eventBus.on('capability:executed', (e) => {
      this.recordAccess(e.slaveId)
    })
  }

  start(): void {
    this.stop()
    void this.runLoop()
  }

  stop(): void {
    if (this.timer) clearInterval(this.timer)
    this.timer = null
  }

  private async runLoop(): Promise<void> {
    const policy = await this.policyStore.get<EvictionPolicy>('fleet.eviction')
    if (!policy?.enabled) {
      this.timer = setInterval(() => void this.runLoop(), 60_000)
      return
    }

    this.timer = setInterval(() => void this.check(policy), policy.checkIntervalMs ?? 60_000)
  }

  private async check(policy: EvictionPolicy): Promise<void> {
    const slaves = this.governor.getAllSlaves()
    const now = Date.now()

    // [1] Evict idle slaves past TTL
    if (policy.idleTtlMs > 0) {
      for (const slave of slaves) {
        const info = this.accessLog.get(slave.slaveId)
        const lastAccess = info?.lastAccess ?? slave.lastHealthCheck
        const idleMs = now - lastAccess

        // Check minimum uptime
        const uptime = now - (info?.createdAt ?? slave.lastHealthCheck)
        if (uptime < (policy.minUptimeBeforeEvictMs ?? 30_000)) continue

        if (idleMs > policy.idleTtlMs) {
          // Check warm pool
          if (slaves.length - 1 < policy.warmPoolSize) continue

          console.log(`[eviction] Evicting idle slave ${slave.slaveId} (idle ${idleMs}ms)`)
          await this.evict(slave.slaveId, policy)
        }
      }
    }

    // [2] Enforce max concurrent
    if (policy.maxConcurrentSlaves > 0) {
      const activeSlaves = this.governor.getAllSlaves()
        .filter(s => s.status === 'running')
      while (activeSlaves.length > policy.maxConcurrentSlaves) {
        const victim = this.selectVictim(activeSlaves, policy.evictionStrategy)
        if (!victim) break
        await this.evict(victim.slaveId, policy)
        activeSlaves.splice(activeSlaves.indexOf(victim), 1)
      }
    }

    // [3] Enforce max per provider
    if (policy.maxPerProvider > 0) {
      const providers = new Map<string, typeof slaves>()
      for (const s of slaves) {
        const arr = providers.get(s.providerId) ?? []
        arr.push(s)
        providers.set(s.providerId, arr)
      }
      for (const [providerId, providerSlaves] of providers) {
        while (providerSlaves.length > policy.maxPerProvider) {
          const victim = this.selectVictim(providerSlaves, policy.evictionStrategy)
          if (!victim) break
          await this.evict(victim.slaveId, policy)
          providerSlaves.splice(providerSlaves.indexOf(victim), 1)
        }
      }
    }
  }

  private selectVictim(
    slaves: ChromeSlave[],
    strategy: string,
  ): ChromeSlave | null {
    if (slaves.length === 0) return null

    switch (strategy) {
      case 'lru':
        return slaves.sort((a, b) => {
          const aAccess = this.accessLog.get(a.slaveId)?.lastAccess ?? a.lastHealthCheck
          const bAccess = this.accessLog.get(b.slaveId)?.lastAccess ?? b.lastHealthCheck
          return aAccess - bAccess
        })[0] ?? null

      case 'lfu':
        return slaves.sort((a, b) => {
          const aCount = this.accessLog.get(a.slaveId)?.accessCount ?? 0
          const bCount = this.accessLog.get(b.slaveId)?.accessCount ?? 0
          return aCount - bCount
        })[0] ?? null

      case 'fifo':
        return slaves.sort((a, b) => a.lastHealthCheck - b.lastHealthCheck)[0] ?? null

      case 'idle_first':
      default:
        return slaves.sort((a, b) => {
          const aIdle = Date.now() - (this.accessLog.get(a.slaveId)?.lastAccess ?? a.lastHealthCheck)
          const bIdle = Date.now() - (this.accessLog.get(b.slaveId)?.lastAccess ?? b.lastHealthCheck)
          return bIdle - aIdle
        })[0] ?? null
    }
  }

  private async evict(slaveId: string, policy: EvictionPolicy): Promise<void> {
    this.eventBus.emit({ type: 'fleet:evicting', slaveId, reason: 'policy' })
    await this.governor.kill(slaveId)
    this.accessLog.delete(slaveId)
  }

  recordAccess(slaveId: string): void {
    const info = this.accessLog.get(slaveId) ?? {
      lastAccess: Date.now(),
      accessCount: 0,
      createdAt: Date.now(),
    }
    info.lastAccess = Date.now()
    info.accessCount++
    this.accessLog.set(slaveId, info)
  }
}
```

## Test

```typescript
test('idle slave evicted after TTL', async () => {
  await policyStore.set('fleet.eviction', {
    enabled: true,
    idleTtlMs: 1000,
    maxConcurrentSlaves: 10,
    evictionStrategy: 'idle_first',
    checkIntervalMs: 500,
    warmPoolSize: 0,
    minUptimeBeforeEvictMs: 0,
    ...defaults,
  })

  const slave = await governor.spawn('chatgpt', 'work')
  await delay(2000)  // wait past TTL

  // Slave should be evicted
  expect(governor.getSlave(slave.slaveId)).toBeNull()
})

test('warm pool prevents eviction', async () => {
  await policyStore.set('fleet.eviction', {
    ...policy,
    warmPoolSize: 1,
  })

  const slave = await governor.spawn('chatgpt', 'work')
  await delay(2000)

  // Should still be alive — it's the warm pool
  expect(governor.getSlave(slave.slaveId)).not.toBeNull()
})
```
