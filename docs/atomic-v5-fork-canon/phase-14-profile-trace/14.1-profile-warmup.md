# Unit 14.1 — ProfileWarmupEngine: History/Cookie/Trust Building

**Phase:** 14 | **ID:** 14.1 | **Dependencies:** 11.2

## Problem

Fresh Chrome profiles with zero browsing history, zero cookies, and zero favicons are a strong bot signal. Anti-bot systems check profile age, cookie count, and history depth. A real user's browser has months of history, dozens of cookies, and cached favicons.

## Spec

```typescript
// src/engines/stealth/profile-warmup-engine.ts

export class ProfileWarmupModule implements StealthModule {
  name = 'profile_warmup'
  detectionVector = 'Fresh profile detection (empty history, zero cookies, no favicons)'
  description = 'Pre-populates browser history, cookies, and favicons before first real use to build trust'
  priority = 5 // runs early — before other modules interact with pages

  configSchema = z.object({
    historySites: z.array(z.object({
      url: z.string(),
      title: z.string().optional(),
      visitCount: z.number().default(1),
    })).optional(),
    cookieSites: z.array(z.object({
      domain: z.string(),
      cookies: z.array(z.object({
        name: z.string(),
        value: z.string(),
        path: z.string().default('/'),
        secure: z.boolean().default(true),
        sameSite: z.enum(['Strict', 'Lax', 'None']).default('Lax'),
      })),
    })).optional(),
    warmupDelayMs: z.number().default(2000),
    skipIfWarm: z.boolean().default(true), // skip if profile already has history
  })

  private defaultHistorySites = [
    { url: 'https://www.google.com', title: 'Google', visitCount: 3 },
    { url: 'https://www.google.com/search?q=hello', title: 'hello - Google Search', visitCount: 1 },
    { url: 'https://www.youtube.com', title: 'YouTube', visitCount: 2 },
    { url: 'https://github.com', title: 'GitHub: Let\'s build from here', visitCount: 1 },
    { url: 'https://news.ycombinator.com', title: 'Hacker News', visitCount: 1 },
    { url: 'https://stackoverflow.com', title: 'Stack Overflow', visitCount: 1 },
    { url: 'https://www.wikipedia.org', title: 'Wikipedia', visitCount: 1 },
  ]

  async apply(config: Record<string, unknown>, ctx: StealthContext): Promise<void> {
    const cfg = this.configSchema.parse(config)
    const historySites = cfg.historySites ?? this.defaultHistorySites

    // Check if profile is already warm
    if (cfg.skipIfWarm) {
      const isWarm = await this.checkProfileWarmth(ctx)
      if (isWarm) {
        ctx.logger?.info('Profile already warm, skipping warmup')
        return
      }
    }

    // ── 1. Inject browsing history via CDP ──────────────
    for (const site of historySites) {
      for (let i = 0; i < site.visitCount; i++) {
        await this.addHistoryEntry(ctx, site.url, site.title)
      }
    }

    // ── 2. Visit sites to generate real cookies + favicons ──
    for (const site of historySites.slice(0, 3)) {
      try {
        await ctx.cdp.send(ctx.slaveId, 'Page.navigate', { url: site.url })
        await Bun.sleep(cfg.warmupDelayMs)
        // Page visit generates real cookies, cached resources, favicons
      } catch {
        // Non-critical — continue with other sites
      }
    }

    // ── 3. Navigate to provider to pre-cache its resources ──
    // This will be done by the caller (FleetSupervisor) after warmup

    ctx.logger?.info('Profile warmup complete', {
      historyEntries: historySites.reduce((s, h) => s + h.visitCount, 0),
    })
  }

  private async addHistoryEntry(
    ctx: StealthContext,
    url: string,
    title?: string,
  ): Promise<void> {
    // Use CDP to add history entry
    // Page.navigate creates a real history entry
    // But we don't want to actually load the full page
    // Alternative: use Runtime.evaluate to manipulate history API

    // For a more realistic approach: navigate, wait briefly, then continue
    // This creates real entries in Chrome's internal history database
    try {
      await ctx.cdp.send(ctx.slaveId, 'Page.navigate', { url })
      // Wait for minimal page load
      await Bun.sleep(500)

      // Set title if provided
      if (title) {
        await ctx.cdp.send(ctx.slaveId, 'Runtime.evaluate', {
          expression: `document.title = ${JSON.stringify(title)}`,
        }).catch(() => {})
      }
    } catch {
      // Non-critical
    }
  }

  private async checkProfileWarmth(ctx: StealthContext): Promise<boolean> {
    // Check if the profile has existing cookies/history
    // by checking the cookie jar
    try {
      const result = await ctx.cdp.send(ctx.slaveId, 'Network.getCookies', {})
      const cookies = (result as any)?.cookies ?? []
      return cookies.length > 5 // arbitrary threshold
    } catch {
      return false
    }
  }

  async verify(ctx: StealthContext): Promise<boolean> {
    try {
      const result = await ctx.cdp.send(ctx.slaveId, 'Network.getCookies', {})
      const cookies = (result as any)?.cookies ?? []
      return cookies.length > 0
    } catch {
      return false
    }
  }
}
```

### Provider-Specific Warmup

Different providers may need different warmup strategies. The config can be per-provider:

```json
// seeds/stealth-profiles/warmup-chatgpt.json
{
  "historySites": [
    { "url": "https://chatgpt.com", "title": "ChatGPT", "visitCount": 2 },
    { "url": "https://openai.com", "title": "OpenAI", "visitCount": 1 },
    { "url": "https://www.google.com", "title": "Google", "visitCount": 3 },
    { "url": "https://www.youtube.com", "title": "YouTube", "visitCount": 1 }
  ],
  "warmupDelayMs": 3000,
  "skipIfWarm": true
}
```

### Warmup on First Spawn

```typescript
// FleetSupervisor.spawn(), after Chrome launches:
const profile = await launchProfileEngine.resolve(providerSlug)
if (profile.stealthProfileId) {
  const moduleProfile = stealthModuleEngine.getProfile(profile.stealthProfileId)
  const warmupModule = moduleProfile?.modules.find(m => m.name === 'profile_warmup')
  if (warmupModule?.enabled) {
    // Run warmup BEFORE navigating to provider
    // This ensures the profile has history when the provider first sees it
    await warmupEngine.apply(warmupModule.config, ctx)

    // Then navigate to provider
    await cdp.send(slaveId, 'Page.navigate', { url: providerUrl })
  }
}
```

```typescript
test('warmup adds cookies to profile', async () => {
  // Check cookies before
  const before = await ctx.cdp.send(slaveId, 'Network.getCookies', {})
  expect((before as any).cookies.length).toBe(0)

  await module.apply({}, ctx)

  // Check cookies after
  const after = await ctx.cdp.send(slaveId, 'Network.getCookies', {})
  expect((after as any).cookies.length).toBeGreaterThan(0)
})

test('skipIfWarm prevents redundant warmup', async () => {
  // Pre-populate cookies
  await ctx.cdp.send(slaveId, 'Network.setCookie', {
    name: 'test', value: '1', domain: 'example.com',
  })

  let navigated = false
  mockCdp.on('Page.navigate', () => { navigated = true })

  await module.apply({ skipIfWarm: true }, ctx)

  // Should not navigate because profile already warm
  expect(navigated).toBe(false)
})
```
