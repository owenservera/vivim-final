# Unit 12.2 — WebGlSpoofEngine: GPU Renderer + Vendor Spoofing

**Phase:** 12 | **ID:** 12.2 | **Dependencies:** 11.2

## Problem

`WebGLRenderingContext.getParameter(UNMASKED_RENDERER_WEBGL)` returns the GPU model string. Headless Chrome returns `"SwiftShader"` — an immediate bot signal. Real Chrome returns strings like `"ANGLE (NVIDIA, NVIDIA GeForce RTX 3080 Direct3D11 vs_5_0 ps_5_0)"`.

## Spec

```typescript
// src/engines/stealth/webgl-spoof-engine.ts

const REALISTIC_GPUS = [
  { vendor: 'Google Inc. (NVIDIA)', renderer: 'ANGLE (NVIDIA, NVIDIA GeForce RTX 3080 Direct3D11 vs_5_0 ps_5_0)' },
  { vendor: 'Google Inc. (NVIDIA)', renderer: 'ANGLE (NVIDIA, NVIDIA GeForce RTX 3060 Direct3D11 vs_5_0 ps_5_0)' },
  { vendor: 'Google Inc. (Intel)', renderer: 'ANGLE (Intel, Intel(R) UHD Graphics 630 Direct3D11 vs_5_0 ps_5_0)' },
  { vendor: 'Google Inc. (AMD)', renderer: 'ANGLE (AMD, AMD Radeon RX 6700 XT Direct3D11 vs_5_0 ps_5_0)' },
  { vendor: 'Google Inc. (Intel)', renderer: 'ANGLE (Intel, Intel(R) Iris(R) Xe Graphics Direct3D11 vs_5_0 ps_5_0)' },
]

export class WebGlSpoofModule implements StealthModule {
  name = 'webgl_spoof'
  detectionVector = 'WebGL renderer + vendor fingerprinting'
  description = 'Spoofs WebGL UNMASKED_RENDERER and UNMASKED_VENDOR to realistic GPU strings'
  priority = 11
  configSchema = z.object({
    renderer: z.string().default('auto'),  // 'auto' = pick from list, or explicit string
    vendor: z.string().default('auto'),
    seed: z.number().optional(),  // for deterministic selection
  })

  async apply(config: Record<string, unknown>, ctx: StealthContext): Promise<void> {
    const cfg = this.configSchema.parse(config)

    let gpu = REALISTIC_GPUS[0]!
    if (cfg.renderer === 'auto') {
      // Pick a random GPU (or seeded for consistency)
      const idx = cfg.seed
        ? cfg.seed % REALISTIC_GPUS.length
        : Math.floor(Math.random() * REALISTIC_GPUS.length)
      gpu = REALISTIC_GPUS[idx]!
    } else {
      gpu = { vendor: cfg.vendor !== 'auto' ? cfg.vendor : 'Google Inc. (NVIDIA)', renderer: cfg.renderer }
    }

    const script = `
      (function() {
        var vendor = ${JSON.stringify(gpu.vendor)};
        var renderer = ${JSON.stringify(gpu.renderer)};

        var UNMASKED_VENDOR = 37445;
        var UNMASKED_RENDERER = 37446;

        function patchGetParameter(proto) {
          if (!proto) return;
          var orig = proto.getParameter;
          proto.getParameter = function(param) {
            if (param === UNMASKED_VENDOR) return vendor;
            if (param === UNMASKED_RENDERER) return renderer;
            return orig.call(this, param);
          };
        }

        patchGetParameter(WebGLRenderingContext.prototype);
        if (typeof WebGL2RenderingContext !== 'undefined') {
          patchGetParameter(WebGL2RenderingContext.prototype);
        }

        // Also patch WebGLDebugRendererInfo for older detection methods
        var oldExtensionEnabled = WebGLRenderingContext.prototype.getExtension;
        WebGLRenderingContext.prototype.getExtension = function(name) {
          if (name === 'WEBGL_debug_renderer_info') {
            var ext = oldExtensionEnabled.call(this, name);
            if (ext) {
              return { UNMASKED_VENDOR_WEBGL: UNMASKED_VENDOR, UNMASKED_RENDERER_WEBGL: UNMASKED_RENDERER };
            }
          }
          return oldExtensionEnabled.call(this, name);
        };
      })();
    `

    await ctx.cdp.send(ctx.slaveId, 'Page.addScriptToEvaluateOnNewDocument', { source: script })
  }

  async verify(ctx: StealthContext): Promise<boolean> {
    const result = await ctx.cdp.send(ctx.slaveId, 'Runtime.evaluate', {
      expression: `(() => {
        const canvas = document.createElement('canvas');
        const gl = canvas.getContext('webgl');
        if (!gl) return false;
        const ext = gl.getExtension('WEBGL_debug_renderer_info');
        if (!ext) return false;
        const renderer = gl.getParameter(ext.UNMASKED_RENDERER_WEBGL);
        return renderer !== 'SwiftShader' && !renderer.includes('SwiftShader');
      })()`,
      returnByValue: true,
    })
    return (result as any)?.result?.value === true
  }
}
```

## Test

```typescript
test('WebGL renderer is not SwiftShader', async () => {
  await module.apply({ renderer: 'auto' }, ctx)
  const ok = await module.verify!(ctx)
  expect(ok).toBe(true)
})

test('explicit renderer string is used', async () => {
  await module.apply({ renderer: 'ANGLE (NVIDIA, GeForce RTX 4090)', vendor: 'Google Inc. (NVIDIA)' }, ctx)
  const result = await ctx.cdp.send(slaveId, 'Runtime.evaluate', {
    expression: `(() => {
      const gl = document.createElement('canvas').getContext('webgl');
      const ext = gl.getExtension('WEBGL_debug_renderer_info');
      return gl.getParameter(ext.UNMASKED_RENDERER_WEBGL);
    })()`,
    returnByValue: true,
  })
  expect((result as any).result.value).toContain('RTX 4090')
})
```
