# Unit 2.8 — Frontend: Render Single Response in Conversation Surface

**Phase:** 2 | **ID:** 2.8 | **Dependencies:** 2.7

## Problem

The sandbox frontend has a CapabilityHarness and CapabilityCatalog but no dedicated conversation surface. The user needs a chat-like interface where they can:
1. See the conversation history (user + assistant messages)
2. Type and send a new message
3. See the response rendered with proper formatting (markdown, code blocks, etc.)

## Spec

### New Component: ConversationSurface

```tsx
// web/sandbox/src/features/conversation-surface.tsx

import { useState, useEffect, useCallback } from 'react'
import { useCapabilityStore } from '../store/capability-store.js'

interface Message {
  id: string
  role: 'user' | 'assistant'
  content: string
  blocks?: ContentBlock[]
  createdAt: number
}

export function ConversationSurface({ conversationId }: { conversationId: string }) {
  const [messages, setMessages] = useState<Message[]>([])
  const [input, setInput] = useState('')
  const [sending, setSending] = useState(false)

  // Load message history
  useEffect(() => {
    fetch(`/api/conversations/${conversationId}/messages`)
      .then(r => r.json())
      .then(msgs => setMessages(msgs))
      .catch(console.error)
  }, [conversationId])

  // WebSocket for real-time updates
  useEffect(() => {
    const ws = new WebSocket('ws://localhost:9420/ws')
    ws.onopen = () => {
      ws.send(JSON.stringify({
        type: 'subscribe',
        entityType: 'conversation',
        entityId: conversationId,
      }))
    }
    ws.onmessage = (ev) => {
      const msg = JSON.parse(ev.data)
      if (msg.type === 'conversation:complete') {
        setMessages(prev => [...prev, msg.message])
      }
    }
    return () => ws.close()
  }, [conversationId])

  const handleSend = useCallback(async () => {
    if (!input.trim() || sending) return
    setSending(true)
    
    // Optimistic: immediately show user message
    const userMsg: Message = {
      id: `temp_${Date.now()}`,
      role: 'user',
      content: input,
      createdAt: Date.now(),
    }
    setMessages(prev => [...prev, userMsg])
    setInput('')

    try {
      const resp = await fetch(`/api/conversations/${conversationId}/send`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ message: userMsg.content }),
      })
      const result = await resp.json()
      
      if (result.ok) {
        // Assistant response will arrive via WebSocket
        // But also add directly as fallback
        if (result.text) {
          const assistantMsg: Message = {
            id: result.messageId,
            role: 'assistant',
            content: result.text,
            blocks: result.blocks,
            createdAt: Date.now(),
          }
          setMessages(prev => [...prev, assistantMsg])
        }
      } else {
        // Remove optimistic message on error
        setMessages(prev => prev.filter(m => m.id !== userMsg.id))
        console.error('Send failed:', result.error)
      }
    } finally {
      setSending(false)
    }
  }, [input, conversationId, sending])

  return (
    <div className="flex flex-col h-full">
      {/* Message list */}
      <div className="flex-1 overflow-y-auto px-4 py-2">
        {messages.map(msg => (
          <MessageBubble key={msg.id} message={msg} />
        ))}
      </div>
      
      {/* Input */}
      <div className="border-t p-3 flex gap-2">
        <textarea
          value={input}
          onChange={e => setInput(e.target.value)}
          onKeyDown={e => {
            if (e.key === 'Enter' && !e.shiftKey) {
              e.preventDefault()
              handleSend()
            }
          }}
          placeholder="Type a message..."
          className="flex-1 px-3 py-2 border rounded-lg resize-none"
          rows={1}
        />
        <button
          onClick={handleSend}
          disabled={!input.trim() || sending}
          className="px-4 py-2 bg-blue-600 text-white rounded-lg"
        >
          {sending ? '...' : 'Send'}
        </button>
      </div>
    </div>
  )
}

function MessageBubble({ message }: { message: Message }) {
  const isUser = message.role === 'user'
  return (
    <div className={`mb-3 ${isUser ? 'text-right' : ''}`}>
      <div
        className={`inline-block px-4 py-2 rounded-lg max-w-[80%] ${
          isUser ? 'bg-blue-600 text-white' : 'bg-gray-100'
        }`}
      >
        <div className="whitespace-pre-wrap text-left">{message.content}</div>
      </div>
    </div>
  )
}
```

### ContentBlock Renderer

For richer rendering (code blocks, thinking, artifacts):

```tsx
function renderBlocks(blocks: ContentBlock[]) {
  return blocks.map((block, i) => {
    switch (block.kind) {
      case 'text':
        return <p key={i}>{block.content}</p>
      case 'code':
        return <pre key={i}><code className="block bg-gray-800 text-gray-100 p-3 rounded">{block.content}</code></pre>
      case 'thinking':
        return <details key={i} className="text-gray-500 italic"><summary>Thinking</summary>{block.content}</details>
      default:
        return null
    }
  })
}
```

### Integration with SandboxApp

Update `sandbox-app.tsx` to include the conversation surface:

```tsx
export function SandboxApp() {
  const [conversationId, setConversationId] = useState<string | null>(null)
  
  // Create conversation on mount if none exists
  useEffect(() => {
    fetch('/api/conversations', {
      method: 'POST',
      body: JSON.stringify({ providerId: 'chatgpt' })
    })
      .then(r => r.json())
      .then(conv => setConversationId(conv.id))
  }, [])

  return (
    <div className="flex h-screen">
      <ConversationSurface conversationId={conversationId} />
      <DebugPanel />
    </div>
  )
}
```
