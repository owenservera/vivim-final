# 16.2 — Kernel MCP Tools

**Phase:** 16 · **Domain:** Kernel Surfaces · **Status:** DRAFT

## Purpose

MCP tools that let the system query itself. Registered in the existing MCP server
so AI agents can ask questions about the system's state.

## Interface

```typescript
// Registered in src/mcp/server.ts — DiscoveryMcpServer
const kernelMcpTools = [
  {
    name: 'system.describe',
    description: 'Describe the full system topology — engines, stores, capabilities, routes, wiring',
    inputSchema: { type: 'object', properties: {} },
    handler: async () => {
      return kernel.registry.describe()
    }
  },
  {
    name: 'system.health',
    description: 'Get aggregate health snapshot of all engines',
    inputSchema: { type: 'object', properties: {} },
    handler: async () => {
      return await kernel.oracle.health()
    }
  },
  {
    name: 'system.diagnose',
    description: 'Scan for problems — stubs, broken wires, missing deps, health issues',
    inputSchema: {
      type: 'object',
      properties: {
        engineId: { type: 'string', description: 'Optional: scan specific engine' }
      }
    },
    handler: async (args: { engineId?: string }) => {
      if (args.engineId) {
        return await kernel.oracle.diagnoseEngine(args.engineId)
      }
      return await kernel.oracle.diagnose()
    }
  },
  {
    name: 'system.explain',
    description: 'Explain why something happened — trace causal chain for a trace ID',
    inputSchema: {
      type: 'object',
      properties: {
        target: { type: 'string', description: 'Trace ID or engine ID to explain' }
      },
      required: ['target']
    },
    handler: async (args: { target: string }) => {
      return await kernel.oracle.explain(args.target)
    }
  },
  {
    name: 'system.heal',
    description: 'Trigger corrective action for a detected problem',
    inputSchema: {
      type: 'object',
      properties: {
        issueId: { type: 'string', description: 'Diagnostic issue ID to heal' }
      },
      required: ['issueId']
    },
    handler: async (args: { issueId: string }) => {
      return await kernel.oracle.heal(args.issueId)
    }
  },
  {
    name: 'system.config',
    description: 'Get or set engine configuration',
    inputSchema: {
      type: 'object',
      properties: {
        engineId: { type: 'string', description: 'Engine to configure' },
        config: { type: 'object', description: 'New config (omit to read)' }
      },
      required: ['engineId']
    },
    handler: async (args: { engineId: string; config?: Record<string, unknown> }) => {
      if (args.config) {
        await kernel.config.updateScoped(args.engineId, args.config)
        return { ok: true, engineId: args.engineId }
      }
      return kernel.config.readScoped(args.engineId)
    }
  },
  {
    name: 'system.trace',
    description: 'Get trace spans for a specific trace ID',
    inputSchema: {
      type: 'object',
      properties: {
        traceId: { type: 'string', description: 'Trace ID to look up' }
      },
      required: ['traceId']
    },
    handler: async (args: { traceId: string }) => {
      return kernel.tracer.getTrace(args.traceId)
    }
  }
]
```

## Registration

```typescript
// src/mcp/server.ts — add to DiscoveryMcpServer constructor:
for (const tool of kernelMcpTools) {
  this.registerTool(tool)
}
```
