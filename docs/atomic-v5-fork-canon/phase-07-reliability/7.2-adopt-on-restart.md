# Unit 7.2 — PortReaper Adopt-on-Restart: Reconnect Instead of Kill

**Phase:** 7 | **ID:** 7.2 | **Dependencies:** 7.1

## Problem

`FleetSupervisor.boot()` calls `portReaper.reap()` which **kills** any Chrome process found on ports in the configured range. This destroys valid Chrome sessions that survived a server restart.

## Re-Programmable Design

An **AdoptionPolicy** controls what happens to orphaned Chrome instances:

```typescript
interface AdoptionPolicy {
  strategy: 'adopt' | 'kill' | 'ask'  // 'ask' = emit event, let handler decide
  validateProfile: boolean             // check if profile dir exists before adopting
  validateLogin: boolean               // check if session is still logged in
  maxAdoptAge: number                  // don't adopt instances older than N ms
}
```

### Implementation

```typescript
// New method on PortReaper:
async findAdoptable(
  portRange: [number, number],
  policy: AdoptionPolicy,
): Promise<AdoptableInstance[]> {
  const orphans = await this.findOrphans(portRange)
  const adoptable: AdoptableInstance[] = []

  for (const orphan of orphans) {
    const cmd = orphan.cmd
    // Parse --user-data-dir from command line
    const profileDirMatch = cmd.match(/--user-data-dir=([^\s]+)/)
    const portMatch = cmd.match(/--remote-debugging-port=(\d+)/)
    const profileDir = profileDirMatch?.[1]
    const port = Number(portMatch?.[1])

    if (!profileDir || !port) continue

    // Validate profile dir exists
    if (policy.validateProfile && !await fileExists(profileDir)) continue

    // Check age
    const ageMs = await getProcessAge(orphan.pid)
    if (policy.maxAdoptAge && ageMs > policy.maxAdoptAge) continue

    adoptable.push({
      pid: orphan.pid,
      port,
      profileDir,
      cmd,
      ageMs,
    })
  }

  return adoptable
}

// FleetSupervisor.boot():
async boot(): Promise<void> {
  const policy = await this.policyStore.get<AdoptionPolicy>('fleet.adoption')

  switch (policy?.strategy ?? 'adopt') {
    case 'adopt': {
      const adoptable = await this.portReaper.findAdoptable(this.opts.portRange, policy)
      for (const instance of adoptable) {
        await this.adoptInstance(instance)
      }
      // Kill remaining orphans that weren't adoptable
      await this.portReaper.reap(this.opts.portRange)
      break
    }
    case 'kill': {
      await this.portReaper.reap(this.opts.portRange)
      break
    }
    case 'ask': {
      const adoptable = await this.portReaper.findAdoptable(this.opts.portRange, policy)
      this.eventBus.emit({ type: 'fleet:adoptable_found', instances: adoptable })
      // Wait for handler to decide
      break
    }
  }

  if (this.opts.autoRestart) {
    this.startHealthProbe()
  }
}

private async adoptInstance(info: AdoptableInstance): Promise<void> {
  // Verify CDP is responsive
  try {
    const cdp = new BunCdpClient(`ws://127.0.0.1:${info.port}/devtools/browser`)
    await cdp.connect()
    await cdp.send('Browser.getVersion')
    await cdp.disconnect()
  } catch {
    // Chrome alive but CDP not responding — kill it
    await this.portReaper.reapProcess(info.pid)
    return
  }

  // Parse provider/account from profile dir path
  // Path pattern: {base}/{provider}/{account}
  const parts = info.profileDir.replace(/\\/g, '/').split('/')
  const account = parts[parts.length - 1] || 'unknown'
  const provider = parts[parts.length - 2] || 'unknown'

  const id = `${provider}_${account}_adopted_${Date.now()}`

  this.instances.set(id, {
    id,
    providerSlug: provider,
    accountId: account,
    debugPort: info.port,
    profileDir: info.profileDir,
    status: 'running',
    pid: info.pid,
    consecutiveFailures: 0,
    lastHealthCheck: Date.now(),
    createdAt: Date.now() - info.ageMs,
  })

  this.portReaper.trackPid(info.port, info.pid)
  console.log(`[fleet] Adopted Chrome pid=${info.pid} port=${info.port} profile=${info.profileDir}`)
}
```

## Test

```typescript
test('adopt-on-restart preserves Chrome session', async () => {
  // Spawn Chrome via setup
  await launchChrome({ visible: true, debugPort: 9301, profileDir: '/tmp/chatgpt/work' })

  // Simulate server restart — create new supervisor
  const newSupervisor = new FleetSupervisor(store, {
    portRange: [9300, 9400],
    adoptionStrategy: 'adopt',
    ...opts,
  })
  await newSupervisor.boot()

  // Chrome should be adopted, not killed
  const instances = newSupervisor.getAllInstances()
  expect(instances.length).toBe(1)
  expect(instances[0].debugPort).toBe(9301)

  // Verify Chrome is still alive
  expect(await isChromeRunning(instances[0].pid!)).toBe(true)
})
```
