# Unit 9.4 — Audit Trail for All User + System Actions

**Phase:** 9 | **ID:** 9.4 | **Dependencies:** 9.1

## Problem

No audit trail. You can't answer "who did what, when?" for compliance, debugging, or analytics. Actions happen but aren't recorded in a queryable format.

## Re-Programmable Design

An **AuditSink** plugin system (same pattern as logging/metrics):

```typescript
interface AuditSink {
  name: string
  record(entry: AuditEntry): Promise<void>
  query?(filter: AuditFilter): Promise<AuditEntry[]>
}

interface AuditEntry {
  id: string
  ts: number
  actor: string           // 'user' | 'system' | 'agent'
  actorId?: string        // user ID, session ID, or agent ID
  action: string          // 'conversation.send' | 'fleet.spawn' | 'provider.login'
  targetType: string      // 'conversation' | 'slave' | 'provider' | 'config'
  targetId?: string
  result: 'success' | 'failure' | 'denied'
  durationMs?: number
  details: Record<string, unknown>
  ipAddress?: string
  userAgent?: string
}

interface AuditPolicy {
  enabled: boolean
  sinks: Array<{ name: string; enabled: boolean; config: Record<string, unknown> }>
  actions: string[]       // which actions to audit ('*' = all)
  redactFields: string[]  // fields to redact in details (e.g. 'message')
  retentionDays: number
}
```

### Audit Engine

```typescript
// New file: src/engines/audit-trail.ts

export class AuditTrail {
  private sinks: AuditSink[] = []
  private policy: AuditPolicy

  constructor(policy: AuditPolicy, private logger: StructuredLogger) {
    this.policy = policy
    this.loadSinks()
  }

  record(entry: Omit<AuditEntry, 'id' | 'ts'>): void {
    if (!this.policy.enabled) return

    // Check if this action should be audited
    if (!this.policy.actions.includes('*') && !this.policy.actions.includes(entry.action)) {
      return
    }

    const full: AuditEntry = {
      ...entry,
      id: `audit_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
      ts: Date.now(),
      details: this.redact(entry.details),
    }

    // Write to all enabled sinks
    for (const sink of this.sinks) {
      sink.record(full).catch(err => {
        this.logger.error('Audit sink failed', err instanceof Error ? err : undefined, { sink: sink.name })
      })
    }
  }

  // Convenience methods
  recordUserAction(action: string, actorId: string, details: Record<string, unknown>, result: 'success' | 'failure' = 'success'): void {
    this.record({ actor: 'user', actorId, action, targetType: 'unknown', result, details })
  }

  recordSystemAction(action: string, details: Record<string, unknown>, result: 'success' | 'failure' = 'success'): void {
    this.record({ actor: 'system', action, targetType: 'unknown', result, details })
  }

  async query(filter: AuditFilter): Promise<AuditEntry[]> {
    // Query from first sink that supports query
    for (const sink of this.sinks) {
      if (sink.query) {
        return sink.query(filter)
      }
    }
    return []
  }

  private redact(details: Record<string, unknown>): Record<string, unknown> {
    const out = { ...details }
    for (const field of this.policy.redactFields) {
      if (field in out) out[field] = '[REDACTED]'
    }
    return out
  }
}
```

### Auto-Instrumentation

Wire audit trail into all action entry points:

```typescript
// In ConversationManager.send():
const auditTimer = Date.now()
try {
  const result = await this.sendInternal(...)
  this.audit.record({
    actor: 'user', actorId: 'session_xxx',
    action: 'conversation.send',
    targetType: 'conversation', targetId: conversationId,
    result: 'success', durationMs: Date.now() - auditTimer,
    details: { providerId: conv.providerId, messageLength: message.length },
  })
  return result
} catch (err) {
  this.audit.record({
    actor: 'user', actorId: 'session_xxx',
    action: 'conversation.send',
    targetType: 'conversation', targetId: conversationId,
    result: 'failure', durationMs: Date.now() - auditTimer,
    details: { error: String(err) },
  })
  throw err
}
```
