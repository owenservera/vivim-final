# Unit 9.3 — Error Tracking Integration

**Phase:** 9 | **ID:** 9.3 | **Dependencies:** 9.1

## Problem

Errors are logged but not tracked. No error grouping, no deduplication, no release tracking, no alerting thresholds. The same error can fire 1000 times and fill up logs.

## Re-Programmable Design

An **ErrorReporter** plugin system:

```typescript
interface ErrorReporter {
  name: string
  report(error: TrackedError): Promise<void>
}

interface TrackedError {
  id: string               // dedup hash
  name: string
  message: string
  stack?: string
  level: 'error' | 'fatal'
  engine?: string
  context: Record<string, unknown>
  firstSeen: number
  lastSeen: number
  count: number
}

interface ErrorTrackingPolicy {
  enabled: boolean
  reporters: Array<{ name: string; enabled: boolean; config: Record<string, unknown> }>
  dedupWindowMs: number    // group identical errors within this window
  maxContextSize: number   // truncate context to prevent memory bloat
  sampleRate: number       // 0-1, for high-volume errors
  ignorePatterns: string[] // error messages to ignore
}
```

### Implementation

```typescript
// New file: src/engines/error-tracker.ts

export class ErrorTracker {
  private reporters: ErrorReporter[] = []
  private seen = new Map<string, { error: TrackedError; lastFlush: number }>()
  private policy: ErrorTrackingPolicy

  constructor(
    policy: ErrorTrackingPolicy,
    private logger: StructuredLogger,
  ) {
    this.policy = policy
    this.loadReporters()
    // Periodic flush
    setInterval(() => void this.flush(), 5000)
  }

  capture(error: Error, context: Record<string, unknown> = {}): void {
    if (!this.policy.enabled) return

    // Check ignore patterns
    for (const pattern of this.policy.ignorePatterns) {
      if (error.message.includes(pattern)) return
    }

    // Sampling for high-volume errors
    if (this.policy.sampleRate < 1 && Math.random() > this.policy.sampleRate) return

    const id = this.dedupHash(error, context)
    const now = Date.now()

    const existing = this.seen.get(id)
    if (existing) {
      // Dedup: increment count, update last seen
      existing.error.count++
      existing.error.lastSeen = now
      // Don't re-report within window
      if (now - existing.lastFlush < this.policy.dedupWindowMs) return
    } else {
      const tracked: TrackedError = {
        id,
        name: error.name,
        message: error.message,
        stack: error.stack,
        level: 'error',
        engine: context.engine as string | undefined,
        context: this.truncateContext(context),
        firstSeen: now,
        lastSeen: now,
        count: 1,
      }
      this.seen.set(id, { error: tracked, lastFlush: now })
    }

    // Report
    const entry = this.seen.get(id)!
    entry.lastFlush = now
    for (const reporter of this.reporters) {
      reporter.report(entry.error).catch(() => {})
    }

    // Also log via structured logger
    this.logger.error(error.message, error, context)
  }

  private dedupHash(error: Error, context: Record<string, unknown>): string {
    // Hash based on error name + first line of stack + engine
    const firstStackLine = error.stack?.split('\n')[1]?.trim() ?? ''
    const engine = context.engine ?? ''
    return hash(`${error.name}:${firstStackLine}:${engine}`)
  }

  private truncateContext(context: Record<string, unknown>): Record<string, unknown> {
    const max = this.policy.maxContextSize ?? 4096
    const json = JSON.stringify(context)
    if (json.length <= max) return context
    return { truncated: true, preview: json.slice(0, max), originalSize: json.length }
  }

  private async flush(): Promise<void> {
    // Send aggregated counts to reporters
    for (const [, entry] of this.seen) {
      if (entry.error.count > 1) {
        for (const reporter of this.reporters) {
          if (reporter.updateCount) {
            await reporter.updateCount(entry.error.id, entry.error.count)
          }
        }
        entry.error.count = 0 // reset after flush
      }
    }
  }
}

// Built-in reporters
class SentryReporter implements ErrorReporter {
  name = 'sentry'
  constructor(private config: { dsn: string; environment: string; release?: string }) {}
  async report(error: TrackedError): Promise<void> {
    // POST to Sentry API
  }
}

class ConsoleErrorReporter implements ErrorReporter {
  name = 'console'
  async report(error: TrackedError): Promise<void> {
    console.error(`[${error.level}] ${error.name}: ${error.message}`, error.context)
  }
}
```

test('identical errors are deduped', () => {
  const reporter = { name: 'mock', report: vi.fn(async () => {}) }
  tracker.reporters = [reporter]

  const error = new Error('Test error')
  tracker.capture(error, { engine: 'test' })
  tracker.capture(error, { engine: 'test' }) // duplicate
  tracker.capture(error, { engine: 'test' }) // duplicate

  // Should only report once within window
  expect(reporter.report).toHaveBeenCalledTimes(1)
})

test('ignored patterns are skipped', () => {
  tracker.capture(new Error('Chrome not found'), {})
  tracker.capture(new Error('ECONNREFUSED'), {})
  // If 'ECONNREFUSED' is in ignorePatterns, it's skipped
})
