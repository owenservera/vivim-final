# Unit 9.1 — Structured Logging with Pluggable Transports

**Phase:** 9 | **ID:** 9.1 | **Dependencies:** none

## Problem

All logging is `console.log` / `console.error` — unstructured, unsearchable, no log levels, no correlation IDs, no pluggable destinations.

## Re-Programmable Design

A **LogTransport** plugin system where each transport is independently configurable:

```typescript
interface LogTransport {
  name: string
  write(entry: StructuredLog): Promise<void>
  flush?(): Promise<void>
}

interface StructuredLog {
  ts: number
  level: 'trace' | 'debug' | 'info' | 'warn' | 'error' | 'fatal'
  msg: string
  engine?: string
  conversationId?: string
  slaveId?: string
  providerId?: string
  traceId?: string
  durationMs?: number
  data?: Record<string, unknown>
  error?: { message: string; stack?: string; name: string }
}

interface LoggingPolicy {
  minLevel: 'trace' | 'debug' | 'info' | 'warn' | 'error' | 'fatal'
  transports: TransportConfig[]
  correlationHeader: string  // header name for trace correlation
  redactPaths: string[]      // fields to redact (e.g. ['data.token', 'data.password'])
}

interface TransportConfig {
  name: string         // 'console' | 'file' | 'loki' | 'datadog' | 'otlp'
  enabled: boolean
  level?: string       // override minLevel for this transport
  config: Record<string, unknown>  // transport-specific config
}
```

### Logger Implementation

```typescript
// New file: src/engines/logger.ts

export class StructuredLogger {
  private transports: LogTransport[] = []
  private minLevel: number
  private redactPaths: Set<string>

  private static LEVEL_RANK: Record<string, number> = {
    trace: 0, debug: 1, info: 2, warn: 3, error: 4, fatal: 5,
  }

  constructor(private policy: LoggingPolicy) {
    this.minLevel = this.LEVEL_RANK[policy.minLevel] ?? 2
    this.redactPaths = new Set(policy.redactPaths ?? [])
    this.loadTransports()
  }

  private loadTransports(): void {
    for (const cfg of this.policy.transports) {
      if (!cfg.enabled) continue
      const transport = this.createTransport(cfg.name, cfg.config)
      if (transport) this.transports.push(transport)
    }
  }

  private createTransport(name: string, config: Record<string, unknown>): LogTransport | null {
    switch (name) {
      case 'console':
        return new ConsoleTransport(config)
      case 'file':
        return new FileTransport(config)
      case 'loki':
        return new LokiTransport(config)
      case 'datadog':
        return new DatadogTransport(config)
      case 'otlp':
        return new OTLPTransport(config)
      default:
        // Check plugin registry
        return LogTransportRegistry.get(name)?.(config) ?? null
    }
  }

  log(entry: Partial<StructuredLog>): void {
    const level = entry.level ?? 'info'
    if (this.LEVEL_RANK[level] < this.minLevel) return

    const full: StructuredLog = {
      ts: Date.now(),
      level,
      msg: entry.msg ?? '',
      ...entry,
      data: entry.data ? this.redact(entry.data) : undefined,
    }

    for (const transport of this.transports) {
      transport.write(full).catch(() => {}) // fire-and-forget
    }
  }

  // Convenience methods
  info(msg: string, data?: Record<string, unknown>): void {
    this.log({ level: 'info', msg, data })
  }
  warn(msg: string, data?: Record<string, unknown>): void {
    this.log({ level: 'warn', msg, data })
  }
  error(msg: string, error?: Error, data?: Record<string, unknown>): void {
    this.log({ level: 'error', msg, error: error ? {
      message: error.message, stack: error.stack, name: error.name,
    } : undefined, data })
  }

  private redact(data: Record<string, unknown>): Record<string, unknown> {
    const out = { ...data }
    for (const path of this.redactPaths) {
      const parts = path.split('.')
      let obj: any = out
      for (let i = 0; i < parts.length - 1; i++) {
        obj = obj?.[parts[i]]
        if (!obj) break
      }
      if (obj && typeof obj === 'object') {
        obj[parts[parts.length - 1]] = '[REDACTED]'
      }
    }
    return out
  }

  private LEVEL_RANK = StructuredLogger.LEVEL_RANK
}
```

### Built-in Transports

```typescript
class ConsoleTransport implements LogTransport {
  name = 'console'
  constructor(private config: { colorize?: boolean; format?: 'json' | 'pretty' }) {}

  async write(entry: StructuredLog): Promise<void> {
    const format = this.config.format ?? 'json'
    if (format === 'json') {
      console.log(JSON.stringify(entry))
    } else {
      const color = this.config.colorize ? this.colorFor(entry.level) : ''
      const reset = this.config.colorize ? '\x1b[0m' : ''
      console.log(`${color}[${entry.level.toUpperCase()}]${reset} ${entry.ts} ${entry.msg}`)
    }
  }
}
```

### Plugin Registry (for custom transports)

```typescript
// Transport plugins registered at runtime
const LogTransportRegistry = new Map<string, (config: Record<string, unknown>) => LogTransport>()

export function registerLogTransport(name: string, factory: (config: Record<string, unknown>) => LogTransport): void {
  LogTransportRegistry.set(name, factory)
}
```

## Test

```typescript
test('logger writes to all enabled transports', async () => {
  const mockTransport = { name: 'mock', write: vi.fn(async () => {}) }
  const logger = new StructuredLogger({
    minLevel: 'info',
    transports: [{ name: 'mock', enabled: true, config: {} }],
    correlationHeader: 'X-Trace-Id',
    redactPaths: ['data.token'],
  })
  logger.transports = [mockTransport]

  logger.info('Test message', { token: 'secret', data: 'public' })

  expect(mockTransport.write).toHaveBeenCalledWith(
    expect.objectContaining({
      msg: 'Test message',
      data: { token: '[REDACTED]', data: 'public' },
    })
  )
})
```
