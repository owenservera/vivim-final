# Unit 20.8: Schema + Store: Sovereign Tables

**Phase:** 20 | **File:** `prisma/schema.prisma` + `src/storage/impl/`
**Depends:** — | **Source:** upgrade-03-schema.md §Phase 20

## Purpose
Add 2 new tables for sovereign data: `SyncLog` and `SyncPeer`. Implement store contract `SyncStore` used by SyncEngine (20.6).

## Models (from upgrade-03-schema.md)

### SyncLog
```prisma
model SyncLog {
  id         String  @id
  deviceId   String  @map("device_id")
  table      String
  recordId   String  @map("record_id")
  operation  String
  dataJson   String  @map("data_json")
  ts         Int
  syncedAt   Int?    @map("synced_at")

  @@index([deviceId, syncedAt], map: "idx_sl_device")
  @@index([table, recordId], map: "idx_sl_record")
  @@map("sync_log")
}
```

### SyncPeer
```prisma
model SyncPeer {
  id           String  @id
  deviceId     String  @map("device_id")
  name         String
  publicKey    String  @map("public_key")
  lastSyncAt   Int?    @map("last_sync_at")
  status       String  @default("pending")
  pairedAt     Int?    @map("paired_at")

  @@unique([deviceId])
  @@map("sync_peer")
}
```

## Store Contract (SyncStore)
```typescript
export interface SyncStore {
  createLogEntry(entry: {
    id: string
    deviceId: string
    table: string
    recordId: string
    operation: string
    dataJson: string
    ts: number
    syncedAt: number | null
  }): Promise<void>

  getUnsyncedEntries(deviceId: string, limit?: number): Promise<Array<{
    id: string
    deviceId: string
    table: string
    recordId: string
    operation: string
    dataJson: string
    ts: number
    syncedAt: number | null
  }>>

  markSynced(ids: string[]): Promise<void>

  createPeer(peer: {
    id: string
    deviceId: string
    name: string
    publicKey: string
    lastSyncAt: number | null
    status: string
    pairedAt: number | null
  }): Promise<void>

  updatePeer(deviceId: string, patch: Partial<{
    name: string
    publicKey: string
    lastSyncAt: number
    status: string
    pairedAt: number
  }>): Promise<void>

  getPeers(): Promise<Array<{
    id: string
    deviceId: string
    name: string
    publicKey: string
    lastSyncAt: number | null
    status: string
    pairedAt: number | null
  }>>

  getPeer(deviceId: string): Promise<{
    id: string
    deviceId: string
    name: string
    publicKey: string
    lastSyncAt: number | null
    status: string
    pairedAt: number | null
  } | null>
}
```

## Tests
- SyncLog.createLogEntry inserts row
- SyncLog.getUnsyncedEntries returns rows where syncedAt IS NULL
- SyncLog.markSynced updates syncedAt timestamp
- SyncPeer.createPeer inserts row
- SyncPeer.updatePeer updates status to 'paired'
- SyncPeer.getPeers returns all peers
- SyncPeer.getPeer by deviceId returns single peer
- Unique constraint on deviceId enforced
