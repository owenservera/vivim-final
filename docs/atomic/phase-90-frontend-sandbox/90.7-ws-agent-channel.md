## Interface

```typescript
// src/server/websocket.ts — additions
// Maintain sessionId → ws registry
const sessions = new Map<string, WebSocket>()

// Handle inbound messages:
ws.on('message', (raw) => {
  const msg = JSON.parse(raw)
  if (msg.type === 'hello' && msg.role === 'frontend') {
    sessions.set(msg.sessionId, ws)
  }
  if (msg.type === 'agent:command') {
    // Route to target sessionId's ws
    const target = sessions.get(msg.targetSessionId)
    if (target) target.send(JSON.stringify(msg))
  }
  if (msg.type === 'agent:discover') {
    // Forward to frontend, collect response
  }
  if (msg.type === 'agent:result') {
    // Forward back to agent's ws
  }
})
```
