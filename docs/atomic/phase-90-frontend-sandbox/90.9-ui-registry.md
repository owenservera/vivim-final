export interface CapabilityRenderer {
  slug: string

  bestPracticeNote?: string
}

export const CapabilityRegistry = {
  register(slug: string, renderer: CapabilityRenderer): void

  list(): CapabilityRenderer[]

## Store Contract

- Generic contract-driven renderer in `web/ui/src/`
- Proven bespoke renderers graduate from sandbox to `web/ui/registry/`
- Prod renderer reads ledger to pick bespoke vs generic

## Gate Criteria

- [ ] `CapabilityRegistry` exports register/get/list
- [ ] `docs/sandbox/PROMOTED.md` exists
- [ ] Generic renderer can render any capability from its 21-field contract
