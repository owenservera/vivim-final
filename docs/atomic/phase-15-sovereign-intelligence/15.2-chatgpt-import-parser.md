# Unit 15.2: ChatGPT Export Parser

**Phase:** 15 | **File:** `src/engines/parsers/chatgpt-import.ts`
**Depends:** 15.1 KnowledgeIngestionEngine | **Produces:** Parse ChatGPT conversations.json
**Source:** upgrade-05-memory-intelligence.md §Import Parser Specifications

## Purpose

Parse the ChatGPT export format (conversations.json) into normalized conversations that KnowledgeIngestionEngine can ingest.

## Interface

```typescript
export interface ParsedConversation {
  externalId: string
  title: string
  source: 'chatgpt'
  createdAt: number
  updatedAt: number
  messages: ParsedMessage[]
}

export interface ParsedMessage {
  externalId: string
  role: 'user' | 'assistant' | 'system'
  content: string
  model?: string
  createdAt: number
  parentExternalId?: string
}

export interface ChatGPTExportParser {
  parse(rawJson: string): ParsedConversation[]
}
```

## Implementation

```typescript
export class ChatGPTExportParserImpl implements ChatGPTExportParser {
  parse(rawJson: string): ParsedConversation[] {
    const data = JSON.parse(rawJson) as ChatGPTRawConversation[]
    return data.map(conv => this.parseConversation(conv))
  }

  private parseConversation(conv: ChatGPTRawConversation): ParsedConversation {
    // Flatten mapping tree into ordered message list
    const messages = Object.values(conv.mapping)
      .filter(node => node.message)
      .map(node => this.parseMessage(node))
      .sort((a, b) => a.createdAt - b.createdAt)

    return {
      externalId: conv.title + '_' + conv.create_time,
      title: conv.title,
      source: 'chatgpt',
      createdAt: conv.create_time * 1000,
      updatedAt: (conv.update_time ?? conv.create_time) * 1000,
      messages,
    }
  }

  private parseMessage(node: ChatGPTRawNode): ParsedMessage {
    const msg = node.message!
    const parts = msg.content?.parts ?? []
    const content = parts.filter(p => typeof p === 'string').join('\n')
    return {
      externalId: msg.id,
      role: msg.author?.role ?? 'user',
      content,
      model: msg.model,
      createdAt: msg.create_time ? msg.create_time * 1000 : Date.now(),
    }
  }
}
```
