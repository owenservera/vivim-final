## Purpose

Add REST API routes for knowledge ingestion, search, synthesis, and management.

## Routes

```typescript
export function createKnowledgeRouter(ctx: ServerContext) {
  return async (req: Request): Promise<Response> => {
    const url = new URL(req.url)
    const { pathname } = url

    // POST /api/knowledge/ingest
    if (pathname === '/api/knowledge/ingest' && req.method === 'POST') {
      const body = await req.json() as { source: string; filePath: string }
      const result = await ctx.knowledgeIngestion?.ingest({
        source: body.source, filePath: body.filePath,
        deduplicate: true, extractEntities: true, extractDecisions: true, generateEmbeddings: true,
      })
      return json(result)
    }

    // GET /api/knowledge/search?q=X
    if (pathname === '/api/knowledge/search' && req.method === 'GET') {
      const q = url.searchParams.get('q') ?? ''
      const results = await ctx.semanticSearch?.search({ text: q, limit: 20 })
      return json(results)
    }

    // POST /api/knowledge/synthesize
    if (pathname === '/api/knowledge/synthesize' && req.method === 'POST') {
      const body = await req.json() as { question: string }
      const result = await ctx.synthesizer?.synthesize({
        question: body.question, maxSources: 10, synthesisStyle: 'summary',
        scope: {},
      })
      return json(result)
    }

    // GET /api/knowledge/export
    if (pathname === '/api/knowledge/export' && req.method === 'GET') {
      const result = await ctx.exportEngine?.export({
        format: 'json', scope: 'full', outputPath: `export_${Date.now()}.json`,
        includeEmbeddings: true, includeChromeProfiles: false,
      })
      return json(result)
    }

    // GET /api/knowledge/entities?type=X
    // GET /api/knowledge/decisions?conversationId=X
    // GET /api/knowledge/topics
    // POST /api/knowledge/topics
    // GET /api/knowledge/jobs
    // GET /api/knowledge/jobs/:id

    return errorResponse('Not found', 'NotFoundError', 404)
  }
}
```
