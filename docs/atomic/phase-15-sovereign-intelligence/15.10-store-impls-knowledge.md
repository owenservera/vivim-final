## Pattern (same for all)

```typescript
import type { CapStoreDb } from '../db.js'

export class KnowledgeIngestionStoreImpl implements KnowledgeIngestionStore {
  constructor(private db: CapStoreDb) {}

  async createImportJob(job: { id: string; source: string; ... }): Promise<void> {
    await this.db.prisma.importJob.create({ data: job })
  }

  async findExistingConversation(source: string, externalId: string): Promise<string | null> {
    const row = await this.db.prisma.conversation.findFirst({
      where: { source, externalId },
      select: { id: true },
    })
    return row?.id ?? null
  }

  // ... all other methods ...
}
```
