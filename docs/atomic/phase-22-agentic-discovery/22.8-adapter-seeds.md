# 22.8 — Adapter Seed Modules

**Phase:** 22 — Agentic Discovery Tooling
**Depends On:** 22.7 (Enhanced CapabilityShapeRegistry)
**Effort:** Medium
**Status:** PENDING

---

## Purpose

Create adapter modules that transform shape-specific capabilities into the universal capability contract. Loaded by `CapabilityShapeRegistry.loadAdaptersFromDir()`.

---

## File Structure

```
seeds/adapters/
  ├── chat_app.adapter.ts
  ├── coding_ide.adapter.ts
  ├── search_engine.adapter.ts
  └── custom.adapter.ts
```

---

## Interface

```typescript
// src/engines/capability-shape-registry.ts (existing)

export interface CapabilityAdapter {
  shapeId: string
  toUniversal(
    shapeSpecific: Record<string, unknown>,
    shapeContext: CapabilityShape,
  ): Record<string, unknown>
  fromUniversal(
    action: Record<string, unknown>,
    shapeContext: CapabilityShape,
  ): Record<string, unknown>
  projectState(
    rawState: Record<string, unknown>,
    shapeContext: CapabilityShape,
  ): Record<string, unknown>
}
```

## Adapter Specifications

### chat_app.adapter.ts

Identity adapter — chat apps already use the universal capability contract. Minimal transformation.

```typescript
export default {
  shapeId: 'chat_app',
  toUniversal(cap) { return cap },  // identity
  fromUniversal(action) { return action },
  projectState(state) { return state },
}
```

### coding_ide.adapter.ts

Transforms IDE-specific capabilities (run_code, open_file) to universal contract.

```typescript
export default {
  shapeId: 'coding_ide',
  toUniversal(cap) {
    // Map IDE-specific capability shapes to universal
    if (cap.type === 'run_code') {
      return { ...cap, uiComponent: 'action_button', uiPosition: 'header' }
    }
    return cap
  },
  fromUniversal(action) {
    if (action.slug === 'send_message') {
      // Map universal send_message to IDE's "run in terminal"
      return { ...action, type: 'execute_in_terminal' }
    }
    return action
  },
  projectState(rawState) {
    // Extract IDE-specific state into universal mirror state
    return {
      composerValue: rawState.editorContent ?? '',
      activeFile: rawState.filePath ?? null,
      isRunning: rawState.isExecuting ?? false,
    }
  },
}
```

### search_engine.adapter.ts

Transforms search-specific capabilities (search, filter_results) to universal contract.

```typescript
export default {
  shapeId: 'search_engine',
  toUniversal(cap) {
    if (cap.type === 'search') {
      return { ...cap, uiComponent: 'text_input', uiPosition: 'composer' }
    }
    return cap
  },
  fromUniversal(action) {
    if (action.slug === 'send_message') {
      return { ...action, type: 'search_query' }
    }
    return action
  },
  projectState(rawState) {
    return {
      composerValue: rawState.searchQuery ?? '',
      resultCount: rawState.results?.length ?? 0,
    }
  },
}
```

### custom.adapter.ts

Generic fallback — tries to map any capability to universal without transformation.

```typescript
export default {
  shapeId: 'custom',
  toUniversal(cap) { return cap },
  fromUniversal(action) { return action },
  projectState(state) { return state },
}
```
