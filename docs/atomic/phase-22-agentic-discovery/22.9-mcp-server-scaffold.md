## Server Entry

```typescript
// src/mcp/server.ts

import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js'
import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js'
import { registerDiscoveryTools } from './discovery-tools.js'

export interface DiscoveryServerContext {
  discoveryEngine: ProviderDiscoveryEngine
  manifestInference: ManifestInferenceEngine
  shapeRegistry: CapabilityShapeRegistry
  providerRegistrar: ProviderRegistrar
}

export async function createDiscoveryMcpServer(
  ctx: DiscoveryServerContext,
): Promise<McpServer> {
  const server = new McpServer({
    name: 'vivim-discovery',
    version: '1.0.0',
  })

  registerDiscoveryTools(server, ctx)

  const transport = new StdioServerTransport()
  await server.connect(transport)

  return server
}
```

## Tool Registration Pattern

```typescript
// src/mcp/discovery-tools.ts

import { z } from 'zod'

export function registerDiscoveryTools(
  server: McpServer,
  ctx: DiscoveryServerContext,
) {
  // ── Session Management ─────────────────────────────────────

  server.tool(
    'discover_start',
    'Start a provider discovery session for a URL',
    {
      url: z.string().url(),
      shapeHint: z.string().optional(),
      providerNameHint: z.string().optional(),
    },
    async ({ url, shapeHint, providerNameHint }) => {
      const session = await ctx.discoveryEngine.createSession(url, {
        shapeHint,
        providerNameHint,
      })
      return {
        content: [{ type: 'text', text: JSON.stringify(session) }],
      }
    },
  )

  server.tool(
    'discover_get_session',
    'Get current state of a discovery session',
    { sessionId: z.string() },
    async ({ sessionId }) => {
      const session = await ctx.discoveryEngine.getSession(sessionId)
      if (!session) {
        return { content: [{ type: 'text', text: 'Session not found' }], isError: true }
      }
      return { content: [{ type: 'text', text: JSON.stringify(session) }] }
    },
  )

  // ... (23 more tools follow same pattern)
}
```
